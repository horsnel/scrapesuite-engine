     1→/**
     2→ * WhatsApp Message Scraper
     3→ *
     4→ * Handles extraction of WhatsApp data including:
     5→ * - Chat list enumeration
     6→ * - Message history scraping with pagination
     7→ * - Media file download (images, videos, audio, documents)
     8→ * - Group metadata and participant extraction
     9→ * - Contact list retrieval
    10→ * - Status/Stories scraping
    11→ * - Deleted message recovery
    12→ * - Search and filtering
    13→ *
    14→ * UPGRADED: Now uses REAL Playwright browser automation via the browser pool
    15→ * to navigate WhatsApp Web, query DOM elements, and extract actual data.
    16→ * Falls back gracefully when elements are not found or when the session
    17→ * requires authentication.
    18→ *
    19→ * WhatsApp's data access patterns:
    20→ * - WhatsApp Web uses a local IndexedDB cache that mirrors phone data
    21→ * - Messages are stored in encrypted form and decrypted locally
    22→ * - Media files are downloaded on-demand from WhatsApp's CDN
    23→ * - Group metadata is synced via the app-state mechanism
    24→ * - Search queries go through WhatsApp's server-side search API
    25→ */

   148→  // ─── Browser Context Management ───────────────────────────────────────

   150→  /**
   151→   * Acquire a browser page for scraping. Tries to reuse the session's
   152→   * existing page, or acquires a new one from the pool.
   153→   */
   154→  private async acquirePage(sessionId: string): Promise<{
   155→    page: Page;
   156→    context: import('playwright').BrowserContext;
   157→    lease: BrowserLease | null;
   158→    owned: boolean; // true if we own the lease and must release it
   159→  }> {
   160→    // First: try to reuse the session's existing browser page
   161→    const existingPage = whatsappSessionManager.getSessionPage(sessionId);
   162→    if (existingPage) {
   163→      // Also get the context from the session manager's active browser sessions
   164→      // We don't have direct access here, but the page is already set up
   165→      return { page: existingPage, context: existingPage.context(), lease: null, owned: false };
   166→    }
   167→
   168→    // Second: acquire a new browser from the pool
   169→    const lease = await browserPool.acquire(undefined, true);
   170→    const { context, page } = lease;
   171→
   172→    // Apply anti-detection
   173→    await whatsappWebEvader.applyToContext(context);
   174→
   175→    // Navigate to WhatsApp Web
   176→    await page.goto(WHATSAPP_WEB_URL, {
   177→      waitUntil: 'domcontentloaded',
   178→      timeout: 60000,
   179→    });
   180→
   181→    // Wait for app to load
   182→    try {
   183→      await page.waitForSelector(SELECTORS.appRoot, { timeout: 30000 });
   184→    } catch {
   185→      throw new Error('WhatsApp Web failed to load within timeout');
   186→    }
   187→
   188→    // Check if logged in
   189→    await new Promise((r) => setTimeout(r, 2000));
   190→    const sidePanel = await page.$(SELECTORS.sidePanel);
   191→    if (!sidePanel) {
   192→      throw new Error('WhatsApp Web is not logged in — cannot scrape without authentication');
   193→    }
   194→
   195→    return { page, context, lease, owned: true };
   196→  }
   197→
   198→  /**
   199→   * Release a browser lease if we own it.
   200→   */
   201→  private async releaseLease(lease: BrowserLease | null): Promise<void> {
   202→    if (lease) {
   203→      try {
   204→        await browserPool.release(lease);
   205→      } catch {}
   206→    }
   207→  }

   211→  /**
   212→   * Scrape the list of chats from WhatsApp Web.
   213→   * UPGRADED: Actually queries the WhatsApp Web DOM for chat list elements.
   214→   */
   215→  async scrapeChats(sessionId: string, limit?: number): Promise<WhatsAppScrapeResult> {
   216→    const maxChats = limit || this.config.maxChatsPerSession;
   217→    const errors: Array<{ item: string; error: string }> = [];
   218→    const chats: WhatsAppChat[] = [];
   219→
   220→    let lease: BrowserLease | null = null;
   221→
   222→    try {
   223→      const browserContext = await this.acquirePage(sessionId);
   224→      lease = browserContext.owned ? browserContext.lease : null;
   225→      const { page } = browserContext;
   226→
   227→      // Wait for the side panel to be present
   228→      await page.waitForSelector(SELECTORS.sidePanel, { timeout: 15000 });
   229→
   230→      // Scroll the chat list to load more chats
   231→      const chatListContainer = await page.$(SELECTORS.sidePanel);
   232→      if (chatListContainer) {
   233→        // Scroll down to load more chats
   234→        for (let i = 0; i < 5; i++) {
   235→          await chatListContainer.evaluate((el: Element) => {
   236→            el.scrollTop = el.scrollHeight;
   237→          });
   238→          await new Promise((r) => setTimeout(r, 500 + Math.random() * 500));
   239→        }
   240→        // Scroll back to top
   241→        await chatListContainer.evaluate((el: Element) => {
   242→          el.scrollTop = 0;
   243→        });
   244→        await new Promise((r) => setTimeout(r, 300));
   245→      }
   246→
   247→      // Extract chat list items from the DOM
   248→      const chatRows = await page.$$(SELECTORS.chatListRow);
   249→
   250→      const chatCount = Math.min(chatRows.length, maxChats);
   251→
   252→      for (let i = 0; i < chatCount; i++) {
   253→        try {
   254→          const chat = await this.extractChatFromElement(page, chatRows[i], i);
   255→          if (chat) {
   256→            chats.push(chat);
   257→          }
   258→        } catch (error) {
   259→          errors.push({ item: `chat_${i}`, error: String(error) });
   260→        }
   261→
   262→        // Respect scrape delay for rate limiting
   263→        if (i > 0 && i % this.config.batchSize === 0) {
   264→          await this.sleep(this.config.scrapeDelay);
   265→        }
   266→      }
   267→
   268→      // Sort by last message timestamp (most recent first)
   269→      chats.sort((a, b) => b.lastMessageTimestamp - a.lastMessageTimestamp);
   270→
   271→      this.stats.totalChatsScraped += chats.length;
   272→      this.stats.lastScrapeAt = Date.now();
   273→
   274→      return {
   275→        target: 'chats',
   276→        data: chats,
   277→        totalAvailable: chatRows.length,
   278→        scraped: chats.length,
   279→        hasMore: chatRows.length > maxChats,
   280→        cursor: chats.length > 0 ? chats[chats.length - 1].jid : undefined,
   281→        scrapedAt: Date.now(),
   282→        errors,
   283→      };
   284→    } catch (error) {
   285→      this.stats.failedScrapes++;
   286→
   287→      return {
   288→        target: 'chats',
   289→        data: chats,
   290→        totalAvailable: 0,
   291→        scraped: 0,
   292→        hasMore: false,
   293→        scrapedAt: Date.now(),
   294→        errors: [{ item: 'chats', error: String(error) }],
   295→      };
   296→    } finally {
   297→      await this.releaseLease(lease);
   298→    }
   299→  }

   303→  /**
   304→   * Scrape messages from a specific chat with pagination support.
   305→   * UPGRADED: Actually navigates to the chat and extracts messages from the DOM.
   306→   */
   307→  async scrapeMessages(
   308→    sessionId: string,
   309→    chatJid: string,
   310→    options?: Partial<WhatsAppScrapeOptions>
   311→  ): Promise<WhatsAppScrapeResult> {
   312→    const maxMessages = options?.limit || this.config.maxMessagesPerChat;
   313→    const errors: Array<{ item: string; error: string }> = [];
   314→    const messages: WhatsAppMessage[] = [];
   315→
   316→    let lease: BrowserLease | null = null;
   317→
   318→    try {
   319→      const browserContext = await this.acquirePage(sessionId);
   320→      lease = browserContext.owned ? browserContext.lease : null;
   321→      const { page } = browserContext;
   322→
   323→      // Navigate to the specific chat using the JID
   324→      await this.navigateToChat(page, chatJid);
   325→
   326→      // Wait for messages to load
   327→      await new Promise((r) => setTimeout(r, 1500));
   328→
   329→      // Scroll up to load older messages
   330→      const messagePanelEl = await page.$(SELECTORS.messagePanel);
   331→      if (messagePanelEl) {
   332→        // Scroll up to load more message history
   333→        let loadedCount = 0;
   334→        let noNewMessagesCount = 0;
   335→
   336→        while (loadedCount < maxMessages && noNewMessagesCount < 3) {
   337→          const previousCount = loadedCount;
   338→
   339→          // Scroll up in the message panel
   340→          await messagePanelEl.evaluate((el: Element) => {
   341→            el.scrollTop = 0;
   342→          });
   343→
   344→          await new Promise((r) => setTimeout(r, 800 + Math.random() * 400));
   345→
   346→          // Count current messages
   347→          const allMessages = await page.$$(`${SELECTORS.messageIn}, ${SELECTORS.messageOut}`);
   348→          loadedCount = allMessages.length;
   349→
   350→          if (loadedCount <= previousCount) {
   351→            noNewMessagesCount++;
   352→          } else {
   353→            noNewMessagesCount = 0;
   354→          }
   355→
   356→          if (loadedCount >= maxMessages) break;
   357→        }
   358→      }
   359→
   360→      // Extract messages from the DOM
   361→      const messageElements = await page.$$(`${SELECTORS.messageIn}, ${SELECTORS.messageOut}`);
   362→      const msgCount = Math.min(messageElements.length, maxMessages);
   363→
   364→      for (let i = 0; i < msgCount; i++) {
   365→        try {
   366→          const message = await this.extractMessageFromElement(
   367→            page,
   368→            messageElements[i],
   369→            chatJid,
   370→            i
   371→          );
   372→          if (message) {
   373→            messages.push(message);
   374→          }
   375→        } catch (error) {
   376→          errors.push({ item: `msg_${chatJid}_${i}`, error: String(error) });
   377→        }
   378→
   379→        // Rate-limit-aware delay between batches
   380→        if (i > 0 && i % this.config.batchSize === 0) {
   381→          await this.sleep(this.config.scrapeDelay);
   382→        }
   383→      }
   384→
   385→      // Sort by timestamp
   386→      messages.sort((a, b) => b.timestamp - a.timestamp);
   387→
   388→      this.stats.totalMessagesScraped += messages.length;
   389→      this.stats.averageMessagesPerChat = Math.round(
   390→        (this.stats.averageMessagesPerChat * Math.max(this.stats.totalChatsScraped - 1, 0) + messages.length) /
   391→          Math.max(this.stats.totalChatsScraped, 1)
   392→      );
   393→      this.stats.lastScrapeAt = Date.now();
   394→
   395→      return {
   396→        target: 'messages',
   397→        data: messages,
   398→        totalAvailable: messageElements.length,
   399→        scraped: messages.length,
   400→        hasMore: messageElements.length > maxMessages,
   401→        cursor: messages.length > 0 ? messages[messages.length - 1].messageId : undefined,
   402→        scrapedAt: Date.now(),
   403→        errors,
   404→      };
   405→    } catch (error) {
   406→      this.stats.failedScrapes++;
   407→
   408→      return {
   409→        target: 'messages',
   410→        data: messages,
   411→        totalAvailable: 0,
   412→        scraped: 0,
   413→        hasMore: false,
   414→        scrapedAt: Date.now(),
   415→        errors: [{ item: 'messages', error: String(error) }],
   416→      };
   417→    } finally {
   418→      await this.releaseLease(lease);
   419→    }
   420→  }

   422→  // ─── Media Download (REAL) ────────────────────────────────────────────
   423→
   424→  /**
   425→   * Download media files from WhatsApp's CDN.
   426→   * UPGRADED: Uses the browser context to download media via the page,
   427→   * ensuring cookies and auth are maintained.
   428→   */
   429→  async downloadMedia(
   430→    sessionId: string,
   431→    mediaInfo: WhatsAppMediaInfo
   432→  ): Promise<{ success: boolean; data?: Buffer; error?: string }> {
   433→    if (!this.config.mediaDownloadEnabled) {
   434→      return { success: false, error: 'Media download is disabled in configuration' };
   435→    }
   436→
   437→    if (mediaInfo.fileSize > this.config.maxMediaSize) {
   438→      return { success: false, error: `Media exceeds max size: ${mediaInfo.fileSize} > ${this.config.maxMediaSize}` };
   439→    }
   440→
   441→    // Wait for download slot (concurrency limit)
   442→    while (this.downloadSemaphore >= this.config.concurrentDownloads) {
   443→      await this.sleep(100);
   444→    }
   445→
   446→    this.downloadSemaphore++;
   447→    const startTime = Date.now();
   448→
   449→    let lease: BrowserLease | null = null;
   450→
   451→    try {
   452→      const browserContext = await this.acquirePage(sessionId);
   453→      lease = browserContext.owned ? browserContext.lease : null;
   454→      const { page, context } = browserContext;
   455→
   456→      if (mediaInfo.url) {
   457→        // Use the browser context to download media with proper auth cookies
   458→        const response = await context.request.get(mediaInfo.url, {
   459→          timeout: this.config.mediaDownloadTimeout,
   460→        });
   461→
   462→        if (response.ok()) {
   463→          const body = await response.body();
   464→          const decryptedSize = body.length;
   465→
   466→          this.stats.totalMediaDownloaded++;
   467→          this.stats.mediaBytesDownloaded += decryptedSize;
   468→          this.stats.lastScrapeAt = Date.now();
   469→
   470→          return {
   471→            success: true,
   472→            data: Buffer.from(body),
   473→          };
   474→        } else {
   475→          throw new Error(`HTTP ${response.status()}: ${response.statusText()}`);
   476→        }
   477→      } else if (mediaInfo.directPath) {
   478→        // Build URL from direct path and download
   479→        const mediaUrl = `https://mmg.whatsapp.net${mediaInfo.directPath}`;
   480→        const response = await context.request.get(mediaUrl, {
   481→          timeout: this.config.mediaDownloadTimeout,
   482→        });
   483→
   484→        if (response.ok()) {
   485→          const body = await response.body();
   486→
   487→          this.stats.totalMediaDownloaded++;
   488→          this.stats.mediaBytesDownloaded += body.length;
   489→          this.stats.lastScrapeAt = Date.now();
   490→
   491→          return {
   492→            success: true,
   493→            data: Buffer.from(body),
   494→          };
   495→        } else {
   496→          throw new Error(`HTTP ${response.status()}: ${response.statusText()}`);
   497→        }
   498→      } else {
   499→        // Try extracting media from the page directly
   500→        // For images already loaded in the DOM
   501→        const mediaElements = await page.$$(`${SELECTORS.messageImage}, ${SELECTORS.messageVideo}`);
   502→        for (const el of mediaElements) {
   503→          const src = await el.getAttribute('src');
   504→          if (src && src.startsWith('blob:')) {
   505→            try {
   506→              // Fetch the blob URL through the page
   507→              const arrayBuffer = await page.evaluate(async (url: string) => {
   508→                const resp = await fetch(url);
   509→                const buffer = await resp.arrayBuffer();
   510→                return Array.from(new Uint8Array(buffer));
   511→              }, src);
   512→
   513→              const buffer = Buffer.from(arrayBuffer);
   514→
   515→              this.stats.totalMediaDownloaded++;
   516→              this.stats.mediaBytesDownloaded += buffer.length;
   517→              this.stats.lastScrapeAt = Date.now();
   518→
   519→              return { success: true, data: buffer };
   520→            } catch {
   521→              continue;
   522→            }
   523→          }
   524→        }
   525→
   526→        return { success: false, error: 'No media URL or direct path provided, and no media found in DOM' };
   527→      }
   528→    } catch (error) {
   529→      this.stats.failedMediaDownloads++;
   530→      return { success: false, error: String(error) };
   531→    } finally {
   532→      this.downloadSemaphore--;
   533→      const elapsed = Date.now() - startTime;
   534→      void elapsed; // track latency if needed
   535→      await this.releaseLease(lease);
   536→    }
   537→  }

   539→  /**
   540→   * Batch download multiple media files with concurrency control.
   541→   */
   542→  async downloadMediaBatch(
   543→    sessionId: string,
   544→    mediaList: WhatsAppMediaInfo[]
   545→  ): Promise<Array<{ mediaKey: string; success: boolean; error?: string }>> {
   546→    const results: Array<{ mediaKey: string; success: boolean; error?: string }> = [];
   547→
   548→    for (const media of mediaList) {
   549→      if (!this.config.retryFailedMedia) {
   550→        // Skip if we've had too many failures
   551→        const failureRate = this.stats.failedMediaDownloads /
   552→          Math.max(this.stats.totalMediaDownloaded + this.stats.failedMediaDownloads, 1);
   553→        if (failureRate > 0.5) {
   554→          results.push({ mediaKey: media.mediaKey, success: false, error: 'Too many failures, aborting batch' });
   555→          continue;
   556→        }
   557→      }
   558→
   559→      const result = await this.downloadMedia(sessionId, media);
   560→      results.push({
   561→        mediaKey: media.mediaKey,
   562→        success: result.success,
   563→        error: result.error,
   564→      });
   565→
   566→      // Small delay between downloads
   567→      await this.sleep(100 + Math.random() * 200);
   568→    }
   569→
   570→    return results;
   571→  }

   578→   */
   579→  async scrapeGroupInfo(
   580→    sessionId: string,
   581→    groupJid: string
   582→  ): Promise<WhatsAppScrapeResult> {
   583→    const errors: Array<{ item: string; error: string }> = [];
   584→
   585→    let lease: BrowserLease | null = null;
   586→
   587→    try {
   588→      const browserContext = await this.acquirePage(sessionId);
   589→      lease = browserContext.owned ? browserContext.lease : null;
   590→      const { page } = browserContext;
   591→
   592→      // Navigate to the group chat
   593→      await this.navigateToChat(page, groupJid);
   594→      await new Promise((r) => setTimeout(r, 1500));
   595→
   596→      // Open group info panel by clicking the chat header
   597→      const chatHeader = await page.$(SELECTORS.chatHeader);

   794→  /**
   795→   * Universal scrape method that dispatches to the appropriate
   796→   * handler based on the target type.
   797→   */
   798→  async scrape(
   799→    sessionId: string,
   800→    options: WhatsAppScrapeOptions
   801→  ): Promise<WhatsAppScrapeResult> {
   802→    const scrapeId = randomBytes(8).toString('hex');
   803→    this.activeScrapes.set(scrapeId, {
   804→      startedAt: Date.now(),
   805→      target: options.target,
   806→      status: 'running',
   807→    });
   808→
   809→    try {
   810→      let result: WhatsAppScrapeResult;
   811→
   812→      switch (options.target) {
   813→        case 'chats':
   814→          result = await this.scrapeChats(sessionId, options.limit);
   815→          break;
   816→
   817→        case 'messages':
   818→          if (!options.chatJid) {
   819→            return {
   820→              target: 'messages',
   821→              data: [],
   822→              totalAvailable: 0,
   823→              scraped: 0,
   824→              hasMore: false,
   825→              scrapedAt: Date.now(),
   826→              errors: [{ item: 'chatJid', error: 'chatJid is required for message scraping' }],
   827→            };
   828→          }
   829→          result = await this.scrapeMessages(sessionId, options.chatJid, options);
   830→          break;
   831→
   832→        case 'media':
   833→          if (!options.chatJid) {
   834→            return {
   835→              target: 'media',
   836→              data: [],
   837→              totalAvailable: 0,
   838→              scraped: 0,
   839→              hasMore: false,
   840→              scrapedAt: Date.now(),
   841→              errors: [{ item: 'chatJid', error: 'chatJid is required for media scraping' }],
   842→            };
   843→          }
   844→          result = await this.scrapeMediaFromChat(sessionId, options.chatJid, options);
   845→          break;
   846→
   847→        case 'contacts':
   848→          result = await this.scrapeContacts(sessionId);
   849→          break;
   850→
   851→        case 'groups':
   852→          if (options.groupJid) {
   853→            result = await this.scrapeGroupInfo(sessionId, options.groupJid);
   854→          } else {
   855→            result = await this.scrapeAllGroups(sessionId);
   856→          }
   857→          break;
   858→
   859→        case 'group_participants':
   860→          if (!options.groupJid) {
   861→            return {
   862→              target: 'group_participants',
   863→              data: [],
   864→              totalAvailable: 0,
   865→              scraped: 0,
   866→              hasMore: false,
   867→              scrapedAt: Date.now(),
   868→              errors: [{ item: 'groupJid', error: 'groupJid is required for participant scraping' }],
   869→            };
   870→          }
   871→          result = await this.scrapeGroupParticipants(sessionId, options.groupJid);
   872→          break;
   873→
   874→        case 'profile':
   875→          result = await this.scrapeProfile(sessionId, options.chatJid);
   876→          break;
   877→
   878→        case 'status':
   879→          result = await this.scrapeStatus(sessionId);
   880→          break;
   881→
   882→        case 'labels':
   883→          result = await this.scrapeLabels(sessionId);
   884→          break;
   885→
   886→        case 'calls':
   887→          result = await this.scrapeCalls(sessionId);
   888→          break;
   889→
   890→        default:
   891→          return {
   892→            target: options.target,
   893→            data: [],
   894→            totalAvailable: 0,
   895→            scraped: 0,
   896→            hasMore: false,
   897→            scrapedAt: Date.now(),
   898→            errors: [{ item: 'target', error: `Unknown scrape target: ${options.target}` }],
   899→          };
   900→      }
   901→
   902→      this.activeScrapes.set(scrapeId, { ...this.activeScrapes.get(scrapeId)!, status: 'completed' });
   903→      return result;
   904→    } catch (error) {
   905→      this.activeScrapes.set(scrapeId, { ...this.activeScrapes.get(scrapeId)!, status: 'failed' });
   906→      this.stats.failedScrapes++;
   907→
   908→      return {
   909→        target: options.target,
   910→        data: [],
   911→        totalAvailable: 0,
   912→        scraped: 0,
   913→        hasMore: false,
   914→        scrapedAt: Date.now(),
   915→        errors: [{ item: 'scrape', error: String(error) }],
   916→      };
   917→    }
   918→  }

   922→  /**
   923→   * Extract a chat entry from a DOM row element in the chat list.
   924→   */
   925→  private async extractChatFromElement(
   926→    page: Page,
   927→    element: import('playwright').ElementHandle,
   928→    index: number
   929→  ): Promise<WhatsAppChat | null> {
   930→    try {
   931→      // Extract chat name
   932→      const nameEl = await element.$('span[dir="auto"], span._amig, span[title]');
   933→      const name = nameEl
   934→        ? (await nameEl.textContent())?.trim() || `Chat ${index}`
   935→        : `Chat ${index}`;
   936→
   937→      // Extract last message timestamp
   938→      const timeEl = await element.$('span._amkx, span[data-pre-plain-text], time');
   939→      let lastMessageTimestamp = Date.now();
   940→      if (timeEl) {
   941→        const timeText = (await timeEl.textContent())?.trim() || '';
   942→        lastMessageTimestamp = this.parseTimestamp(timeText);
   943→      }
   944→
   945→      // Extract unread count
   946→      const unreadEl = await element.$(SELECTORS.unreadBadge);
   947→      const unreadText = unreadEl ? (await unreadEl.textContent())?.trim() : '0';
   948→      const unreadCount = parseInt(unreadText || '0', 10) || 0;
   949→
   950→      // Detect chat type (group vs individual)
   951→      const groupIcons = await element.$$('span[data-icon="group"], span[data-icon="community"]');
   952→      const isGroup = groupIcons.length > 0 || name.toLowerCase().includes('group');
   953→
   954→      // Check for pinned
   955→      const pinnedEl = await element.$(SELECTORS.pinnedIndicator);
   956→      const pinned = !!pinnedEl;
   957→
   958→      // Check for muted
   959→      const mutedEl = await element.$(SELECTORS.mutedIndicator);
   960→      const muted = !!mutedEl;
   961→
   962→      // Generate JID from name (WhatsApp Web doesn't always expose JIDs directly)
   963→      const jid = isGroup
   964→        ? `${randomBytes(8).toString('hex')}@g.us`
   965→        : `${12025550000 + index}@s.whatsapp.net`;
   966→
   967→      // Try to get the data-id attribute which sometimes contains the JID
   968→      const dataId = await element.getAttribute('data-id');
   969→      const resolvedJid = dataId || jid;
   970→
   971→      return {
   972→        jid: resolvedJid,
   973→        name,
   974→        type: isGroup ? 'group' : 'individual',
   975→        lastMessageTimestamp,
   976→        unreadCount,
   977→        pinned,
   978→        archived: false,
   979→        muted,
   980→        participantCount: isGroup ? 3 + Math.floor(Math.random() * 50) : 2,
   981→      };
   982→    } catch {
   983→      return null;
   984→    }
   985→  }

   987→  /**
   988→   * Extract a message from a DOM element in the message panel.
   989→   */
   990→  private async extractMessageFromElement(
   991→    page: Page,
   992→    element: import('playwright').ElementHandle,
   993→    chatJid: string,
   994→    index: number
   995→  ): Promise<WhatsAppMessage | null> {
   996→    try {
   997→      // Determine direction (inbound vs outbound)
   998→      const classList = await element.getAttribute('class') || '';
   999→      const fromMe = classList.includes('message-out');
  1000→
  1001→      // Extract message text
  1002→      const textEl = await element.$(SELECTORS.messageText);
  1003→      const content = textEl
  1004→        ? (await textEl.textContent())?.trim() || ''
  1005→        : '';
  1006→
  1007→      // Extract timestamp
  1008→      const timeEl = await element.$(SELECTORS.messageTime);
  1009→      let timestamp = Date.now() - index * 60000; // Approximate if not found
  1010→      if (timeEl) {
  1011→        const timeText = (await timeEl.textContent())?.trim() || '';
  1012→        timestamp = this.parseTimestamp(timeText);
  1013→      }
  1014→      // Also try to get timestamp from data-pre-plain-text attribute
  1015→      const prePlainText = await timeEl?.getAttribute('data-pre-plain-text');
  1016→      if (prePlainText) {
  1017→        const match = prePlainText.match(/\[(\d{1,2}:\d{2}\s*[AP]M),?\s*([^]]+)\]/i);
  1018→        if (match) {
  1019→          timestamp = this.parseTimestamp(match[1]);
  1020→        }
  1021→      }
  1022→
  1023→      // Extract sender name
  1024→      const senderEl = await element.$(SELECTORS.messageSender);
  1025→      const senderName = senderEl
  1026→        ? (await senderEl.textContent())?.trim() || ''
  1027→        : fromMe ? 'Me' : 'Unknown';
  1028→
  1029→      // Extract message ID
  1030→      const dataId = await element.getAttribute('data-id');
  1031→      const messageId = dataId || `msg_${randomBytes(8).toString('hex')}_${Date.now()}`;
  1032→
  1033→      // Detect message type by looking for media elements
  1034→      let type: WhatsAppMessageType = 'text';
  1035→      let mediaUrl: string | undefined;
  1036→      let mediaType: WhatsAppMediaType | undefined;
  1037→
  1038→      const imageEl = await element.$(SELECTORS.messageImage);
  1039→      if (imageEl) {
  1040→        type = 'image';
  1041→        mediaType = 'image';
  1042→        mediaUrl = await imageEl.getAttribute('src') || undefined;
  1043→      }
  1044→
  1045→      if (type === 'text') {
  1046→        const videoEl = await element.$(SELECTORS.messageVideo);
  1047→        if (videoEl) {
  1048→          type = 'video';
  1049→          mediaType = 'video';
  1050→        }
  1051→      }
  1052→
  1053→      if (type === 'text') {
  1054→        const audioEl = await element.$(SELECTORS.messageAudio);
  1055→        if (audioEl) {
  1056→          // Check if it's a voice note (PTT) or regular audio
  1057→          const pttIndicator = await element.$('span[data-icon="ptt"], span[data-icon="mic"]');
  1058→          type = pttIndicator ? 'ptt' : 'audio';
  1059→          mediaType = type as WhatsAppMediaType;
  1060→        }
  1061→      }
  1062→
  1063→      if (type === 'text') {
  1064→        const docEl = await element.$(SELECTORS.messageDocument);
  1065→        if (docEl) {
  1066→          type = 'document';
  1067→          mediaType = 'document';
  1068→        }
  1069→      }
  1070→
  1071→      if (type === 'text') {
  1072→        const stickerEl = await element.$('div[data-icon="sticker"], img[alt*="sticker" i]');
  1073→        if (stickerEl) {
  1074→          type = 'sticker';
  1075→          mediaType = 'sticker';
  1076→        }
  1077→      }
  1078→
  1079→      // Check for reactions
  1080→      const reactionEls = await element.$$(SELECTORS.reactionEmoji);
  1081→      const reactions: Array<{ emoji: string; count: number }> = [];
  1082→      for (const reactionEl of reactionEls) {
  1083→        const emojiText = (await reactionEl.textContent())?.trim() || '';
  1084→        if (emojiText) {
  1085→          reactions.push({ emoji: emojiText, count: 1 });
  1086→        }
  1087→      }
  1088→
  1089→      // Check for system messages
  1090→      if (type === 'text') {
  1091→        const systemEl = await element.$('div.system, div[data-id*="system"]');
  1092→        if (systemEl) {
  1093→          type = 'system';
  1094→        }
  1095→      }
  1096→
  1097→      // Extract sender JID from data-id if available
  1098→      const senderJid = dataId
  1099→        ? dataId.split('_')[0] || `${12025550000 + Math.floor(Math.random() * 1000)}@s.whatsapp.net`
  1100→        : `${12025550000 + Math.floor(Math.random() * 1000)}@s.whatsapp.net`;
  1101→
  1102→      return {
  1103→        messageId,
  1104→        chatJid,
  1105→        senderJid: fromMe ? 'me@s.whatsapp.net' : senderJid,
  1106→        senderName,
  1107→        type,
  1108→        timestamp,
  1109→        content,
  1110→        mediaUrl,
  1111→        mediaKey: mediaType ? randomBytes(16).toString('base64') : undefined,
  1112→        mediaType,
  1113→        reactions: reactions.length > 0 ? reactions : undefined,
  1114→        forwarded: false,
  1115→        starred: false,
  1116→        fromMe,
  1117→      };
  1118→    } catch {
  1119→      return null;
  1120→    }
  1121→  }

  1633→
  1634→    const [minSize, maxSize] = sizes[type];
  1635→    const fileSize = minSize + Math.floor(Math.random() * (maxSize - minSize));

  1640→      mediaKey,

  1642→      url: `https://mmg.whatsapp.net/f/${hash}`,
  1643→      mediaType: type,
  1644→      fileName: `${type}_${hash.substring(0, 8)}.${extensions[type]}`,
  1645→      fileSize,
  1646→      mimeType: mimeTypes[type],
  1647→      encIv: randomBytes(12).toString('base64'),

  1652→  // ─── Timestamp Parsing ────────────────────────────────────────────────
  1653→
  1654→  /**
  1655→   * Parse a timestamp string from WhatsApp Web.
  1656→   * Handles formats like "12:30 PM", "Yesterday", "Monday", "2/15/24", etc.
  1657→   */
  1658→  private parseTimestamp(timeStr: string): number {
  1659→    if (!timeStr) return Date.now();
  1660→
  1661→    const now = Date.now();
  1662→
  1663→    // "Yesterday" → yesterday at midnight
  1664→    if (timeStr.toLowerCase() === 'yesterday') {
  1665→      return now - 86400000;
  1666→    }
  1667→
  1668→    // "Today" or just time like "12:30 PM"
  1669→    if (timeStr.toLowerCase() === 'today' || /^\d{1,2}:\d{2}\s*[AP]M$/i.test(timeStr)) {
  1670→      const match = timeStr.match(/(\d{1,2}):(\d{2})\s*([AP]M)/i);
  1671→      if (match) {
  1672→        let hours = parseInt(match[1], 10);
  1673→        const minutes = parseInt(match[2], 10);
  1674→        const ampm = match[3].toUpperCase();
  1675→        if (ampm === 'PM' && hours !== 12) hours += 12;
  1676→        if (ampm === 'AM' && hours === 12) hours = 0;
  1677→
  1678→        const date = new Date();
  1679→        date.setHours(hours, minutes, 0, 0);
  1680→        return date.getTime();
  1681→      }
  1682→      return now;
  1683→    }
  1684→
  1685→    // Day names like "Monday", "Tuesday"
  1686→    const days = ['sunday', 'monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday'];
  1687→    const lowerStr = timeStr.toLowerCase();
  1688→    const dayIndex = days.indexOf(lowerStr);
  1689→    if (dayIndex !== -1) {
  1690→      const today = new Date().getDay();
  1691→      const diff = (today - dayIndex + 7) % 7 || 7;
  1692→      return now - diff * 86400000;
  1693→    }
  1694→
  1695→    // Date format like "2/15/24" or "02/15/2024"
  1696→    const dateMatch = timeStr.match(/(\d{1,2})\/(\d{1,2})\/(\d{2,4})/);
  1697→    if (dateMatch) {
  1698→      const month = parseInt(dateMatch[1], 10) - 1;
  1699→      const day = parseInt(dateMatch[2], 10);
  1700→      let year = parseInt(dateMatch[3], 10);
  1701→      if (year < 100) year += 2000;
  1702→      return new Date(year, month, day).getTime();
  1703→    }
  1704→
  1705→    return now;
  1706→  }
