   584→        tlsHead.defeated = hasTls;
   585→        tlsHead.confidence = hasTls ? 0.8 : 0.4;
   586→        tlsHead.lastCheckAt = Date.now();
   587→      }
   588→    } catch (err: any) {
   589→      const tlsHead = heads.find(h => h.head === 'tls');
   590→      if (tlsHead) {
   591→        // TLS engine not available -- assume OK if we're using a real browser
   592→        tlsHead.defeated = true;
   593→        tlsHead.confidence = 0.5;
   594→        tlsHead.lastCheckAt = Date.now();
   595→      }
   596→    }
   597→
   598→    // Step 5: HTTP/2 head -- delegate to deep browser patcher
   599→    try {
   600→      const { deepBrowserPatcher } = await import('./deep-patcher');
   601→      const hasHeaderOrder = deepBrowserPatcher.getHeaderOrder('chrome').length > 0;
   602→      const h2Head = heads.find(h => h.head === 'http2');
   603→      if (h2Head) {
   604→        h2Head.defeated = hasHeaderOrder;
   605→        h2Head.confidence = hasHeaderOrder ? 0.75 : 0.4;
   606→        h2Head.lastCheckAt = Date.now();
   607→      }
   608→    } catch (err: any) {
   609→      const h2Head = heads.find(h => h.head === 'http2');
   610→      if (h2Head) {
   611→        h2Head.defeated = true;
   612→        h2Head.confidence = 0.5;
   613→        h2Head.lastCheckAt = Date.now();
   614→      }
   615→    }
   616→
   617→    // Step 6: Behavioral head -- simulate human interactions
   618→    try {
   619→      await this.simulateBehavior(page);
   620→      const behavioralHead = heads.find(h => h.head === 'behavioral');
   621→      if (behavioralHead) {
   622→        behavioralHead.defeated = true;
   623→        behavioralHead.confidence = 0.7;
   624→        behavioralHead.lastCheckAt = Date.now();
   625→      }
   626→    } catch (err: any) {
   627→      const behavioralHead = heads.find(h => h.head === 'behavioral');
   628→      if (behavioralHead) behavioralHead.errors.push(err.message);
   629→    }
   630→
   631→    return heads;
   632→  }

   634→  // --- Challenge Resolution ------------------------------------------------
   635→
   636→  private async waitForChallengeResolution(page: Page, timeout: number): Promise<boolean> {
   637→    const startTime = Date.now();
   638→    const checkInterval = 500;
   639→
   640→    while (Date.now() - startTime < timeout) {
   641→      try {
   642→        // Check if challenge elements are gone
   643→        let challengeGone = true;
   644→        for (const selector of AKAMAI_CHALLENGE_SELECTORS) {
   645→          try {
   646→            const el = await page.$(selector);
   647→            if (el) { challengeGone = false; break; }
   648→          } catch { /* selector check failed */ }
   649→        }
   650→
   651→        // Check for Akamai cookies
   652→        let hasAkamaiCookies = false;
   653→        try {
   654→          const cookieAttr = await page.evaluate(() =>
   655→            document.documentElement.getAttribute('data-hydra-cookies')
   656→          );
   657→          if (cookieAttr) {
   658→            const parsed = JSON.parse(cookieAttr);
   659→            hasAkamaiCookies = !!(parsed.ak_bmsc || parsed._abck);
   660→          }
   661→        } catch { /* cookie check failed */ }
   662→
   663→        if (!hasAkamaiCookies) {
   664→          try {
   665→            const cookies = await page.context().cookies();
   666→            hasAkamaiCookies = cookies.some(c => c.name === '_abck' || c.name === 'ak_bmsc');
   667→          } catch { /* cookie access failed */ }
   668→        }
   669→
   670→        if ((challengeGone && hasAkamaiCookies) || hasAkamaiCookies) {
   671→          return true;
   672→        }
   673→
   674→        await this.sleep(checkInterval);
   675→      } catch {
   676→        // Page may have navigated
   677→        try {
   678→          await page.evaluate(() => document.title);
   679→        } catch {
   680→          return true; // Page navigated away -- likely solved
   681→        }
   682→      }
   683→    }
   684→
   685→    return false;
   686→  }

   688→  // --- Behavioral Simulation -----------------------------------------------
   689→
   690→  private async simulateBehavior(page: Page): Promise<void> {
   691→    try {
   692→      // Simulate mouse movements with human-like patterns
   693→      const viewport = page.viewportSize() || { width: 1920, height: 1080 };
   694→      const steps = 5 + Math.floor(Math.random() * 8);
   695→
   696→      for (let i = 0; i < steps; i++) {
   697→        const x = Math.floor(Math.random() * viewport.width * 0.8 + viewport.width * 0.1);
   698→        const y = Math.floor(Math.random() * viewport.height * 0.8 + viewport.height * 0.1);
   699→        await page.mouse.move(x, y, { steps: 3 + Math.floor(Math.random() * 5) });
   700→        await this.sleep(50 + Math.random() * 200);
   701→      }
   702→
   703→      // Simulate a scroll
   704→      await page.mouse.wheel(0, 100 + Math.floor(Math.random() * 300));
   705→      await this.sleep(200 + Math.random() * 500);
   706→
   707→      // Simulate another scroll
   708→      await page.mouse.wheel(0, -50 - Math.floor(Math.random() * 150));
   709→      await this.sleep(100 + Math.random() * 300);
   710→    } catch (err: any) {
   711→      logger.debug({ err: err.message }, 'Behavioral simulation failed (non-critical)');
   712→    }
   713→  }

   715→  // --- Cookie Extraction ---------------------------------------------------
   716→
   717→  private async extractCookies(
   718→    ctx: BypassContext,
   719→    domain: string,
   720→    strategy: BypassStrategy
   721→  ): Promise<ManagedCookie[]> {
   722→    const tokens: ManagedCookie[] = [];
   723→    const now = Date.now();
   724→
   725→    try {
   726→      const cookies = await ctx.context.cookies();
   727→      for (const cookie of cookies) {
   728→        if (AKAMAI_COOKIE_NAMES.includes(cookie.name)) {
   729→          tokens.push(this.createManagedCookie({
   730→            name: cookie.name,
   731→            value: cookie.value,
   732→            domain: cookie.domain || domain,
   733→            path: cookie.path || '/',
   734→            httpOnly: cookie.httpOnly,
   735→            secure: cookie.secure,
   736→            sameSite: (cookie.sameSite as 'Strict' | 'Lax' | 'None') || 'Lax',
   737→          }, this.config.tokenFreshnessMs));
   738→        }
   739→      }
   740→
   741→      // Also try extracting from DOM
   742→      try {
   743→        const cookieAttr = await ctx.page.evaluate(() =>
   744→          document.documentElement.getAttribute('data-hydra-cookies')
   745→        );
   746→        if (cookieAttr) {
   747→          const parsed = JSON.parse(cookieAttr);
   748→          for (const [name, value] of Object.entries(parsed)) {
   749→            if (!tokens.some(t => t.name === name)) {
   750→              tokens.push(this.createManagedCookie({
   751→                name,
   752→                value: value as string,
   753→                domain,
   754→              }, this.config.tokenFreshnessMs));
   755→            }
   756→          }
   757→        }
   758→      } catch { /* DOM extraction failed */ }
   759→
   760→      // Store tokens
   761→      if (tokens.length > 0) {
   762→        await this.storeTokens(domain, tokens);
   763→      }
   764→    } catch (err: any) {
   765→      logger.error({ err: err.message, domain }, 'Akamai cookie extraction failed');
   766→    }
   767→
   768→    return tokens;
   769→  }

   773→  private estimateRechallengeTime(heads: HydraHeadStatus[]): number {
   774→    // If sensor head was barely defeated, rechallenge will come sooner
   775→    const sensorHead = heads.find(h => h.head === 'sensor');

   777→      return 60000 + Math.random() * 120000; // 1-3 minutes
   778→    }
   779→    return 300000 + Math.random() * 300000; // 5-10 minutes
   780→  }
