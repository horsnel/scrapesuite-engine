   761→  /**
   762→   * Solve a Proof-of-Work challenge by finding a nonce
   763→   * that produces a hash with the required difficulty prefix.
   764→   */
   765→  private async solvePowChallenge(input: string, difficulty: number): Promise<string> {
   766→    const targetPrefix = '0'.repeat(difficulty);
   767→    let nonce = 0;
   768→    const maxAttempts = 5_000_000; // Safety limit
   769→    const timeout = Date.now() + this.config.wasmChallengeTimeout;
   770→
   771→    while (nonce < maxAttempts && Date.now() < timeout) {
   772→      const hash = createHmac('sha256', `${input}${nonce}`).digest('hex');
   773→      if (hash.startsWith(targetPrefix)) {
   774→        return `${nonce}:${hash}`;
   775→      }
   776→      nonce++;
   777→
   778→      // Yield to event loop every 100K iterations
   779→      if (nonce % 100000 === 0) {
   780→        await new Promise((resolve) => setTimeout(resolve, 0));
   781→      }
   782→    }
   783→
   784→    throw new Error(`PoW challenge not solved within ${maxAttempts} attempts or timeout`);
   785→  }
