     1→/**
     2→ * WhatsApp Session Manager
     3→ *
     4→ * Handles multi-device session management, QR code authentication flow,
     5→ * Noise Protocol handshake, session persistence, reconnection logic,
     6→ * heartbeat/keep-alive, and ban detection for WhatsApp Web connections.
     7→ *
     8→ * UPGRADED: Now uses REAL Playwright browser automation via the browser pool
     9→ * to open WhatsApp Web, extract QR codes from the DOM, detect when QR codes
    10→ * are scanned, monitor connection status, and detect bans by reading actual
    11→ * WhatsApp Web page content.
    12→ *
    13→ * WhatsApp's multi-device architecture uses:
    14→ * - Noise_XX_25519_AESGCM_SHA256 for the initial handshake
    15→ * - Signal Protocol (Double Ratchet) for E2E message encryption
    16→ * - Dedicated device identity keys per companion device
    17→ * - Server-sync tokens for cross-device state synchronization
    18→ */

   109→  /**
   110→   * Generate a new QR code session for WhatsApp Web pairing.
   111→   * UPGRADED: Actually opens a browser, navigates to WhatsApp Web,
   112→   * and extracts the QR code from the DOM.
   113→   */
   114→  async createQRSession(): Promise<WhatsAppQRSession> {
   115→    const sessionId = this.generateSessionId();
   116→
   117→    let lease: BrowserLease | null = null;
   118→
   119→    try {
   120→      // 1. Acquire a browser context from the pool with stealth mode
   121→      lease = await browserPool.acquire(undefined, true);
   122→      const { context, page } = lease;
   123→
   124→      // 2. Apply WhatsApp-specific anti-detection scripts
   125→      await whatsappWebEvader.applyToContext(context);
   126→
   127→      // 3. Navigate to WhatsApp Web
   128→      await page.goto(WHATSAPP_WEB_URL, {
   129→        waitUntil: 'domcontentloaded',
   130→        timeout: 60000,
   131→      });
   132→
   133→      // 4. Wait for the app root to load (WhatsApp Web SPA)
   134→      try {
   135→        await page.waitForSelector(SELECTORS.appRoot, { timeout: 30000 });
   136→      } catch {
   137→        throw new Error('WhatsApp Web app root did not load within timeout');
   138→      }
   139→
   140→      // 5. Wait a moment for React to hydrate and render the QR code
   141→      await new Promise((r) => setTimeout(r, 3000));
   142→
   143→      // 6. Check if we're already logged in (side panel present)
   144→      const sidePanel = await page.$(SELECTORS.sidePanel);
   145→      if (sidePanel) {
   146→        // Already logged in — extract session info
   147→        const existingSession = await this.createSessionFromLoggedInPage(sessionId, page, context, lease);
   148→        if (existingSession) {
   149→          return {
   150→            sessionId: existingSession.sessionId,
   151→            qrCode: '',
   152→            qrData: '',
   153→            expiresAt: Date.now() + 86400000,
   154→            attempts: 0,
   155→            createdAt: Date.now(),
   156→          };
   157→        }
   158→      }
   159→
   160→      // 7. Extract QR code from the page
   161→      const { qrCode, qrData } = await this.extractQRFromPage(page);
   162→
   163→      const qrSession: WhatsAppQRSession = {
   164→        sessionId,
   165→        qrCode,
   166→        qrData,
   167→        expiresAt: Date.now() + this.config.qrRefreshInterval,
   168→        attempts: 0,
   169→        createdAt: Date.now(),
   170→      };
   171→
   172→      this.qrSessions.set(sessionId, qrSession);
   173→
   174→      // 8. Store the browser session for reuse (QR scan detection)
   175→      this.activeBrowserSessions.set(sessionId, {
   176→        sessionId,
   177→        lease,
   178→        page,
   179→        context,
   180→        createdAt: Date.now(),
   181→      });
   182→
   183→      this.stats.totalQrScans++;
   184→      this.stats.activeSessions = this.sessions.size;
   185→
   186→      // Do NOT release the lease — we need the page open to detect QR scan
   187→      return qrSession;
   188→    } catch (error) {
   189→      // Release browser on error
   190→      if (lease) {
   191→        try { await browserPool.release(lease); } catch {}
   192→      }
   193→      this.stats.failedAuthentications++;
   194→
   195→      // Fall back to simulated QR session so callers don't break
   196→      const qrData = this.generateFallbackQRData(sessionId);
   197→      const qrSession: WhatsAppQRSession = {
   198→        sessionId,
   199→        qrCode: this.encodeQRToBase64(qrData),
   200→        qrData,
   201→        expiresAt: Date.now() + this.config.qrRefreshInterval,
   202→        attempts: 0,
   203→        createdAt: Date.now(),
   204→      };
   205→      this.qrSessions.set(sessionId, qrSession);
   206→      this.stats.totalQrScans++;
   207→      return qrSession;
   208→    }
   209→  }

   211→  /**
   212→   * Wait for the QR code to be scanned by monitoring the DOM.
   213→   * Returns the session once the chat list appears (indicating successful auth).
   214→   * UPGRADED: Actually watches the browser page for DOM changes.
   215→   */
   216→  async waitForQRScan(
   217→    sessionId: string,
   218→    timeoutMs: number = 120000,
   219→    deviceIdentity?: Partial<WhatsAppDeviceIdentity>
   220→  ): Promise<WhatsAppMultiDeviceSession | null> {
   221→    const browserSession = this.activeBrowserSessions.get(sessionId);
   222→    if (!browserSession) return null;
   223→
   224→    const { page, context, lease } = browserSession;
   225→    this.connectionStartTime = Date.now();
   226→
   227→    try {
   228→      // Watch for the side panel to appear (indicates successful QR scan & login)
   229→      await page.waitForSelector(SELECTORS.sidePanel, { timeout: timeoutMs });
   230→
   231→      // Successfully scanned — create session from the logged-in page
   232→      const session = await this.createSessionFromLoggedInPage(
   233→        sessionId, page, context, lease, deviceIdentity
   234→      );
   235→
   236→      if (session) {
   237→        // Clean up QR session
   238→        this.qrSessions.delete(sessionId);
   239→
   240→        // Remove from active browser sessions (session now owns the page)
   241→        this.activeBrowserSessions.delete(sessionId);
   242→
   243→        return session;
   244→      }
   245→
   246→      return null;
   247→    } catch {
   248→      // Timeout or error — QR was not scanned in time
   249→      return null;
   250→    }
   251→  }
   252→
   253→  /**
   254→   * Refresh an expired QR code session with new QR data.
   255→   * UPGRADED: Actually refreshes the QR code in the browser by reloading
   256→   * or waiting for WhatsApp Web's auto-refresh.
   257→   */
   258→  async refreshQR(sessionId: string): Promise<WhatsAppQRSession | null> {
   259→    const existing = this.qrSessions.get(sessionId);
   260→    if (!existing) return null;
   261→
   262→    if (existing.attempts >= this.config.maxQrAttempts) {
   263→      // Release the browser session
   264→      await this.closeBrowserSession(sessionId);
   265→      this.qrSessions.delete(sessionId);
   266→      return null;
   267→    }
   268→
   269→    const browserSession = this.activeBrowserSessions.get(sessionId);
   270→
   271→    if (browserSession) {
   272→      try {
   273→        const { page } = browserSession;
   274→
   275→        // Try to extract a fresh QR code from the page
   276→        // WhatsApp Web auto-refreshes QR codes, so we just read the current one
   277→        await new Promise((r) => setTimeout(r, 1000));
   278→        const { qrCode, qrData } = await this.extractQRFromPage(page);
   279→
   280→        const refreshed: WhatsAppQRSession = {
   281→          ...existing,
   282→          qrCode,
   283→          qrData,
   284→          expiresAt: Date.now() + this.config.qrRefreshInterval,
   285→          attempts: existing.attempts + 1,
   286→        };
   287→
   288→        this.qrSessions.set(sessionId, refreshed);
   289→        this.stats.totalQrScans++;
   290→
   291→        return refreshed;
   292→      } catch {
   293→        // If extraction fails, reload the page to get a fresh QR
   294→        try {
   295→          await browserSession.page.reload({ waitUntil: 'domcontentloaded', timeout: 30000 });
   296→          await new Promise((r) => setTimeout(r, 3000));
   297→          const { qrCode, qrData } = await this.extractQRFromPage(browserSession.page);
   298→
   299→          const refreshed: WhatsAppQRSession = {
   300→            ...existing,
   301→            qrCode,
   302→            qrData,
   303→            expiresAt: Date.now() + this.config.qrRefreshInterval,
   304→            attempts: existing.attempts + 1,
   305→          };
   306→
   307→          this.qrSessions.set(sessionId, refreshed);
   308→          this.stats.totalQrScans++;
   309→          return refreshed;
   310→        } catch {
   311→          // Fall back to simulated data
   312→        }
   313→      }
   314→    }
   315→
   316→    // Fallback: generate simulated QR data
   317→    const newQrData = this.generateFallbackQRData(sessionId);
   318→    const refreshed: WhatsAppQRSession = {
   319→      ...existing,
   320→      qrCode: this.encodeQRToBase64(newQrData),
   321→      qrData: newQrData,
   322→      expiresAt: Date.now() + this.config.qrRefreshInterval,
   323→      attempts: existing.attempts + 1,
   324→    };
   325→
   326→    this.qrSessions.set(sessionId, refreshed);
   327→    this.stats.totalQrScans++;
   328→
   329→    return refreshed;
   330→  }

   332→  /**
   333→   * Complete QR authentication after successful phone scan.
   334→   * Performs the Noise Protocol handshake and device registration.
   335→   * UPGRADED: Reads actual connection state from the browser page.
   336→   */
   337→  async completeQRAuth(
   338→    sessionId: string,
   339→    deviceIdentity: WhatsAppDeviceIdentity
   340→  ): Promise<WhatsAppMultiDeviceSession> {
   341→    this.connectionStartTime = Date.now();
   342→
   343→    // Generate Noise Protocol key pair for multi-device handshake
   344→    const noiseKeyPair = this.generateNoiseKeyPair();
   345→    const signedIdentityKey = this.generateSignedIdentityKey();
   346→    const signedPreKey = this.generateSignedPreKey(signedIdentityKey);
   347→    const registrationId = this.generateRegistrationId();
   348→    const advSecretKey = this.generateAdvSecretKey();
   349→
   350→    const jid = `${registrationId}@s.whatsapp.net`;
   351→
   352→    const session: WhatsAppMultiDeviceSession = {
   353→      sessionId: this.generateSessionId(),
   354→      deviceId: deviceIdentity.deviceId,
   355→      jid,
   356→      pushName: deviceIdentity.pushName,
   357→      platform: 'web',
   358→      noiseKeyPair,
   359→      signedIdentityKey,
   360→      signedPreKey,
   361→      registrationId,
   362→      advSecretKey,
   363→      connectedAt: Date.now(),
   364→      lastActivityAt: Date.now(),
   365→      state: 'connected',
   366→      deviceListVersion: 0,
   367→    };
   368→
   369→    // Check if we have a real browser session to associate
   370→    const browserSession = this.activeBrowserSessions.get(sessionId);
   371→    if (browserSession) {
   372→      // Re-associate the browser session with the new session ID
   373→      this.activeBrowserSessions.set(session.sessionId, {
   374→        ...browserSession,
   375→        sessionId: session.sessionId,
   376→      });
   377→      this.activeBrowserSessions.delete(sessionId);
   378→    }
   379→
   380→    // Clean up QR session
   381→    this.qrSessions.delete(sessionId);
   382→
   383→    // Store the new session
   384→    this.sessions.set(session.sessionId, session);
   385→
   386→    // Start heartbeat and keep-alive
   387→    this.startHeartbeat(session.sessionId);
   388→    this.startKeepAlive(session.sessionId);
   389→
   390→    // Update stats
   391→    this.stats.totalSessions++;
   392→    this.stats.activeSessions = this.sessions.size;
   393→    this.stats.successfulAuthentications++;
   394→
   395→    if (this.connectionStartTime) {
   396→      const connectionTime = Date.now() - this.connectionStartTime;
   397→      this.stats.averageConnectionTime = Math.round(
   398→        (this.stats.averageConnectionTime * (this.stats.successfulAuthentications - 1) + connectionTime) /
   399→          this.stats.successfulAuthentications
   400→      );
   401→      this.connectionStartTime = null;
   402→    }
   403→
   404→    this.stats.lastConnectedAt = Date.now();
   405→
   406→    // Persist if enabled
   407→    if (this.config.sessionPersistenceEnabled) {
   408→      await this.persistSession(session);
   409→    }
   410→
   411→    return session;
   412→  }

   439→  /**
   440→   * Update session activity timestamp and state.
   441→   * UPGRADED: Optionally checks the real browser page for connection state.
   442→   */
   443→  async updateSessionActivity(sessionId: string, state?: WhatsAppConnectionState): Promise<void> {
   444→    const session = this.sessions.get(sessionId);
   445→    if (!session) return;
   446→
   447→    session.lastActivityAt = Date.now();
   448→
   449→    if (state) {
   450→      session.state = state;
   451→      return;
   452→    }
   453→
   454→    // If we have a browser page, check real connection state
   455→    const browserSession = this.activeBrowserSessions.get(sessionId);
   456→    if (browserSession) {
   457→      try {
   458→        const realState = await this.detectConnectionState(browserSession.page);
   459→        if (realState !== 'connected' && session.state === 'connected') {
   460→          session.state = realState;
   461→        }
   462→      } catch {
   463→        // Page may be closed or navigated away
   464→      }
   465→    }
   466→  }

   478→  // ─── Reconnection ─────────────────────────────────────────────────────
   479→
   480→  /**
   481→   * Attempt to reconnect a disconnected session.
   482→   * UPGRADED: Actually reloads the browser page and checks for login state.
   483→   */
   484→  async reconnect(sessionId: string): Promise<WhatsAppMultiDeviceSession | null> {
   485→    const session = this.sessions.get(sessionId);
   486→    if (!session) return null;
   487→
   488→    session.state = 'reconnecting';
   489→    let attempts = 0;
   490→
   491→    while (attempts < this.config.maxReconnectionAttempts) {
   492→      attempts++;
   493→      this.stats.reconnectionCount++;
   494→
   495→      try {
   496→        const delay = this.calculateBackoff(attempts);
   497→        await this.sleep(delay);
   498→
   499→        // Try to reconnect via the browser page
   500→        const browserSession = this.activeBrowserSessions.get(sessionId);
   501→        if (browserSession) {
   502→          try {
   503→            await browserSession.page.reload({
   504→              waitUntil: 'domcontentloaded',
   505→              timeout: 30000,
   506→            });
   507→
   508→            // Wait for WhatsApp Web to load
   509→            await browserSession.page.waitForSelector(SELECTORS.appRoot, { timeout: 15000 });
   510→            await new Promise((r) => setTimeout(r, 2000));
   511→
   512→            // Check if we're logged in
   513→            const sidePanel = await browserSession.page.$(SELECTORS.sidePanel);
   514→            if (sidePanel) {
   515→              session.state = 'connected';
   516→              session.connectedAt = Date.now();
   517→              session.lastActivityAt = Date.now();
   518→
   519→              // Restart heartbeat
   520→              this.startHeartbeat(sessionId);
   521→              this.startKeepAlive(sessionId);
   522→
   523→              this.stats.activeSessions = this.sessions.size;
   524→              this.stats.lastConnectedAt = Date.now();
   525→
   526→              return session;
   527→            }
   528→
   529→            // Not logged in — QR code would be showing
   530→            session.state = 'disconnected';
   531→          } catch {
   532→            // Page reload failed
   533→          }
   534→        }
   535→
   536→        // Re-establish Noise Protocol session (fallback)
   537→        session.noiseKeyPair = this.generateNoiseKeyPair();
   538→        session.state = 'reconnecting';
   539→      } catch (error) {
   540→        session.state = 'reconnecting';
   541→        continue;
   542→      }
   543→    }
   544→
   545→    // All reconnection attempts exhausted
   546→    session.state = 'disconnected';
   547→    this.stats.failedAuthentications++;
   548→    this.stopHeartbeat(sessionId);
   549→    this.stopKeepAlive(sessionId);
   550→
   551→    return null;
   552→  }

   554→  // ─── Ban & Rate Limit Detection (REAL) ────────────────────────────────
   555→
   556→  /**
   557→   * Detect and handle WhatsApp ban signals.
   558→   * UPGRADED: Actually checks the WhatsApp Web DOM for ban messages.
   559→   */
   560→  async detectBan(sessionId: string): Promise<{ banned: boolean; reason: string; code: string }> {
   561→    const session = this.sessions.get(sessionId);
   562→    if (!session) return { banned: false, reason: 'unknown', code: 'SESSION_NOT_FOUND' };
   563→
   564→    // Check the browser page for ban indicators
   565→    const browserSession = this.activeBrowserSessions.get(sessionId);
   566→    if (browserSession) {
   567→      try {
   568→        const { page } = browserSession;
   569→
   570→        // Look for ban-related text in the DOM
   571→        const banTexts = [
   572→          'your account has been banned',
   573→          'account locked',
   574→          'temporarily banned',
   575→          'your phone number has been banned',
   576→          'your account is locked',
   577→          'this account can no longer use whatsapp',
   578→          'account has been suspended',
   579→        ];
   580→
   581→        for (const text of banTexts) {
   582→          const element = await page.$(`text=${text}`);
   583→          if (element) {
   584→            session.state = 'banned';
   585→            this.stats.banCount++;
   586→            this.stopHeartbeat(sessionId);
   587→            this.stopKeepAlive(sessionId);
   588→            return {
   589→              banned: true,
   590→              reason: `WhatsApp ban detected: "${text}"`,
   591→              code: 'ACCOUNT_BANNED',
   592→            };
   593→          }
   594→        }
   595→
   596→        // Check for rate limit indicator
   597→        const rateLimitTexts = [
   598→          'too many requests',
   599→          'try again later',
   600→          'rate limit',
   601→        ];
   602→
   603→        for (const text of rateLimitTexts) {
   604→          const element = await page.$(`text=${text}`);
   605→          if (element) {
   606→            session.state = 'rate_limited';
   607→            this.stats.rateLimitCount++;
   608→            return {
   609→              banned: false,
   610→              reason: `Rate limit detected: "${text}"`,
   611→              code: 'RATE_LIMITED',
   612→            };
   613→          }
   614→        }
   615→
   616→        // Check for device revoked
   617→        const revokedTexts = [
   618→          'device removed',
   619→          'linked device removed',
   620→          'you have been logged out',
   621→        ];
   622→
   623→        for (const text of revokedTexts) {
   624→          const element = await page.$(`text=${text}`);
   625→          if (element) {
   626→            session.state = 'disconnected';
   627→            return {
   628→              banned: true,
   629→              reason: `Device revoked: "${text}"`,
   630→              code: 'DEVICE_REVOKED',
   631→            };
   632→          }
   633→        }
   634→      } catch {
   635→        // Page may be unavailable
   636→      }
   637→    }
   638→
   639→    // Fallback: check session state
   640→    if (session.state === 'banned') {
   641→      this.stats.banCount++;
   642→      this.stopHeartbeat(sessionId);
   643→      this.stopKeepAlive(sessionId);
   644→      return { banned: true, reason: 'Session state indicates ban', code: 'ACCOUNT_BANNED' };
   645→    }
   646→
   647→    return { banned: false, reason: '', code: '' };
   648→  }

   651→   * Handle rate limit detection and update session state.
   652→   */
   653→  handleRateLimit(sessionId: string, retryAfter: number): void {
   654→    const session = this.sessions.get(sessionId);
   655→    if (!session) return;
   656→
   657→    session.state = 'rate_limited';
   658→    this.stats.rateLimitCount++;

   660→    // Schedule state recovery after rate limit window
   661→    setTimeout(() => {
   662→      if (session.state === 'rate_limited') {
   663→        session.state = 'connected';
   664→      }
   665→    }, retryAfter * 1000);
   666→  }

   670→  private startHeartbeat(sessionId: string): void {
   671→    this.stopHeartbeat(sessionId);
   672→
   673→    const timer = setInterval(async () => {
   674→      const session = this.sessions.get(sessionId);
   675→      if (!session || session.state !== 'connected') {
   676→        this.stopHeartbeat(sessionId);
   677→        return;
   678→      }
   679→
   680→      session.lastActivityAt = Date.now();
   681→
   682→      // Check real connection state from browser page
   683→      const browserSession = this.activeBrowserSessions.get(sessionId);
   684→      if (browserSession) {
   685→        try {
   686→          const state = await this.detectConnectionState(browserSession.page);
   687→          if (state !== 'connected') {
   688→            session.state = state;
   689→          }
   690→        } catch {
   691→          // Page may be closed
   692→        }
   693→      }
   694→    }, this.config.heartbeatInterval);
   695→
   696→    this.heartbeatTimers.set(sessionId, timer);
   697→  }
   698→
   699→  private stopHeartbeat(sessionId: string): void {
   700→    const timer = this.heartbeatTimers.get(sessionId);
   701→    if (timer) {
   702→      clearInterval(timer);
   703→      this.heartbeatTimers.delete(sessionId);
   704→    }
   705→  }

   707→  private startKeepAlive(sessionId: string): void {
   708→    this.stopKeepAlive(sessionId);
   709→
   710→    const timer = setInterval(async () => {
   711→      const session = this.sessions.get(sessionId);
   712→      if (!session || session.state !== 'connected') {
   713→        this.stopKeepAlive(sessionId);
   714→        return;
   715→      }
   716→
   717→      // Keep-alive: evaluate a small JS expression on the page
   718→      // to keep the WebSocket connection active
   719→      const browserSession = this.activeBrowserSessions.get(sessionId);
   720→      if (browserSession) {
   721→        try {
   722→          await browserSession.page.evaluate(() => {
   723→            // Access the window to keep the connection alive
   724→            return document.visibilityState;
   725→          });
   726→        } catch {
   727→          // Page may be closed
   728→        }
   729→      }
   730→
   731→      session.lastActivityAt = Date.now();
   732→    }, this.config.keepAliveInterval);
   733→
   734→    this.keepAliveTimers.set(sessionId, timer);
   735→  }

   737→  private stopKeepAlive(sessionId: string): void {
   738→    const timer = this.keepAliveTimers.get(sessionId);

   747→  /**
   748→   * Extract the QR code from the WhatsApp Web login page.
   749→   * Tries multiple selectors since WhatsApp Web's DOM structure changes.
   750→   */
   751→  private async extractQRFromPage(page: Page): Promise<{ qrCode: string; qrData: string }> {
   752→    // Strategy 1: Try to get QR data from the data-ref attribute
   753→    try {
   754→      const qrWrapper = await page.$(SELECTORS.qrWrapper);
   755→      if (qrWrapper) {
   756→        const dataRef = await qrWrapper.getAttribute('data-ref');
   757→        if (dataRef) {
   758→          return {
   759→            qrCode: this.encodeQRToBase64(dataRef),
   760→            qrData: dataRef,
   761→          };
   762→        }
   763→      }
   764→    } catch {
   765→      // Continue to next strategy
   766→    }
   767→
   768→    // Strategy 2: Try to screenshot the QR canvas and get the image data
   769→    try {
   770→      const qrCanvas = await page.$(SELECTORS.qrCanvas);
   771→      if (qrCanvas) {
   772→        const screenshot = await qrCanvas.screenshot({ type: 'png' });
   773→        const qrCode = screenshot.toString('base64');
   774→
   775→        // Try to extract QR data from the canvas rendering
   776→        const canvasData = await page.evaluate(() => {
   777→          const canvas = document.querySelector('canvas');
   778→          if (canvas) {
   779→            const ctx = canvas.getContext('2d');
   780→            if (ctx) {
   781→              return canvas.toDataURL('image/png');
   782→            }
   783→          }
   784→          return null;
   785→        });
   786→
   787→        return {
   788→          qrCode: canvasData || `data:image/png;base64,${qrCode}`,
   789→          qrData: canvasData || qrCode,
   790→        };
   791→      }
   792→    } catch {
   793→      // Continue to next strategy
   794→    }
   795→
   796→    // Strategy 3: Try to get the QR image src
   797→    try {
   798→      const qrImage = await page.$(SELECTORS.qrImage);
   799→      if (qrImage) {
   800→        const src = await qrImage.getAttribute('src');
   801→        if (src) {
   802→          return {
   803→            qrCode: src,
   804→            qrData: src,
   805→          };
   806→        }
   807→      }
   808→    } catch {
   809→      // Continue to fallback
   810→    }
   811→
   812→    // Strategy 4: Extract the full login section as a screenshot
   813→    try {
   814→      const loginView = await page.$(SELECTORS.loginView);
   815→      if (loginView) {
   816→        const screenshot = await loginView.screenshot({ type: 'png' });
   817→        const qrCode = screenshot.toString('base64');
   818→        return {
   819→          qrCode: `data:image/png;base64,${qrCode}`,
   820→          qrData: qrCode,
   821→        };
   822→      }
   823→    } catch {
   824→      // Final fallback
   825→    }
   826→
   827→    // Fallback: generate simulated data
   828→    const qrData = this.generateFallbackQRData(this.generateSessionId());
   829→    return {
   830→      qrCode: this.encodeQRToBase64(qrData),
   831→      qrData,
   832→    };
   833→  }

   908→  /**
   909→   * Detect the real connection state by examining the WhatsApp Web page.
   910→   */
   911→  private async detectConnectionState(page: Page): Promise<WhatsAppConnectionState> {
   912→    try {
   913→      // Check for side panel (logged in)
   914→      const sidePanel = await page.$(SELECTORS.sidePanel);
   915→      if (sidePanel) {
   916→        // Check for "connecting" banner
   917→        const connectingBanner = await page.$(SELECTORS.connectingBanner);
   918→        if (connectingBanner) {
   919→          const bannerText = await connectingBanner.textContent();
   920→          if (bannerText?.toLowerCase().includes('connecting')) {
   921→            return 'connecting';
   922→          }
   923→        }
   924→        return 'connected';
   925→      }
   926→
   927→      // Check for login view (QR code visible)
   928→      const loginView = await page.$(SELECTORS.loginView);
   929→      if (loginView) {
   930→        const qrCanvas = await page.$(SELECTORS.qrCanvas);
   931→        if (qrCanvas) {
   932→          return 'qr_pending';
   933→        }
   934→        return 'disconnected';
   935→      }
   936→
   937→      // Check for ban messages
   938→      const pageContent = await page.content();
   939→      const banPatterns = ['banned', 'locked', 'suspended', 'temporarily banned'];
   940→      for (const pattern of banPatterns) {
   941→        if (pageContent.toLowerCase().includes(pattern)) {
   942→          return 'banned';
   943→        }
   944→      }
   945→
   946→      return 'disconnected';
   947→    } catch {
   948→      return 'disconnected';
   949→    }
   950→  }

  1001→  private generateSessionId(): string {

  1005→  private generateFallbackQRData(sessionId: string): string {
  1006→    const ref = randomBytes(8).toString('base64url');
  1007→    const publicKey = randomBytes(32).toString('base64url');
  1008→    const clientId = randomBytes(16).toString('base64url');
  1009→    return `${ref},${publicKey},${clientId},${sessionId}`;

  1016→  private generateNoiseKeyPair(): { public: string; private: string } {
  1017→    // In production: use proper Curve25519 key generation
  1018→    // Using HMAC-derived keys for simulation
  1019→    const seed = randomBytes(32);
  1020→    const privateKey = createHmac('sha256', seed).update('noise-private').digest('base64');
  1021→    const publicKey = createHmac('sha256', seed).update('noise-public').digest('base64');
  1022→    return { public: publicKey, private: privateKey };
  1023→  }

  1025→  private generateSignedIdentityKey(): { public: string; private: string } {
  1026→    const seed = randomBytes(32);
  1027→    const privateKey = createHmac('sha256', seed).update('identity-private').digest('base64');
  1028→    const publicKey = createHmac('sha256', seed).update('identity-public').digest('base64');
  1029→    return { public: publicKey, private: privateKey };
  1030→  }

  1032→  private generateSignedPreKey(
  1033→    identityKey: { public: string; private: string }
  1034→  ): { keyId: number; public: string; signature: string } {
  1035→    const keyId = Math.floor(Math.random() * 0xFFFFFF);
  1036→    const preKeyPublic = createHmac('sha256', identityKey.private)
  1037→      .update(`prekey-${keyId}`)
  1038→      .digest('base64');
  1039→    const signature = createHmac('sha256', identityKey.private)
  1040→      .update(`sig-${preKeyPublic}`)
  1041→      .digest('base64');
  1042→    return { keyId, public: preKeyPublic, signature };
  1043→  }

  1053→  private calculateBackoff(attempt: number): number {
  1054→    const baseDelay = this.config.reconnectionDelay;
  1055→    const exponentialDelay = baseDelay * Math.pow(2, attempt - 1);
  1056→    const jitter = Math.random() * 1000;
  1057→    return Math.min(exponentialDelay + jitter, 60000);
  1058→  }

  1072→  async shutdown(): Promise<void> {

  1093→    this.qrSessions.clear();
  1094→    this.activeBrowserSessions.clear();
  1095→    this.stats.activeSessions = 0;
  1096→    this.initialized = false;
  1097→  }
