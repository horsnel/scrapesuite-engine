   197→   197→  /**
   198→   198→   * Detect Google bot detection signals from a response.
   199→   199→   *
   200→   200→   * Analyzes HTTP response headers, body content, status codes, and timing
   201→   201→   * to identify when YouTube/Google has flagged the request as automated.
   202→   202→   *
   203→   203→   * @param response - The HTTP response to analyze
   204→   204→   * @returns Detected bot signals
   205→   205→   */
   206→   206→  detectBotSignals(response: {
   207→   207→    status: number;
   208→   208→    headers: Record<string, string>;
   209→   209→    body?: string;
   210→   210→    url?: string;
   211→   211→    responseTimeMs?: number;
   212→   212→  }): BotDetectionSignals {
   213→   213→    const signals: BotDetectionSignals = {
   214→   214→      unusualTrafficPage: false,
   215→   215→      captchaDetected: false,
   216→   216→      rateLimited: false,
   217→   217→      consentWall: false,
   218→   218→      loginWall: false,
   219→   219→      verificationRedirect: false,
   220→   220→      responseTimeAnomaly: false,
   221→   221→      statusCode: response.status,
   222→   222→      detectedAt: Date.now(),
   223→   223→    };
   224→   224→
   225→   225→    const body = response.body?.toLowerCase() || '';
   226→   226→    const headers = response.headers || {};
   227→   227→
   228→   228→    // Detect "unusual traffic" page
   229→   229→    for (const pattern of this.config.unusualTrafficPatterns) {
   230→   230→      if (body.includes(pattern.toLowerCase())) {
   231→   231→        signals.unusualTrafficPage = true;
   232→   232→        break;
   233→   233→      }
   234→   234→    }
   235→   235→
   236→   236→    // Detect CAPTCHA
   237→   237→    if (
   238→   238→      body.includes('recaptcha') ||
   239→   239→      body.includes('g-recaptcha') ||
   240→   240→      body.includes('sitekey') ||
   241→   241→      body.includes('captcha') ||
   242→   242→      headers['x-recaptcha'] !== undefined
   243→   243→    ) {
   244→   244→      signals.captchaDetected = true;
   245→   245→
   246→   246→      // Determine CAPTCHA type
   247→   247→      if (body.includes('enterprise') || body.includes('recaptchaenterprise')) {
   248→   248→        signals.captchaType = 'enterprise';
   249→   249→      } else if (body.includes('recaptcha-checkbox') || body.includes("recaptcha['fallback']")) {
   250→   250→        signals.captchaType = 'v2';
   251→   251→      } else {
   252→   252→        signals.captchaType = 'v3';
   253→   253→      }
   254→   254→    }
   255→   255→
   256→   256→    // Detect rate limiting
   257→   257→    if (response.status === 429) {
   258→   258→      signals.rateLimited = true;
   259→   259→      signals.rateLimitDetails = {
   260→   260→        remaining: headers['x-ratelimit-remaining']
   261→   261→          ? parseInt(headers['x-ratelimit-remaining'], 10)
   262→   262→          : undefined,
   263→   263→        resetAt: headers['x-ratelimit-reset']
   264→   264→          ? parseInt(headers['x-ratelimit-reset'], 10)
   265→   265→          : undefined,
   266→   266→        retryAfter: headers['retry-after']
   267→   267→          ? parseInt(headers['retry-after'], 10)
   268→   268→          : undefined,
   269→   269→      };
   270→   270→    }
   271→   271→
   272→   272→    // Detect consent wall
   273→   273→    if (
   274→   274→      body.includes('consent.youtube.com') ||
   275→   275→      body.includes('google.com/consent') ||
   276→   276→      body.includes('"consentBumpV2"') ||
   277→   277→      headers['x-consent-redirect'] !== undefined
   278→   278→    ) {
   279→   279→      signals.consentWall = true;
   280→   280→    }
   281→   281→
   282→   282→    // Detect login wall
   283→   283→    if (
   284→   284→      body.includes('accounts.google.com/signin') ||
   285→   285→      body.includes('"signInEndpoint"') ||
   286→   286→      body.includes('login_required')
   287→   287→    ) {
   288→   288→      signals.loginWall = true;
   289→   289→    }
   290→   290→
   291→   291→    // Detect verification redirect
   292→   292→    if (
   293→   293→      response.status === 302 ||
   294→   294→      response.status === 303 ||
   295→   295→      response.status === 307
   296→   296→    ) {
   297→   297→      const location = headers['location'] || '';
   298→   298→      if (
   299→   299→        location.includes('google.com/sorry') ||
   300→   300→        location.includes('google.com/verify') ||
   301→   301→        location.includes('accounts.google.com') ||
   302→   302→        location.includes('recaptcha')
   303→   303→      ) {
   304→   304→        signals.verificationRedirect = true;
   305→   305→      }
   306→   306→    }
   307→   307→
   308→   308→    // Detect response time anomaly (suspiciously fast = CDN cache or bot detection)
   309→   309→    if (response.responseTimeMs !== undefined && response.responseTimeMs < this.config.fastResponseThresholdMs) {
   310→   310→      signals.responseTimeAnomaly = true;
   311→   311→    }
   312→   312→
   313→   313→    // Determine bot classification based on signals
   314→   314→    const signalCount = [
   315→   315→      signals.unusualTrafficPage,
   316→   316→      signals.captchaDetected,
   317→   317→      signals.rateLimited,
   318→   318→      signals.verificationRedirect,
   319→   319→    ].filter(Boolean).length;
   320→   320→
   321→   321→    if (signalCount === 0) {
   322→   322→      signals.botClassification = 'none';
   323→   323→    } else if (signalCount === 1) {
   324→   324→      signals.botClassification = 'suspicious';
   325→   325→    } else if (signalCount === 2) {
   326→   326→      signals.botClassification = 'bot';
   327→   327→    } else {
   328→   328→      signals.botClassification = 'confirmed_bot';
   329→   329→    }
   330→   330→
   331→   331→    // Update internal state if detection found
   332→   332→    if (signalCount > 0) {
   333→   333→      this.consecutiveDetections++;
   334→   334→      this.lastDetectionTime = Date.now();
   335→   335→      this.stats.totalDetections++;
   336→   336→
   337→   337→      if (signals.unusualTrafficPage) this.stats.unusualTrafficEncounters++;
   338→   338→      if (signals.captchaDetected) this.stats.captchaEncounters++;
   339→   339→      if (signals.rateLimited) this.stats.rateLimitEncounters++;
   340→   340→      if (signals.consentWall) this.stats.consentWallEncounters++;
   341→   341→
   342→   342→      logger.warn({
   343→   343→        signals: {
   344→   344→          unusualTraffic: signals.unusualTrafficPage,
   345→   345→          captcha: signals.captchaDetected,
   346→   346→          captchaType: signals.captchaType,
   347→   347→          rateLimited: signals.rateLimited,
   348→   348→          consentWall: signals.consentWall,
   349→   349→          loginWall: signals.loginWall,
   350→   350→          verificationRedirect: signals.verificationRedirect,
   351→   351→          classification: signals.botClassification,
   352→   352→        },
   353→   353→        consecutiveDetections: this.consecutiveDetections,
   354→   354→      }, 'YouTube bot detection signals detected');
   355→   355→    } else {
   356→   356→      this.consecutiveDetections = Math.max(0, this.consecutiveDetections - 1);
   357→   357→    }
   358→   358→
   359→   359→    return signals;
   360→   360→  }

   506→   506→  /**
   507→   507→   * Generate realistic Chrome headers for YouTube requests.
   508→   508→   *
   509→   509→   * Produces a complete set of HTTP headers that mimic a real Chrome browser
   510→   510→   * accessing YouTube, including proper Accept headers, Sec-CH-UA hints,
   511→   511→   * and YouTube-specific headers like X-YouTube-Client-Name.
   512→   512→   *
   513→   513→   * @param options - Optional overrides for header generation
   514→   514→   * @returns Headers object for HTTP requests
   515→   515→   */
   516→   516→  buildEvadeHeaders(options?: {
   517→   517→    clientName?: string;
   518→   518→    clientVersion?: string;
   519→   519→    referer?: string;

   522→   522→    region?: string;

   527→   527→      ...OS_STRINGS.windows,
   528→   528→      ...OS_STRINGS.mac,
   529→   529→      ...OS_STRINGS.linux,

   534→   534→

   536→   536→    let secChPlatform = '"Windows"';
   537→   537→    if (osString.includes('Macintosh')) secChPlatform = '"macOS"';
   538→   538→    else if (osString.includes('Linux')) secChPlatform = '"Linux"';
   539→   539→
   540→   540→    const headers: Record<string, string> = {
   541→   541→      'User-Agent': `Mozilla/5.0 (${osString}) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/${chromeVersion} Safari/537.36`,
   542→   542→      'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.7',
   543→   543→      'Accept-Language': `${language}-${region},${language};q=0.9,en;q=0.8`,
   544→   544→      'Accept-Encoding': 'gzip, deflate, br',
   545→   545→      'Cache-Control': 'max-age=0',
   546→   546→      'Sec-Ch-Ua': `"Chromium";v="${majorVersion}", "Google Chrome";v="${majorVersion}", "Not-A.Brand";v="99"`,
   547→   547→      'Sec-Ch-Ua-Mobile': '?0',
   548→   548→      'Sec-Ch-Ua-Platform': secChPlatform,
   549→   549→      'Sec-Fetch-Dest': 'document',
   550→   550→      'Sec-Fetch-Mode': 'navigate',
   551→   551→      'Sec-Fetch-Site': 'same-origin',

   562→   562→    if (options?.clientVersion) {

   564→   564→    }

   566→   566→    // Add DNT (Do Not Track) — about 25% of real users have it enabled
   567→   567→    if (Math.random() < 0.25) {
   568→   568→      headers['DNT'] = '1';
   569→   569→    }
   570→   570→
   571→   571→    return headers;
   572→   572→  }

   642→   642→  // ---------------------------------------------------------------------------
   643→   643→
   644→   644→  /**
   645→   645→   * Generate valid YouTube visitor data (_visitor_key and visitorData).
   646→   646→   *
   647→   647→   * YouTube uses visitor tracking to correlate requests and detect
   648→   648→   * automated access. This generates realistic visitor identifiers
   649→   649→   * that are consistent with real browser sessions.
   650→   650→   *
   651→   651→   * @returns Visitor data object with key and data string
   652→   652→   */
   653→   653→  generateYouTubeVisitorData(): {
   654→   654→    visitorKey: string;
   655→   655→    visitorData: string;
   656→   656→    visitorId: string;
   657→   657→  } {
   658→   658→    // _visitor_key — A random hex string (typically 16 characters)
   659→   659→    const visitorKey = randomHex(16);
   660→   660→
   661→   661→    // visitorData — Base64-encoded protobuf containing visitor info
   662→   662→    // Real format: base64-encoded string starting with "Cgs"
   663→   663→    // This is a protobuf-encoded message with:
   664→   664→    //   - field 1: visitor ID (string)
   665→   665→    //   - field 2: timestamp
   666→   666→    // We generate a plausible base64 string
   667→   667→
   668→   668→    const visitorId = randomAlphanumeric(11);
   669→   669→
   670→   670→    // Simulate the protobuf structure:
   671→   671→    // The visitorData in YouTube is a base64-encoded proto that contains
   672→   672→    // a visitor ID and timestamp. We build a realistic-looking one.
   673→   673→    const timestamp = Math.floor(Date.now() / 1000);
   674→   674→
   675→   675→    // Build raw protobuf-like bytes for visitorData
   676→   676→    // Field 1 (length-delimited): visitor ID
   677→   677→    // Field 2 (varint): timestamp
   678→   678→    const idBytes = Buffer.from(visitorId, 'utf-8');
   679→   679→    const protoParts: number[] = [];
   680→   680→
   681→   681→    // Field 1: tag = (1 << 3) | 2 = 0x0A (length-delimited)
   682→   682→    protoParts.push(0x0A);
   683→   683→    protoParts.push(idBytes.length);
   684→   684→    for (let i = 0; i < idBytes.length; i++) {
   685→   685→      protoParts.push(idBytes[i]!);
   686→   686→    }
   687→   687→
   688→   688→    // Field 2: tag = (2 << 3) | 0 = 0x10 (varint)
   689→   689→    protoParts.push(0x10);
   690→   690→    // Encode timestamp as varint
   691→   691→    let ts = timestamp;
   692→   692→    while (ts > 0x7F) {
   693→   693→      protoParts.push((ts & 0x7F) | 0x80);
   694→   694→      ts >>>= 7;
   695→   695→    }
   696→   696→    protoParts.push(ts & 0x7F);
   697→   697→
   698→   698→    const visitorData = Buffer.from(protoParts).toString('base64');
   699→   699→
   700→   700→    logger.debug({
   701→   701→      visitorKey: visitorKey.substring(0, 8) + '...',
   702→   702→      visitorId,
   703→   703→    }, 'Generated YouTube visitor data');
   704→   704→
   705→   705→    return {
   706→   706→      visitorKey,
   707→   707→      visitorData,
   708→   708→      visitorId,
   709→   709→    };
   710→   710→  }

   716→   716→  /**
   717→   717→   * Calculate cooldown duration based on consecutive detection count.
   718→   718→   * Uses exponential backoff with a configurable multiplier and cap.
   719→   719→   *
   720→   720→   * @returns Cooldown duration in seconds
   721→   721→   */
   722→   722→  private calculateCooldown(): number {
   723→   723→    const base = this.config.minCooldownSeconds;
   724→   724→    const multiplier = Math.pow(
   725→   725→      this.config.cooldownMultiplier,
   726→   726→      this.consecutiveDetections,
   727→   727→    );
   728→   728→    const cooldown = Math.min(
   729→   729→      base * multiplier,
   730→   730→      this.config.maxCooldownCapSeconds,
   731→   731→    );
   732→   732→
   733→   733→    // Add jitter (±20%) to avoid synchronized retries
   734→   734→    const jitter = cooldown * 0.2 * (Math.random() * 2 - 1);
   735→   735→
   736→   736→    return Math.max(
   737→   737→      this.config.minCooldownSeconds,
   738→   738→      Math.round(cooldown + jitter),
   739→   739→    );
   740→   740→  }

   743→   743→   * Check if currently in a cooldown period.

   745→   745→   * @returns Remaining cooldown in seconds (0 if not cooling down)
   746→   746→   */
   747→   747→  getRemainingCooldown(): number {
   748→   748→    if (this.currentCooldownEnd <= Date.now()) return 0;
   749→   749→    return Math.ceil((this.currentCooldownEnd - Date.now()) / 1000);
