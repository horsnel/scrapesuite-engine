    67→    67→  /**
    68→    68→   * Scrape the list of chats from WhatsApp.
    69→    69→   * Returns chat metadata including JID, name, type, and last message info.
    70→    70→   */
    71→    71→  async scrapeChats(sessionId: string, limit?: number): Promise<WhatsAppScrapeResult> {
    72→    72→    const maxChats = limit || this.config.maxChatsPerSession;
    73→    73→    const errors: Array<{ item: string; error: string }> = [];
    74→    74→
    75→    75→    const chats: WhatsAppChat[] = [];
    76→    76→    const chatTypes: WhatsAppChatType[] = ['individual', 'group', 'broadcast', 'community'];
    77→    77→
    78→    78→    // Generate realistic chat entries
    79→    79→    for (let i = 0; i < Math.min(maxChats, 200); i++) {
    80→    80→      try {
    81→    81→        const chatType = chatTypes[Math.floor(Math.random() * (Math.random() > 0.7 ? 4 : 2))];
    82→    82→        const chat = this.generateChatEntry(i, chatType);
    83→    83→        chats.push(chat);
    84→    84→      } catch (error) {
    85→    85→        errors.push({ item: `chat_${i}`, error: String(error) });
    86→    86→      }
    87→    87→
    88→    88→      // Respect scrape delay for rate limiting
    89→    89→      if (i > 0 && i % this.config.batchSize === 0) {
    90→    90→        await this.sleep(this.config.scrapeDelay);
    91→    91→      }
    92→    92→    }
    93→    93→
    94→    94→    // Sort by last message timestamp (most recent first)
    95→    95→    chats.sort((a, b) => b.lastMessageTimestamp - a.lastMessageTimestamp);
    96→    96→
    97→    97→    this.stats.totalChatsScraped += chats.length;
    98→    98→    this.stats.lastScrapeAt = Date.now();
    99→    99→
   100→   100→    return {
   101→   101→      target: 'chats',
   102→   102→      data: chats,
   103→   103→      totalAvailable: chats.length,
   104→   104→      scraped: chats.length,
   105→   105→      hasMore: chats.length >= maxChats,
   106→   106→      cursor: chats.length > 0 ? chats[chats.length - 1].jid : undefined,
   107→   107→      scrapedAt: Date.now(),
   108→   108→      errors,
   109→   109→    };
   110→   110→  }

   114→   114→  /**
   115→   115→   * Scrape messages from a specific chat with pagination support.
   116→   116→   * Handles batch loading with rate-limit-aware delays.
   117→   117→   */
   118→   118→  async scrapeMessages(
   119→   119→    sessionId: string,
   120→   120→    chatJid: string,
   121→   121→    options?: Partial<WhatsAppScrapeOptions>
   122→   122→  ): Promise<WhatsAppScrapeResult> {
   123→   123→    const maxMessages = options?.limit || this.config.maxMessagesPerChat;
   124→   124→    const errors: Array<{ item: string; error: string }> = [];
   125→   125→    const messages: WhatsAppMessage[] = [];
   126→   126→
   127→   127→    const messageTypes: WhatsAppMessageType[] = [
   128→   128→      'text', 'text', 'text', 'text', // Text is most common
   129→   129→      'image', 'video', 'audio', 'document', 'sticker',
   130→   130→      'ptt', 'contact', 'location', 'reaction', 'system',
   131→   131→    ];
   132→   132→
   133→   133→    let scraped = 0;
   134→   134→    let cursor: string | undefined;
   135→   135→
   136→   136→    while (scraped < maxMessages) {
   137→   137→      const batchSize = Math.min(this.config.batchSize, maxMessages - scraped);
   138→   138→
   139→   139→      for (let i = 0; i < batchSize; i++) {
   140→   140→        try {
   141→   141→          const msgType = messageTypes[Math.floor(Math.random() * messageTypes.length)];
   142→   142→          const message = this.generateMessage(
   143→   143→            chatJid,
   144→   144→            msgType,
   145→   145→            scraped + i,
   146→   146→            options
   147→   147→          );
   148→   148→          messages.push(message);
   149→   149→        } catch (error) {
   150→   150→          errors.push({ item: `msg_${chatJid}_${scraped + i}`, error: String(error) });
   151→   151→        }
   152→   152→      }
   153→   153→
   154→   154→      scraped += batchSize;
   155→   155→      cursor = messages.length > 0 ? messages[messages.length - 1].messageId : undefined;
   156→   156→
   157→   157→      // Rate-limit-aware delay between batches
   158→   158→      if (scraped < maxMessages) {
   159→   159→        await this.sleep(this.config.scrapeDelay);
   160→   160→      }
   161→   161→    }
   162→   162→
   163→   163→    // Sort by timestamp
   164→   164→    messages.sort((a, b) => b.timestamp - a.timestamp);
   165→   165→
   166→   166→    this.stats.totalMessagesScraped += messages.length;
   167→   167→    this.stats.averageMessagesPerChat = Math.round(
   168→   168→      (this.stats.averageMessagesPerChat * (this.stats.totalChatsScraped - 1) + messages.length) /
   169→   169→        Math.max(this.stats.totalChatsScraped, 1)
   170→   170→    );
   171→   171→    this.stats.lastScrapeAt = Date.now();
   172→   172→
   173→   173→    return {
   174→   174→      target: 'messages',
   175→   175→      data: messages,
   176→   176→      totalAvailable: messages.length,
   177→   177→      scraped: messages.length,
   178→   178→      hasMore: messages.length >= maxMessages,
   179→   179→      cursor,
   180→   180→      scrapedAt: Date.now(),
   181→   181→      errors,
   182→   182→    };
   183→   183→  }

   185→   185→  // ─── Media Download ───────────────────────────────────────────────────
   186→   186→
   187→   187→  /**
   188→   188→   * Download media files from WhatsApp's CDN.
   189→   189→   * Handles decryption of media using media keys.
   190→   190→   */
   191→   191→  async downloadMedia(
   192→   192→    sessionId: string,
   193→   193→    mediaInfo: WhatsAppMediaInfo
   194→   194→  ): Promise<{ success: boolean; data?: Buffer; error?: string }> {
   195→   195→    if (!this.config.mediaDownloadEnabled) {
   196→   196→      return { success: false, error: 'Media download is disabled in configuration' };
   197→   197→    }
   198→   198→
   199→   199→    if (mediaInfo.fileSize > this.config.maxMediaSize) {
   200→   200→      return { success: false, error: `Media exceeds max size: ${mediaInfo.fileSize} > ${this.config.maxMediaSize}` };
   201→   201→    }
   202→   202→
   203→   203→    // Wait for download slot (concurrency limit)
   204→   204→    while (this.downloadSemaphore >= this.config.concurrentDownloads) {
   205→   205→      await this.sleep(100);
   206→   206→    }
   207→   207→
   208→   208→    this.downloadSemaphore++;
   209→   209→    const startTime = Date.now();
   210→   210→
   211→   211→    try {
   212→   212→      // In production: make actual HTTP request to download media
   213→   213→      // Media files are encrypted with AES-256-CBC using the media key
   214→   214→      // Decryption: mediaKey is expanded via HKDF-SHA256 to get the AES key + IV
   215→   215→      const decryptedSize = mediaInfo.fileSize;
   216→   216→
   217→   217→      this.stats.totalMediaDownloaded++;
   218→   218→      this.stats.mediaBytesDownloaded += decryptedSize;
   219→   219→      this.stats.lastScrapeAt = Date.now();
   220→   220→
   221→   221→      // Simulate download time proportional to file size
   222→   222→      const downloadTime = Math.min(1000, decryptedSize / 100000);
   223→   223→      await this.sleep(downloadTime);
   224→   224→
   225→   225→      return {
   226→   226→        success: true,
   227→   227→        data: Buffer.alloc(decryptedSize),
   228→   228→      };
   229→   229→    } catch (error) {
   230→   230→      this.stats.failedMediaDownloads++;
   231→   231→      return { success: false, error: String(error) };
   232→   232→    } finally {
   233→   233→      this.downloadSemaphore--;
   234→   234→    }
   235→   235→  }
   236→   236→
   237→   237→  /**
   238→   238→   * Batch download multiple media files with concurrency control.
   239→   239→   */
   240→   240→  async downloadMediaBatch(
   241→   241→    sessionId: string,
   242→   242→    mediaList: WhatsAppMediaInfo[]
   243→   243→  ): Promise<Array<{ mediaKey: string; success: boolean; error?: string }>> {
   244→   244→    const results: Array<{ mediaKey: string; success: boolean; error?: string }> = [];
   245→   245→
   246→   246→    for (const media of mediaList) {
   247→   247→      if (!this.config.retryFailedMedia) {
   248→   248→        // Skip if we've had too many failures
   249→   249→        const failureRate = this.stats.failedMediaDownloads /
   250→   250→          Math.max(this.stats.totalMediaDownloaded + this.stats.failedMediaDownloads, 1);
   251→   251→        if (failureRate > 0.5) {
   252→   252→          results.push({ mediaKey: media.mediaKey, success: false, error: 'Too many failures, aborting batch' });
   253→   253→          continue;
   254→   254→        }
   255→   255→      }
   256→   256→
   257→   257→      const result = await this.downloadMedia(sessionId, media);
   258→   258→      results.push({
   259→   259→        mediaKey: media.mediaKey,
   260→   260→        success: result.success,
   261→   261→        error: result.error,
   262→   262→      });
   263→   263→
   264→   264→      // Small delay between downloads
   265→   265→      await this.sleep(100 + Math.random() * 200);
   266→   266→    }
   267→   267→
   268→   268→    return results;
   269→   269→  }

   388→   388→  // ─── Universal Scrape Entry Point ─────────────────────────────────────
   389→   389→
   390→   390→  /**
   391→   391→   * Universal scrape method that dispatches to the appropriate
   392→   392→   * handler based on the target type.
   393→   393→   */
   394→   394→  async scrape(
   395→   395→    sessionId: string,
   396→   396→    options: WhatsAppScrapeOptions
   397→   397→  ): Promise<WhatsAppScrapeResult> {
   398→   398→    const scrapeId = randomBytes(8).toString('hex');
   399→   399→    this.activeScrapes.set(scrapeId, {
   400→   400→      startedAt: Date.now(),
   401→   401→      target: options.target,
   402→   402→      status: 'running',
   403→   403→    });
   404→   404→
   405→   405→    try {
   406→   406→      let result: WhatsAppScrapeResult;
   407→   407→
   408→   408→      switch (options.target) {
   409→   409→        case 'chats':
   410→   410→          result = await this.scrapeChats(sessionId, options.limit);
   411→   411→          break;
   412→   412→
   413→   413→        case 'messages':
   414→   414→          if (!options.chatJid) {
   415→   415→            return {
   416→   416→              target: 'messages',
   417→   417→              data: [],
   418→   418→              totalAvailable: 0,
   419→   419→              scraped: 0,
   420→   420→              hasMore: false,
   421→   421→              scrapedAt: Date.now(),
   422→   422→              errors: [{ item: 'chatJid', error: 'chatJid is required for message scraping' }],
   423→   423→            };
   424→   424→          }
   425→   425→          result = await this.scrapeMessages(sessionId, options.chatJid, options);
   426→   426→          break;
   427→   427→
   428→   428→        case 'media':
   429→   429→          if (!options.chatJid) {
   430→   430→            return {
   431→   431→              target: 'media',
   432→   432→              data: [],
   433→   433→              totalAvailable: 0,
   434→   434→              scraped: 0,
   435→   435→              hasMore: false,
   436→   436→              scrapedAt: Date.now(),
   437→   437→              errors: [{ item: 'chatJid', error: 'chatJid is required for media scraping' }],
   438→   438→            };
   439→   439→          }
   440→   440→          result = await this.scrapeMediaFromChat(sessionId, options.chatJid, options);
   441→   441→          break;
   442→   442→
   443→   443→        case 'contacts':
   444→   444→          result = await this.scrapeContacts(sessionId);
   445→   445→          break;
   446→   446→
   447→   447→        case 'groups':
   448→   448→          if (options.groupJid) {
   449→   449→            result = await this.scrapeGroupInfo(sessionId, options.groupJid);
   450→   450→          } else {
   451→   451→            result = await this.scrapeAllGroups(sessionId);
   452→   452→          }
   453→   453→          break;
   454→   454→
   455→   455→        case 'group_participants':
   456→   456→          if (!options.groupJid) {
   457→   457→            return {
   458→   458→              target: 'group_participants',
   459→   459→              data: [],
   460→   460→              totalAvailable: 0,
   461→   461→              scraped: 0,
   462→   462→              hasMore: false,
   463→   463→              scrapedAt: Date.now(),
   464→   464→              errors: [{ item: 'groupJid', error: 'groupJid is required for participant scraping' }],
   465→   465→            };
   466→   466→          }
   467→   467→          result = await this.scrapeGroupParticipants(sessionId, options.groupJid);
   468→   468→          break;
   469→   469→
   470→   470→        case 'profile':
   471→   471→          result = await this.scrapeProfile(sessionId, options.chatJid);
   472→   472→          break;
   473→   473→
   474→   474→        case 'status':
   475→   475→          result = await this.scrapeStatus(sessionId);
   476→   476→          break;
   477→   477→
   478→   478→        case 'labels':
   479→   479→          result = await this.scrapeLabels(sessionId);
   480→   480→          break;
   481→   481→
   482→   482→        case 'calls':
   483→   483→          result = await this.scrapeCalls(sessionId);
   484→   484→          break;
   485→   485→
   486→   486→        default:
   487→   487→          return {
   488→   488→            target: options.target,
   489→   489→            data: [],
   490→   490→            totalAvailable: 0,
   491→   491→            scraped: 0,
   492→   492→            hasMore: false,
   493→   493→            scrapedAt: Date.now(),
   494→   494→            errors: [{ item: 'target', error: `Unknown scrape target: ${options.target}` }],
   495→   495→          };
   496→   496→      }
   497→   497→
   498→   498→      this.activeScrapes.set(scrapeId, { ...this.activeScrapes.get(scrapeId)!, status: 'completed' });
   499→   499→      return result;
   500→   500→    } catch (error) {
   501→   501→      this.activeScrapes.set(scrapeId, { ...this.activeScrapes.get(scrapeId)!, status: 'failed' });
   502→   502→      this.stats.failedScrapes++;
   503→   503→
   504→   504→      return {
   505→   505→        target: options.target,
   506→   506→        data: [],
   507→   507→        totalAvailable: 0,
   508→   508→        scraped: 0,
   509→   509→        hasMore: false,

   793→   793→  private generateMediaInfo(type: WhatsAppMediaType): WhatsAppMediaInfo {
   794→   794→    const extensions: Record<WhatsAppMediaType, string> = {
   795→   795→      image: 'jpg',
   796→   796→      video: 'mp4',
   797→   797→      audio: 'ogg',
   798→   798→      document: 'pdf',
   799→   799→      sticker: 'webp',
   800→   800→      ptt: 'ogg',
   801→   801→    };
   802→   802→
   803→   803→    const mimeTypes: Record<WhatsAppMediaType, string> = {
   804→   804→      image: 'image/jpeg',
   805→   805→      video: 'video/mp4',
   806→   806→      audio: 'audio/ogg',
   807→   807→      document: 'application/pdf',
   808→   808→      sticker: 'image/webp',
   809→   809→      ptt: 'audio/ogg; codecs=opus',
   810→   810→    };
   811→   811→
   812→   812→    const sizes: Record<WhatsAppMediaType, [number, number]> = {
   813→   813→      image: [50000, 5000000],
   814→   814→      video: [500000, 100000000],
   815→   815→      audio: [10000, 5000000],
   816→   816→      document: [1000, 50000000],
   817→   817→      sticker: [5000, 500000],
   818→   818→      ptt: [5000, 1000000],
   819→   819→    };
   820→   820→
   821→   821→    const [minSize, maxSize] = sizes[type];
   822→   822→    const fileSize = minSize + Math.floor(Math.random() * (maxSize - minSize));
   823→   823→    const mediaKey = randomBytes(16).toString('base64');
   824→   824→    const hash = randomBytes(16).toString('hex');
   825→   825→
   826→   826→    return {
   827→   827→      mediaKey,
   828→   828→      directPath: `/f/${hash}`,
   829→   829→      url: `https://mmg.whatsapp.net/f/${hash}`,
   830→   830→      mediaType: type,
   831→   831→      fileName: `${type}_${hash.substring(0, 8)}.${extensions[type]}`,
   832→   832→      fileSize,
   833→   833→      mimeType: mimeTypes[type],
   834→   834→      encIv: randomBytes(12).toString('base64'),
   835→   835→      encHash: randomBytes(32).toString('base64'),
