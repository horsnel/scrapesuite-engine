   254→   254→   254→      }

   259→   259→   259→      );
   260→   260→   260→    } catch (err: unknown) {
   261→   261→   261→      const message = err instanceof Error ? err.message : String(err);
   262→   262→   262→      this.stats.extractionFailures++;

   264→   264→   264→    }

   318→   318→   318→      });

   388→   388→   388→   *
   389→   389→   389→   * Checks the Redis cache first before downloading.
   390→   390→   390→   */

   392→   392→   392→    // Check Redis cache first
   393→   393→   393→    try {
   394→   394→   394→      const cached = await cacheGet<{ js: string; version: string; extractedAt: number }>(
   395→   395→   395→        XBogusCDNExtractor.ALGORITHM_REDIS_KEY,
   396→   396→   396→      );

   398→   398→   398→        this.stats.cacheHits++;

   402→   402→   402→          js: cached.js,
   403→   403→   403→          extractedAt: cached.extractedAt,
   404→   404→   404→          version,
   405→   405→   405→        });
   406→   406→   406→        return cached.js;
   407→   407→   407→      }
   408→   408→   408→    } catch (err: unknown) {
   409→   409→   409→      logger.debug({ error: (err as Error).message }, 'Redis cache read failed -- proceeding to download');
   410→   410→   410→    }

   415→   415→   415→        this.stats.cacheHits++;
   416→   416→   416→        logger.info({ version }, 'Algorithm cache hit (memory)');
   417→   417→   417→        return entry.js;
   418→   418→   418→      }
   419→   419→   419→    }
   420→   420→   420→
   421→   421→   421→    this.stats.cacheMisses++;
   422→   422→   422→
   423→   423→   423→    // Download and scan each candidate script
   424→   424→   424→    const algorithmChunks: string[] = [];
   425→   425→   425→
   426→   426→   426→    for (const url of scriptUrls) {
   427→   427→   427→      try {
   428→   428→   428→        const response = await fetch(url, {
   429→   429→   429→          headers: {
   430→   430→   430→            'User-Agent': DEFAULT_UA,
   431→   431→   431→            'Accept': '*/*',
   432→   432→   432→            'Accept-Language': 'en-US,en;q=0.9',
   433→   433→   433→            Referer: 'https://www.tiktok.com/',
   434→   434→   434→          },
   435→   435→   435→          signal: AbortSignal.timeout(15_000),
   436→   436→   436→        });
   437→   437→   437→
   438→   438→   438→        if (!response.ok) continue;
   439→   439→   439→
   440→   440→   440→        const js = await response.text();
   441→   441→   441→
   442→   442→   442→        // Test if this script contains X-Bogus algorithm indicators
   443→   443→   443→        if (this.containsAlgorithm(js)) {
   444→   444→   444→          algorithmChunks.push(js);
   445→   445→   445→          logger.debug({ url: url.substring(0, 120), size: js.length }, 'Identified X-Bogus algorithm script');
   446→   446→   446→        }
   447→   447→   447→      } catch (err: unknown) {
   448→   448→   448→        logger.debug({ url: url.substring(0, 120), error: (err as Error).message }, 'Failed to download candidate script');
   449→   449→   449→      }
   450→   450→   450→    }
   451→   451→   451→
   452→   452→   452→    if (algorithmChunks.length === 0) {
   453→   453→   453→      logger.warn({ urlCount: scriptUrls.length }, 'No algorithm scripts found among candidates');
   454→   454→   454→      return null;
   455→   455→   455→    }
   456→   456→   456→
   457→   457→   457→    return algorithmChunks.join('\n;//---SEPARATOR---;\n');
   458→   458→   458→  }

   470→   470→   470→        if (matchCount >= 2) return true;

   472→   472→   472→    }

   607→   607→   607→        {

   609→   609→   609→          queryString: params.queryString,
   610→   610→   610→          userAgent: params.userAgent,
   611→   611→   611→          timestamp: params.timestamp,
   612→   612→   612→          body: params.body || '',
   613→   613→   613→        },
   614→   614→   614→      );
   615→   615→   615→
   616→   616→   616→      await page.close();
   617→   617→   617→
   618→   618→   618→      if (typeof xBogus === 'string' && xBogus.length > 10) {
   619→   619→   619→        return xBogus;
   620→   620→   620→      }
   621→   621→   621→
   622→   622→   622→      throw new Error('Sandbox execution did not produce a valid X-Bogus signature');
   623→   623→   623→    } catch (err: unknown) {
   624→   624→   624→      const message = err instanceof Error ? err.message : String(err);
   625→   625→   625→      this.stats.sandboxFailures++;
   626→   626→   626→      logger.error({ error: message }, 'Sandbox execution failed');
   627→   627→   627→      throw err;
   628→   628→   628→    } finally {
   629→   629→   629→      if (context) {
   630→   630→   630→        try { await context.close(); } catch {}
   631→   631→   631→      }
   632→   632→   632→    }
   633→   633→   633→  }

   639→   639→   639→  /**
   640→   640→   640→   * Detect the X-Bogus algorithm version from the extracted JavaScript.
   641→   641→   641→   * Looks for version prefixes like "DFSzswVO" and other identifiers.
   642→   642→   642→   */
   643→   643→   643→  private async detectAlgorithmVersion(js: string): Promise<string> {
   644→   644→   644→    for (const pattern of VERSION_PATTERNS) {
   645→   645→   645→      const match = js.match(pattern);
   646→   646→   646→      if (match) {
   647→   647→   647→        return match[0].replace(/"/g, '');
   648→   648→   648→      }
   649→   649→   649→    }
   650→   650→   650→
   651→   651→   651→    // Fallback: hash the first 512 bytes to create a stable version id
   652→   652→   652→    const sample = js.substring(0, 512);
   653→   653→   653→    let hash = 0;
   654→   654→   654→    for (let i = 0; i < sample.length; i++) {
   655→   655→   655→      const chr = sample.charCodeAt(i);
   656→   656→   656→      hash = ((hash << 5) - hash + chr) | 0;
   657→   657→   657→    }
   658→   658→   658→    return `custom-${Math.abs(hash).toString(16).slice(0, 8)}`;
   659→   659→   659→  }

   665→   665→   665→  /**
   666→   666→   666→   * Retrieve the algorithm JS from in-memory cache, Redis, or by performing
   667→   667→   667→   * a fresh CDN extraction.
   668→   668→   668→   */
   669→   669→   669→  private async getAlgorithm(): Promise<CachedAlgorithm | null> {
   670→   670→   670→    // Check in-memory cache (any version, as long as it's not expired)
   671→   671→   671→    for (const [version, entry] of this.algorithmCache.entries()) {
   672→   672→   672→      if (Date.now() - entry.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   673→   673→   673→        this.stats.cacheHits++;
   674→   674→   674→        return entry;
   675→   675→   675→      }
   676→   676→   676→    }
   677→   677→   677→
   678→   678→   678→    // Check Redis cache
   679→   679→   679→    try {
   680→   680→   680→      const cached = await cacheGet<{ js: string; version: string; extractedAt: number }>(
   681→   681→   681→        XBogusCDNExtractor.ALGORITHM_REDIS_KEY,
   682→   682→   682→      );
   683→   683→   683→      if (cached && cached.js && Date.now() - cached.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   684→   684→   684→        this.stats.cacheHits++;
   685→   685→   685→        this.algorithmCache.set(cached.version, {
   686→   686→   686→          js: cached.js,
   687→   687→   687→          extractedAt: cached.extractedAt,
   688→   688→   688→          version: cached.version,
   689→   689→   689→        });
   690→   690→   690→        return this.algorithmCache.get(cached.version) ?? null;
   691→   691→   691→      }
   692→   692→   692→    } catch (err: unknown) {
   693→   693→   693→      logger.debug({ error: (err as Error).message }, 'Redis cache read failed');
   694→   694→   694→    }
   695→   695→   695→
   696→   696→   696→    // Fresh extraction required
   697→   697→   697→    this.stats.cacheMisses++;
   698→   698→   698→    await this.refreshAlgorithm();
   699→   699→   699→
   700→   700→   700→    // Return from in-memory cache after extraction
   701→   701→   701→    for (const [, entry] of this.algorithmCache.entries()) {
   702→   702→   702→      if (Date.now() - entry.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   703→   703→   703→        return entry;
   704→   704→   704→      }
   705→   705→   705→    }
   706→   706→   706→
   707→   707→   707→    return null;
   708→   708→   708→  }
