    35→// --- Request Deduplication ----------------------------------------------------
    36→
    37→interface PendingRequest {
    38→  promise: Promise<ScrapeJobResult>;
    39→  resolve: (result: ScrapeJobResult) => void;
    40→  reject: (error: Error) => void;
    41→}
    42→
    43→const inflightRequests = new Map<string, PendingRequest>();
    44→
    45→function getDedupeKey(data: ScrapeJobData): string {
    46→  // Dedupe by URL + strategy + extraction instruction + session
    47→  return `${data.url}:${data.strategy || 'auto'}:${data.extract || ''}:${data.proxyTier || ''}:${data.proxyCountry || ''}:${data.proxyCity || ''}:${data.sessionId || ''}`;
    48→}

    50→// --- Orchestrator -------------------------------------------------------------
    51→
    52→export class Orchestrator {
    53→  /**
    54→   * Process a scrape job through the full pipeline:
    55→   *   dedupe → robots → cache → session → intelligence → adaptive rate limit →
    56→   *   proxy → stealth → fetch → smart retry → record → cache →
    57→   *   template extraction → structured → NL extract → captcha →
    58→   *   output pipeline → bill → persist
    59→   */
    60→  async processJob(data: ScrapeJobData): Promise<ScrapeJobResult> {
    61→    // -- Request Deduplication --------------------------------------------
    62→    const dedupeKey = getDedupeKey(data);
    63→    const existing = inflightRequests.get(dedupeKey);
    64→    if (existing) {
    65→      logger.info({ jobId: data.jobId, url: data.url }, 'Deduped -- waiting for identical in-flight request');
    66→      return existing.promise;
    67→    }
    68→
    69→    const pending: Partial<PendingRequest> = {};
    70→    pending.promise = new Promise<ScrapeJobResult>((resolve, reject) => {
    71→      pending.resolve = resolve;
    72→      pending.reject = reject;
    73→    });
    74→    inflightRequests.set(dedupeKey, pending as PendingRequest);
    75→
    76→    try {
    77→      const result = await this._processJob(data);
    78→      pending.resolve!(result);
    79→      return result;
    80→    } catch (err: any) {
    81→      const result: ScrapeJobResult = {
    82→        jobId: data.jobId,
    83→        status: 'failed',
    84→        strategy: 'http',
    85→        creditsUsed: 0,
    86→        creditsCharged: 0,
    87→        statusCode: 502,
    88→        responseMs: 0,
    89→        error: err.message,
    90→        cached: false,
    91→      };
    92→      pending.resolve!(result);
    93→      return result;
    94→    } finally {
    95→      inflightRequests.delete(dedupeKey);
    96→    }
    97→  }
    98→
    99→  private async _processJob(data: ScrapeJobData): Promise<ScrapeJobResult> {
   100→    const startTime = Date.now();
   101→    const domain = data.domain;
   102→    let strategy: JobStrategy = 'http';
   103→    let creditsUsed = 0;
   104→    let creditsCharged = 0;
   105→    let cached = false;
   106→    let html = '';
   107→    let statusCode = 200;
   108→    let extracted: Record<string, any> | undefined;
   109→    let structuredData: Record<string, any> | undefined;
   110→    let proxyId: string | undefined;
   111→    let proxyCountry: string | undefined;
   112→    let proxyUrl: string | undefined;
   113→    let captchaSolved = false;
   114→    let error: string | undefined;
   115→    let retryCount = 0;
   116→    let bandwidthBytes = 0;
   117→    let responseHeaders: Record<string, string> = {};
   118→    let finalUrl = data.url;
   119→    let outputCompressionRatio: number | undefined;
   120→    let appliedOutputFormat: OutputFormat | undefined;
   121→
   122→    // Session state -- populated when sessionId is provided
   123→    let sessionState: import('../session').SessionState | null = null;
   124→
   125→    try {
   126→      // -- Step 1: Check robots.txt --------------------------------------
   127→      if (data.respectRobotsTxt !== false) {
   128→        const robotsCheck = await robotsParser.isAllowed(data.url, 'ScrapeSuite');
   129→        if (!robotsCheck.allowed) {
   130→          logger.info({ jobId: data.jobId, url: data.url }, 'Blocked by robots.txt');
   131→          return {
   132→            jobId: data.jobId,
   133→            status: 'failed',
   134→            strategy: 'cache',
   135→            creditsUsed: 0,
   136→            creditsCharged: 0,
   137→            statusCode: 403,
   138→            responseMs: Date.now() - startTime,
   139→            error: 'URL disallowed by robots.txt',
   140→            cached: false,
   141→          };
   142→        }
   143→      }
   144→
   145→      // -- Step 2: Check Redis cache -------------------------------------
   146→      const cachedResult = await this.checkCache(data.url);
   147→      if (cachedResult) {
   148→        cached = true;
   149→        creditsUsed = 0;
   150→        creditsCharged = 0;
   151→        html = cachedResult.html;
   152→        statusCode = cachedResult.statusCode;
   153→        strategy = 'cache';
   154→        logger.info({ jobId: data.jobId, url: data.url, strategy: 'cache' }, 'Cache hit');
   155→      }
   156→
   157→      if (!cached) {
   158→        // -- Step 3: Session Manager Integration -------------------------
   159→        // If a sessionId is provided, retrieve or create the session state
   160→        // to reuse the same proxy and fingerprint for geo-consistency.
   161→        if (data.sessionId) {
   162→          try {
   163→            sessionState = await sessionManager.getSession(data.sessionId);
   164→            if (sessionState) {
   165→              // Refresh the session TTL on use (sliding window)
   166→              sessionState = await sessionManager.refreshSession(data.sessionId);
   167→              if (sessionState) {
   168→                logger.info(
   169→                  { jobId: data.jobId, sessionId: data.sessionId, proxyId: sessionState.proxyId, proxyCountry: sessionState.proxyCountry },
   170→                  'Session found -- using session proxy and fingerprint',
   171→                );
   172→                // Use the session's proxy and fingerprint
   173→                proxyUrl = sessionState.proxyUrl;
   174→                proxyId = sessionState.proxyId;
   175→                proxyCountry = sessionState.proxyCountry;
   176→              }
   177→            } else {
   178→              // Session doesn't exist or expired -- create a new one
   179→              logger.info(
   180→                { jobId: data.jobId, sessionId: data.sessionId },
   181→                'Session not found or expired -- creating new session',
   182→              );
   183→              try {
   184→                sessionState = await sessionManager.createSession({
   185→                  userId: data.userId,
   186→                  proxyTier: (data.proxyTier as any) || 'residential',
   187→                  proxyCountry: data.proxyCountry,
   188→                  proxyCity: data.proxyCity,
   189→                  proxyAsn: data.proxyAsn,
   190→                  domain,
   191→                });
   192→                proxyUrl = sessionState.proxyUrl;
   193→                proxyId = sessionState.proxyId;
   194→                proxyCountry = sessionState.proxyCountry;
   195→                logger.info(
   196→                  { jobId: data.jobId, sessionId: sessionState.sessionId, proxyId, proxyCountry },
   197→                  'New session created with proxy',
   198→                );
   199→              } catch (sessionErr: any) {
   200→                logger.warn(
   201→                  { jobId: data.jobId, sessionId: data.sessionId, error: sessionErr.message },
   202→                  'Failed to create session -- proceeding without session',
   203→                );
   204→              }
   205→            }
   206→          } catch (sessionErr: any) {
   207→            logger.warn(
   208→              { jobId: data.jobId, sessionId: data.sessionId, error: sessionErr.message },
   209→              'Session manager error -- proceeding without session',
   210→            );
   211→          }
   212→        }
   213→
   214→        // -- Step 4: Consult Domain Intelligence -------------------------
   215→        const recommendation = await domainIntelligence.recommendStrategy(domain);
   216→
   217→        // -- Step 5: Acquire adaptive rate limit token -------------------
   218→        // Uses the new AdaptiveRateLimiter instead of the simple Redis counter.
   219→        try {
   220→          const rateResult = await adaptiveRateLimiter.acquireToken(domain);
   221→          if (!rateResult.allowed) {
   222→            logger.info(
   223→              { jobId: data.jobId, domain, waitMs: rateResult.waitMs, currentRps: rateResult.currentRps },
   224→              'Rate limited -- waiting before proceeding',
   225→            );
   226→            await this.sleep(rateResult.waitMs);
   227→            // Try one more time after waiting
   228→            const retryResult = await adaptiveRateLimiter.acquireToken(domain);
   229→            if (!retryResult.allowed) {
   230→              logger.warn(
   231→                { jobId: data.jobId, domain, waitMs: retryResult.waitMs },
   232→                'Rate limit token still not available after wait -- proceeding anyway',
   233→              );
   234→            }
   235→          }
   236→        } catch (rateErr: any) {
   237→          logger.warn(
   238→            { jobId: data.jobId, domain, error: rateErr.message },
   239→            'Adaptive rate limiter error -- falling back to simple rate limit',
   240→          );
   241→          // Fallback to the old simple rate limiter
   242→          await this.acquireSimpleRateLimitToken(domain, recommendation.safeRps);
   243→        }
   244→
   245→        // -- Step 6: Get proxy from Smart IP Pool (v3.1+) -----------------
   246→        // Uses the unified mega-pool with reputation tracking, IP cooling,
   247→        // auto-scaling, and multi-provider aggregation. Falls back to
   248→        // legacy getProxy() if the smart pool fails.
   249→        // If session already provided a proxy, skip this step.
   250→        if (!proxyUrl) {
   251→          const proxySelection = await proxyManager.getSmartProxy(
   252→            domain,
   253→            (data.proxyTier as any) || recommendation.proxyTier,
   254→            data.proxyCountry,
   255→            {
   256→              city: data.proxyCity,
   257→              asn: data.proxyAsn,
   258→              sessionId: data.sessionId,
   259→              // For domains known to have anti-bot, require IPs with good reputation
   260→              requireReputation: recommendation.strategy === 'stealth-browser',
   261→            },
   262→          );
   263→
   264→          proxyUrl = proxySelection?.proxyUrl;
   265→          proxyId = proxySelection?.proxyId;
   266→          proxyCountry = proxySelection?.country;
   267→        }
   268→
   269→        // -- Step 7: Execute fetch (with smart retry logic) --------------
   270→        const effectiveStrategy: JobStrategy =
   271→          data.strategy === 'auto' || !data.strategy
   272→            ? recommendation.strategy
   273→            : (data.strategy as JobStrategy);
   274→
   275→        // If renderJs is requested, force browser strategy
   276→        const forceBrowser = data.renderJs === true;
   277→        let actualStrategy: JobStrategy = forceBrowser && effectiveStrategy === 'http'
   278→          ? 'browser'
   279→          : effectiveStrategy;
   280→
   281→        // If session has a fingerprint profile, use it for stealth browser requests
   282→        // (handled inside fetchViaStealthBrowser / fetchViaHttp via stealthEngine.getRandomProfile())
   283→
   284→        let fetchResult = await this.executeFetch(
   285→          actualStrategy, data.url, domain, proxyUrl,
   286→          data.waitForSelector, data.timeout, data.headers,
   287→        );
   288→
   289→        html = fetchResult.html;
   290→        statusCode = fetchResult.statusCode;
   291→        strategy = fetchResult.strategy;
   292→        bandwidthBytes = Buffer.byteLength(html, 'utf8');
   293→        responseHeaders = fetchResult.headers || {};
   294→        finalUrl = fetchResult.finalUrl || data.url;
   295→
   296→        // Record the initial response to the adaptive rate limiter
   297→        const isSuccess = statusCode >= 200 && statusCode < 400;
   298→        try {
   299→          await adaptiveRateLimiter.recordResponse(domain, {
   300→            success: isSuccess,
   301→            statusCode,
   302→            responseMs: Date.now() - startTime,
   303→          });
   304→        } catch (rateErr: any) {
   305→          logger.debug({ domain, error: rateErr.message }, 'Failed to record response to adaptive rate limiter');
   306→        }
   307→
   308→        // -- Step 8: Smart Retry -- Empty HTML → browser -----------------
   309→        if (strategy === 'http' && this.isEmptyHtml(html) && retryCount < MAX_RETRIES) {
   310→          retryCount++;
   311→          logger.info({ jobId: data.jobId, url: data.url, retry: retryCount }, 'Empty HTML -- escalating to browser');
   312→
   313→          // Rotate fingerprint before retry
   314→          const rotatedProfile = stealthEngine.getRandomProfile();
   315→          logger.debug(
   316→            { jobId: data.jobId, userAgent: rotatedProfile.userAgent },
   317→            'Rotated fingerprint profile for retry',
   318→          );
   319→
   320→          fetchResult = await this.executeFetch(
   321→            'browser', data.url, domain, proxyUrl,
   322→            data.waitForSelector, data.timeout, data.headers,
   323→          );
   324→          html = fetchResult.html;
   325→          statusCode = fetchResult.statusCode;
   326→          strategy = fetchResult.strategy;
   327→          bandwidthBytes = Buffer.byteLength(html, 'utf8');
   328→          responseHeaders = fetchResult.headers || {};
   329→
   330→          // Record retry outcome
   331→          try {
   332→            await adaptiveRateLimiter.recordResponse(domain, {
   333→              success: statusCode >= 200 && statusCode < 400,
   334→              statusCode,
   335→              responseMs: Date.now() - startTime,
   336→            });
   337→          } catch {}
   338→        }
   339→
   340→        // -- Step 9: Smart Retry -- 429 Rate Limited ---------------------
   341→        if (statusCode === 429 && retryCount < MAX_RETRIES) {
   342→          retryCount++;
   343→          logger.info(
   344→            { jobId: data.jobId, url: data.url, retry: retryCount, currentRps: (await adaptiveRateLimiter.getDomainStatus(domain)).currentRps },
   345→            '429 rate-limited -- waiting cooldown then retrying with different proxy',
   346→          );
   347→
   348→          // Wait based on adaptive rate limiter's cooldown
   349→          const domainStatus = await adaptiveRateLimiter.getDomainStatus(domain).catch(() => null);
   350→          const cooldownMs = domainStatus?.cooldownRemainingMs || 5_000;
   351→          logger.info({ jobId: data.jobId, domain, cooldownMs }, 'Waiting for rate limit cooldown');
   352→          await this.sleep(cooldownMs + 500); // Add small buffer
   353→
   354→          // Get a DIFFERENT proxy for the retry
   355→          const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   356→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   357→
   358→          // Rotate fingerprint
   359→          stealthEngine.getRandomProfile();
   360→
   361→          // Re-acquire rate limit token after cooldown
   362→          try {
   363→            const retryToken = await adaptiveRateLimiter.acquireToken(domain);
   364→            if (!retryToken.allowed) {
   365→              await this.sleep(retryToken.waitMs);
   366→            }
   367→          } catch {}
   368→
   369→          // Retry with stealth browser (more resilient to rate limits)
   370→          fetchResult = await this.executeFetch(
   371→            'stealth-browser', data.url, domain, retryProxyUrl,
   372→            data.waitForSelector, data.timeout, data.headers,
   373→          );
   374→
   375→          if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   376→            html = fetchResult.html;
   377→            statusCode = fetchResult.statusCode;
   378→            strategy = fetchResult.strategy;
   379→            bandwidthBytes = Buffer.byteLength(html, 'utf8');
   380→            responseHeaders = fetchResult.headers || {};
   381→            if (retryProxy) {
   382→              proxyId = retryProxy.proxyId;
   383→              proxyCountry = retryProxy.country;
   384→              proxyUrl = retryProxyUrl;
   385→            }
   386→          }
   387→
   388→          // Record retry outcome
   389→          try {
   390→            await adaptiveRateLimiter.recordResponse(domain, {
   391→              success: fetchResult.statusCode >= 200 && fetchResult.statusCode < 400,
   392→              statusCode: fetchResult.statusCode,
   393→              responseMs: Date.now() - startTime,
   394→            });
   395→          } catch {}
   396→        }
   397→
   398→        // -- Step 10: Smart Retry -- 403 with Anti-bot -------------------
   399→        const antiBot = this.detectAntiBot(html);
   400→        if (
   401→          (statusCode === 403 && (antiBot.confidenceScore > 0.3 || antiBot.cloudflare || antiBot.datadome || antiBot.akamai)) &&
   402→          retryCount < MAX_RETRIES
   403→        ) {
   404→          retryCount++;
   405→          logger.info(
   406→            { jobId: data.jobId, url: data.url, retry: retryCount, antiBotScore: antiBot.confidenceScore },
   407→            '403 with anti-bot detected -- rotating proxy AND fingerprint, retrying with stealth',
   408→          );
   409→
   410→          // Rotate BOTH proxy AND fingerprint
   411→          const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   412→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   413→          const rotatedProfile = stealthEngine.getRandomProfile();
   414→          logger.debug(
   415→            { jobId: data.jobId, rotatedUserAgent: rotatedProfile.userAgent, rotatedPlatform: rotatedProfile.platform },
   416→            'Rotated fingerprint profile for anti-bot retry',
   417→          );
   418→
   419→          fetchResult = await this.executeFetch(
   420→            'stealth-browser', data.url, domain, retryProxyUrl,
   421→            data.waitForSelector, data.timeout, data.headers,
   422→          );
   423→
   424→          if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   425→            html = fetchResult.html;
   426→            statusCode = fetchResult.statusCode;
   427→            strategy = fetchResult.strategy;
   428→            bandwidthBytes = Buffer.byteLength(html, 'utf8');
   429→            responseHeaders = fetchResult.headers || {};
   430→            if (retryProxy) {
   431→              proxyId = retryProxy.proxyId;
   432→              proxyCountry = retryProxy.country;
   433→              proxyUrl = retryProxyUrl;
   434→            }
   435→          }
   436→
   437→          // Record retry outcome
   438→          try {
   439→            await adaptiveRateLimiter.recordResponse(domain, {
   440→              success: fetchResult.statusCode >= 200 && fetchResult.statusCode < 400,
   441→              statusCode: fetchResult.statusCode,
   442→              responseMs: Date.now() - startTime,
   443→              hadAntiBot: true,
   444→            });
   445→          } catch {}
   446→        }
   447→
   448→        // -- Step 11: Smart Retry -- 503 Service Unavailable --------------
   449→        if (statusCode === 503 && retryCount < MAX_RETRIES) {
   450→          retryCount++;
   451→          logger.info(
   452→            { jobId: data.jobId, url: data.url, retry: retryCount },
   453→            '503 service unavailable -- retrying with different proxy after delay',
   454→          );
   455→
   456→          // Short delay before retry
   457→          await this.sleep(PROXY_RETRY_DELAY_MS);
   458→
   459→          // Get a different proxy
   460→          const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   461→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   462→
   463→          // Rotate fingerprint
   464→          stealthEngine.getRandomProfile();
   465→
   466→          fetchResult = await this.executeFetch(
   467→            actualStrategy, data.url, domain, retryProxyUrl,
   468→            data.waitForSelector, data.timeout, data.headers,
   469→          );
   470→
   471→          if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   472→            html = fetchResult.html;
   473→            statusCode = fetchResult.statusCode;
   474→            strategy = fetchResult.strategy;
   475→            bandwidthBytes = Buffer.byteLength(html, 'utf8');
   476→            responseHeaders = fetchResult.headers || {};
   477→            if (retryProxy) {
   478→              proxyId = retryProxy.proxyId;
   479→              proxyCountry = retryProxy.country;
   480→              proxyUrl = retryProxyUrl;
   481→            }
   482→          }
   483→
   484→          // Record retry outcome
   485→          try {
   486→            await adaptiveRateLimiter.recordResponse(domain, {
   487→              success: fetchResult.statusCode >= 200 && fetchResult.statusCode < 400,
   488→              statusCode: fetchResult.statusCode,
   489→              responseMs: Date.now() - startTime,
   490→            });
   491→          } catch {}
   492→        }
   493→
   494→        // -- Step 12: Anti-bot detected (existing escalation) ------------
   495→        // Kept as a fallback for cases where status is 403 but no anti-bot was
   496→        // specifically detected, or anti-bot is detected on other status codes.
   497→        const recheckAntiBot = this.detectAntiBot(html);
   498→        if (
   499→          (statusCode === 403 || recheckAntiBot.confidenceScore > 0.5) &&
   500→          strategy !== 'stealth-browser' &&
   501→          retryCount < MAX_RETRIES &&
   502→          // Don't double-retry if we already handled 403+anti-bot above
   503→          !(statusCode === 403 && (antiBot.confidenceScore > 0.3 || antiBot.cloudflare || antiBot.datadome || antiBot.akamai))
   504→        ) {
   505→          retryCount++;
   506→          logger.info({ jobId: data.jobId, url: data.url, retry: retryCount }, 'Anti-bot detected -- escalating to stealth');
   507→
   508→          // Try getting a different proxy for the retry
   509→          const retryProxy = retryCount > 1
   510→            ? await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random')
   511→            : null;
   512→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   513→
   514→          // Rotate fingerprint
   515→          stealthEngine.getRandomProfile();
   516→
   517→          fetchResult = await this.executeFetch(
   518→            'stealth-browser', data.url, domain, retryProxyUrl,
   519→            data.waitForSelector, data.timeout, data.headers,
   520→          );
   521→          if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   522→            html = fetchResult.html;
   523→            statusCode = fetchResult.statusCode;
   524→            strategy = fetchResult.strategy;
   525→            bandwidthBytes = Buffer.byteLength(html, 'utf8');
   526→            responseHeaders = fetchResult.headers || {};
   527→            if (retryProxy) {
   528→              proxyId = retryProxy.proxyId;
   529→              proxyCountry = retryProxy.country;
   530→              proxyUrl = retryProxyUrl;
   531→            }
   532→          }
   533→
   534→          // Record retry outcome
   535→          try {
   536→            await adaptiveRateLimiter.recordResponse(domain, {
   537→              success: fetchResult.statusCode >= 200 && fetchResult.statusCode < 400,
   538→              statusCode: fetchResult.statusCode,
   539→              responseMs: Date.now() - startTime,
   540→              hadAntiBot: true,
   541→            });
   542→          } catch {}
   543→        }
   544→
   545→        // -- Step 13: CAPTCHA solving (with fallback to proxy rotation) --
   546→        const finalAntiBot = this.detectAntiBot(html);
   547→        const hasCaptcha = finalAntiBot.reCaptcha || finalAntiBot.hCaptcha || finalAntiBot.cloudflareVariant === 'turnstile';
   548→
   549→        if (
   550→          hasCaptcha &&
   551→          data.solveCaptcha !== false &&
   552→          captchaSolver.isConfigured
   553→        ) {
   554→          try {
   555→            const captchaType = finalAntiBot.reCaptcha ? 'recaptcha_v2'
   556→              : finalAntiBot.hCaptcha ? 'hcaptcha'
   557→              : 'turnstile';
   558→
   559→            // Extract site key from HTML
   560→            const siteKey = this.extractCaptchaSiteKey(html, captchaType);
   561→            if (siteKey) {
   562→              logger.info({ jobId: data.jobId, type: captchaType, siteKey }, 'Attempting CAPTCHA solve');
   563→
   564→              const solveResult = await captchaSolver.solve({
   565→                url: data.url,
   566→                siteKey,
   567→                type: captchaType as any,
   568→                proxyUrl,
   569→              });
   570→
   571→              if (solveResult.success) {
   572→                captchaSolved = true;
   573→                creditsUsed += 3; // CAPTCHA_SOLVE cost
   574→
   575→                // Retry with the CAPTCHA token -- inject it into the page
   576→                if (strategy === 'browser' || strategy === 'stealth-browser') {
   577→                  const retryResult = await this.retryWithCaptchaToken(
   578→                    data.url, domain, proxyUrl, solveResult.token,
   579→                    captchaType, data.waitForSelector, data.timeout,
   580→                  );
   581→                  if (retryResult.statusCode >= 200 && retryResult.statusCode < 400) {
   582→                    html = retryResult.html;
   583→                    statusCode = retryResult.statusCode;
   584→                    bandwidthBytes = Buffer.byteLength(html, 'utf8');
   585→                  }
   586→                }
   587→              } else {
   588→                // -- CAPTCHA solving failed -- rotate proxy and retry with stealth --
   589→                if (retryCount < MAX_RETRIES) {
   590→                  retryCount++;
   591→                  logger.info(
   592→                    { jobId: data.jobId, retry: retryCount },
   593→                    'CAPTCHA solving failed -- rotating proxy and retrying with stealth',
   594→                  );
   595→
   596→                  await this.sleep(CAPTCHA_ROTATE_RETRY_DELAY_MS);
   597→
   598→                  const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   599→                  const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   600→                  stealthEngine.getRandomProfile(); // Rotate fingerprint
   601→
   602→                  fetchResult = await this.executeFetch(
   603→                    'stealth-browser', data.url, domain, retryProxyUrl,
   604→                    data.waitForSelector, data.timeout, data.headers,
   605→                  );
   606→
   607→                  if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   608→                    html = fetchResult.html;
   609→                    statusCode = fetchResult.statusCode;
   610→                    strategy = fetchResult.strategy;
   611→                    bandwidthBytes = Buffer.byteLength(html, 'utf8');
   612→                    responseHeaders = fetchResult.headers || {};
   613→                    if (retryProxy) {
   614→                      proxyId = retryProxy.proxyId;
   615→                      proxyCountry = retryProxy.country;
   616→                      proxyUrl = retryProxyUrl;
   617→                    }
   618→                  }
   619→                }
   620→              }
   621→            }
   622→          } catch (captchaErr: any) {
   623→            logger.warn({ jobId: data.jobId, error: captchaErr.message }, 'CAPTCHA solving failed');
   624→
   625→            // On CAPTCHA error, try proxy rotation + stealth as last resort
   626→            if (retryCount < MAX_RETRIES) {
   627→              retryCount++;
   628→              logger.info(
   629→                { jobId: data.jobId, retry: retryCount },
   630→                'CAPTCHA error -- rotating proxy and retrying with stealth',
   631→              );
   632→
   633→              try {
   634→                const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   635→                const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   636→                stealthEngine.getRandomProfile();
   637→
   638→                fetchResult = await this.executeFetch(
   639→                  'stealth-browser', data.url, domain, retryProxyUrl,
   640→                  data.waitForSelector, data.timeout, data.headers,
   641→                );
   642→
   643→                if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   644→                  html = fetchResult.html;
   645→                  statusCode = fetchResult.statusCode;
   646→                  strategy = fetchResult.strategy;
   647→                  bandwidthBytes = Buffer.byteLength(html, 'utf8');
   648→                  responseHeaders = fetchResult.headers || {};
   649→                  if (retryProxy) {
   650→                    proxyId = retryProxy.proxyId;
   651→                    proxyCountry = retryProxy.country;
   652→                    proxyUrl = retryProxyUrl;
   653→                  }
   654→                }
   655→              } catch (rotateErr: any) {
   656→                logger.warn(
   657→                  { jobId: data.jobId, error: rotateErr.message },
   658→                  'Post-CAPTCHA proxy rotation also failed',
   659→                );
   660→              }
   661→            }
   662→          }
   663→        }
   664→
   665→        // -- Step 14: Record anti-bot detection --------------------------
   666→        if (finalAntiBot.cloudflare || finalAntiBot.datadome || finalAntiBot.akamai) {
   667→          logger.warn(
   668→            { jobId: data.jobId, url: data.url, ...finalAntiBot },
   669→            'Anti-bot protection detected',
   670→          );
   671→        }
   672→
   673→        // -- Step 15: Record proxy outcome -------------------------------
   674→        if (proxyId) {
   675→          await proxyManager.recordOutcome({
   676→            proxyId,
   677→            domain,
   678→            success: statusCode >= 200 && statusCode < 400,
   679→            statusCode,
   680→            latencyMs: Date.now() - startTime,
   681→          });
   682→        }
   683→
   684→        // -- Step 16: Record outcome to domain intelligence --------------
   685→        await domainIntelligence.recordOutcome(domain, {
   686→          success: statusCode >= 200 && statusCode < 400,
   687→          statusCode,
   688→          responseMs: Date.now() - startTime,
   689→          strategy: strategy === 'stealth-browser' ? 'browser' : strategy as 'http' | 'browser',
   690→          usedBrowser: strategy === 'browser' || strategy === 'stealth-browser',
   691→          detectedCloudflare: finalAntiBot.cloudflare,
   692→          detectedDataDome: finalAntiBot.datadome,
   693→          pageSizeKb: Math.round(bandwidthBytes / 1024),
   694→          hadEmptyHtml: this.isEmptyHtml(html),
   695→        });
   696→
   697→        // -- Step 17: Record session request (if session is active) ------
   698→        if (data.sessionId && sessionState) {
   699→          try {
   700→            await sessionManager.recordSessionRequest(data.sessionId, {
   701→              success: statusCode >= 200 && statusCode < 400,
   702→              responseMs: Date.now() - startTime,
   703→              bandwidthBytes,
   704→              creditsUsed: 0, // Will be updated after billing
   705→            });
   706→          } catch (sessionErr: any) {
   707→            logger.debug(
   708→              { jobId: data.jobId, sessionId: data.sessionId, error: sessionErr.message },
   709→              'Failed to record session request',
   710→            );
   711→          }
   712→        }
   713→
   714→        // -- Step 18: Cache successful results ---------------------------
   715→        if (statusCode >= 200 && statusCode < 400) {
   716→          const profile = await domainIntelligence.getProfile(domain);
   717→          const ttl = data.cacheTtl || profile.cacheTtlSeconds;
   718→          await cacheSet(`scrape:${data.url}`, { html, statusCode, responseMs: Date.now() - startTime }, ttl);
   719→        }
   720→      }
   721→
   722→      // -- Step 19: Template-based extraction -------------------------------
   723→      // If templateId is specified, or extract is "auto", use the template registry
   724→      // to detect and extract structured data using pre-built templates.
   725→      if (html) {
   726→        try {
   727→          let templateId = data.templateId;
   728→          let templateMatch: import('../templates').TemplateMatch | null = null;
   729→
   730→          // Auto-detect template if templateId not specified but extract is "auto"
   731→          if (!templateId && data.extract === 'auto') {
   732→            templateMatch = templateRegistry.detectTemplate(data.url);
   733→            if (templateMatch && templateMatch.confidence >= 0.5) {
   734→              templateId = templateMatch.templateId;
   735→              logger.info(
   736→                { jobId: data.jobId, url: data.url, templateId, confidence: templateMatch.confidence },
   737→                'Auto-detected template for extraction',
   738→              );
   739→            }
   740→          }
   741→
   742→          // If a template is specified or auto-detected, extract with it
   743→          if (templateId) {
   744→            const templateResult = templateRegistry.extractWithTemplate(templateId, html, data.url);
   745→            if (templateResult && Object.keys(templateResult).length > 0) {
   746→              structuredData = {
   747→                ...(structuredData || {}),
   748→                template: {
   749→                  templateId,
   750→                  data: templateResult,
   751→                },
   752→              };
   753→              logger.info(
   754→                { jobId: data.jobId, templateId, fieldsExtracted: Object.keys(templateResult).length },
   755→                'Template extraction successful',
   756→              );
   757→            }
   758→          }
   759→        } catch (templateErr: any) {
   760→          logger.warn(
   761→            { jobId: data.jobId, templateId: data.templateId, error: templateErr.message },
   762→            'Template extraction failed -- falling back to standard extraction',
   763→          );
   764→        }
   765→      }
   766→
   767→      // -- Step 20: Structured data parsing (existing logic) ---------------
   768→      if (data.structured && html) {
   769→        try {
   770→          const existingStructured = structuredData || {};
   771→          const parsedStructured = this.parseStructuredData(html, data.url);
   772→          structuredData = { ...existingStructured, ...parsedStructured };
   773→          creditsUsed += 2;
   774→        } catch (err: any) {
   775→          logger.warn({ jobId: data.jobId, error: err.message }, 'Structured data parsing failed');
   776→        }
   777→      }
   778→
   779→      // -- Step 21: NL Extraction ------------------------------------------
   780→      if (data.extract && data.extract !== 'auto' && html) {
   781→        try {
   782→          const { nlExtractor } = await import('../extractor/nl-extractor');
   783→          const extractResult = await nlExtractor.extract({
   784→            html,
   785→            url: data.url,
   786→            instruction: data.extract,
   787→          });
   788→          extracted = extractResult.data ?? undefined;
   789→          creditsUsed += 3;
   790→        } catch (extractErr: any) {
   791→          logger.error(
   792→            { jobId: data.jobId, url: data.url, error: extractErr.message },
   793→            'NL extraction failed',
   794→          );
   795→        }
   796→      }
   797→
   798→      // -- Step 22: Output Pipeline Integration ----------------------------
   799→      // If outputFormat is specified, process the HTML through the output pipeline.
   800→      let processedHtml = html;
   801→      if (data.outputFormat && html && data.outputFormat !== 'raw') {
   802→        try {
   803→          const outputResult = outputPipeline.process(html, data.url, {
   804→            format: data.outputFormat as OutputFormat,
   805→          });
   806→
   807→          processedHtml = typeof outputResult.content === 'string'
   808→            ? outputResult.content
   809→            : JSON.stringify(outputResult.content);
   810→          appliedOutputFormat = data.outputFormat;
   811→          outputCompressionRatio = outputResult.compressionRatio;
   812→
   813→          logger.info(
   814→            {
   815→              jobId: data.jobId,
   816→              format: data.outputFormat,
   817→              originalSizeBytes: outputResult.originalSizeBytes,
   818→              outputSizeBytes: outputResult.outputSizeBytes,
   819→              compressionRatio: outputResult.compressionRatio,
   820→              processingMs: outputResult.processingMs,
   821→            },
   822→            'Output pipeline processing completed',
   823→          );
   824→        } catch (outputErr: any) {
   825→          logger.warn(
   826→            { jobId: data.jobId, format: data.outputFormat, error: outputErr.message },
   827→            'Output pipeline processing failed -- returning raw HTML',
   828→          );
   829→          // Gracefully fall back to raw HTML
   830→          appliedOutputFormat = data.outputFormat;
   831→          outputCompressionRatio = 1;
   832→        }
   833→      }
   834→
   835→      // -- Step 23: Calculate credits -------------------------------------
   836→      const isSuccess = statusCode >= 200 && statusCode < 400;
   837→      creditsUsed += calculateScrapeCredits(strategy as 'cache' | 'http' | 'browser', !!data.extract);
   838→
   839→      // -- Step 24: Success-Only Billing ----------------------------------
   840→      if (isSuccess && creditsUsed > 0) {
   841→        const deducted = await deductCredits(data.userId, data.apiKeyId, creditsUsed);
   842→        creditsCharged = deducted ? creditsUsed : 0;
   843→      } else {
   844→        creditsCharged = 0;
   845→      }
   846→
   847→      const responseMs = Date.now() - startTime;
   848→
   849→      // NOTE: Do NOT update job status here -- the worker handles it.
   850→      // This fixes the double-update race condition.
   851→      // Only update non-status fields (result data, credits, etc.)
   852→      await db.scrapeJob.update({
   853→        where: { id: data.jobId },
   854→        data: {
   855→          strategy,
   856→          creditsUsed,
   857→          creditsCharged,
   858→          bandwidthBytes,
   859→          result: { html: processedHtml.substring(0, 50000) },
   860→          extractedData: extracted ?? undefined,
   861→          structuredData: structuredData ?? undefined,
   862→          statusCode,
   863→          responseMs,
   864→          proxyId,
   865→          proxyCountry,
   866→          captchaSolved,
   867→          retryCount,
   868→        },
   869→      });
   870→
   871→      // Update API key bandwidth tracking
   872→      try {
   873→        await db.apiKey.update({
   874→          where: { id: data.apiKeyId },
   875→          data: { bandwidthBytes: { increment: bandwidthBytes } },
   876→        });
   877→      } catch {}
   878→
   879→      // -- Step 25: Fire webhooks -----------------------------------------
   880→      webhookDispatcher.fire(data.userId, 'job_completed', {
   881→        jobId: data.jobId,
   882→        url: data.url,
   883→        strategy,
   884→        statusCode,
   885→        responseMs,
   886→        creditsCharged,
   887→        cached,
   888→        captchaSolved,
   889→        bandwidthBytes,
   890→        outputFormat: appliedOutputFormat,
   891→        outputCompressionRatio,
   892→      }).catch(() => {});
   893→
   894→      logger.info(
   895→        { jobId: data.jobId, url: data.url, strategy, statusCode, responseMs, creditsUsed, creditsCharged, cached, captchaSolved, bandwidthBytes, retryCount, outputFormat: appliedOutputFormat, outputCompressionRatio },
   896→        'Scrape job completed',
   897→      );
   898→
   899→      return {
   900→        jobId: data.jobId,
   901→        status: 'done',
   902→        strategy,
   903→        html: cached ? undefined : processedHtml,
   904→        extracted,
   905→        structuredData,
   906→        statusCode,
   907→        responseMs,
   908→        creditsUsed,
   909→        creditsCharged,
   910→        proxyId,
   911→        proxyCountry,
   912→        captchaSolved,
   913→        cached,
   914→        bandwidthBytes,
   915→        finalUrl,
   916→        outputFormat: appliedOutputFormat,
   917→        outputCompressionRatio,
   918→      };
   919→    } catch (err: any) {
   920→      error = err.message || String(err);
   921→      const responseMs = Date.now() - startTime;
   922→
   923→      logger.error({ jobId: data.jobId, url: data.url, error, responseMs }, 'Scrape job failed');
   924→
   925→      // Fire failure webhook
   926→      webhookDispatcher.fire(data.userId, 'job_failed', {
   927→        jobId: data.jobId,
   928→        url: data.url,
   929→        error,
   930→        statusCode,
   931→      }).catch(() => {});
   932→
   933→      return {
   934→        jobId: data.jobId,
   935→        status: 'failed',
   936→        strategy,
   937→        creditsUsed: 0,
   938→        creditsCharged: 0,
   939→        statusCode,
   940→        responseMs,
   941→        error,
   942→        cached: false,
   943→      };
   944→    }
   945→  }

   981→  /**
   982→   * HTTP fetch with TLS fingerprint randomization via got-scraping.
   983→   * Uses tlsFingerprintFetcher first (randomizes JA3/JA4 fingerprints),
   984→   * then falls back to proxyFetch if got-scraping is unavailable or fails.
   985→   *
   986→   * This is the #1 most impactful fix for bypassing Cloudflare, Akamai, and DataDome,
   987→   * which use TLS fingerprinting to instantly identify automated traffic.
   988→   */
   989→  private async fetchViaHttp(
   990→    url: string,
   991→    _domain: string,
   992→    proxyUrl?: string,
   993→    headers?: Record<string, string>,
   994→    timeout?: number,
   995→  ): Promise<{ html: string; statusCode: number; strategy: JobStrategy; headers: Record<string, string>; finalUrl: string }> {
   996→    const requestTimeout = timeout || DEFAULT_TIMEOUT;
   997→
   998→    // Build fallback headers for proxyFetch (used when got-scraping is unavailable)
   999→    const profile = stealthEngine.getRandomProfile();
  1000→    const fallbackHeaders: Record<string, string> = {
  1001→      'User-Agent': profile.userAgent,
  1002→      Accept: 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8',
  1003→      'Accept-Language': `${profile.locale},en;q=0.9`,
  1004→      'Accept-Encoding': 'gzip, deflate, br',
  1005→      'Cache-Control': 'no-cache',
  1006→      'Sec-Ch-Ua': '"Chromium";v="126", "Google Chrome";v="126", "Not-A.Brand";v="99"',
  1007→      'Sec-Ch-Ua-Mobile': '?0',
  1008→      'Sec-Ch-Ua-Platform': `"${profile.platform}"`,
  1009→      'Sec-Fetch-Dest': 'document',
  1010→      'Sec-Fetch-Mode': 'navigate',
  1011→      'Sec-Fetch-Site': 'none',
  1012→      'Sec-Fetch-User': '?1',
  1013→      'Upgrade-Insecure-Requests': '1',
  1014→      ...headers,
  1015→    };
  1016→
  1017→    // Define the fallback function that uses proxyFetch
  1018→    const fallbackFn = async (
  1019→      fbUrl: string,
  1020→      fbProxyUrl?: string,
  1021→      fbOpts?: any,
  1022→    ): Promise<{ text: string; status: number; headers: Record<string, string>; url: string; ok: boolean }> => {
  1023→      const controller = new AbortController();
  1024→      const timeoutId = setTimeout(() => controller.abort(), requestTimeout);
  1025→      try {
  1026→        return await proxyFetch(fbUrl, fbProxyUrl, {
  1027→          headers: fallbackHeaders,
  1028→          signal: controller.signal,
  1029→          redirect: 'follow',
  1030→        });
  1031→      } finally {
  1032→        clearTimeout(timeoutId);
  1033→      }
  1034→    };
  1035→
  1036→    try {
  1037→      // Try TLS-fingerprint-aware fetch first (got-scraping)
  1038→      // This randomizes JA3/JA4 fingerprints to mimic real browsers
  1039→      const response = await tlsFingerprintFetcher.fetch(
  1040→        url,
  1041→        proxyUrl,
  1042→        {
  1043→          headers,
  1044→          timeout: requestTimeout,
  1045→        },
  1046→        fallbackFn, // Fallback to proxyFetch if got-scraping fails
  1047→      );
  1048→
  1049→      const html = response.text;
  1050→
  1051→      // Extract useful response headers
  1052→      const responseHeaders: Record<string, string> = {};
  1053→      const headerWhitelist = [
  1054→        'content-type', 'server', 'x-powered-by', 'cf-ray',
  1055→        'x-cache', 'content-encoding', 'content-length',
  1056→        'x-request-id', 'strict-transport-security',
  1057→      ];
  1058→      for (const [key, value] of Object.entries(response.headers)) {
  1059→        if (headerWhitelist.includes(key.toLowerCase())) {
  1060→          responseHeaders[key] = value;
  1061→        }
  1062→      }
  1063→
  1064→      const finalUrl = response.url || url;
  1065→
  1066→      return { html, statusCode: response.status, strategy: 'http', headers: responseHeaders, finalUrl };
  1067→    } catch (err: any) {
  1068→      if (err.name === 'AbortError') return { html: '', statusCode: 408, strategy: 'http', headers: {}, finalUrl: url };
  1069→      return { html: '', statusCode: 502, strategy: 'http', headers: {}, finalUrl: url };
  1070→    }
  1071→  }

  1073→  private async fetchViaBrowser(
  1074→    url: string,
  1075→    _domain: string,
  1076→    proxyUrl?: string,
  1077→    waitForSelector?: string,
  1078→    timeout?: number,
  1079→  ): Promise<{ html: string; statusCode: number; strategy: JobStrategy; headers?: Record<string, string>; finalUrl?: string }> {
  1080→    const requestTimeout = timeout || DEFAULT_TIMEOUT;
  1081→    let lease: BrowserLease | undefined;
  1082→
  1083→    try {
  1084→      lease = await browserPool.acquire(proxyUrl, false);
  1085→
  1086→      const response = await lease.page.goto(url, { waitUntil: 'networkidle', timeout: requestTimeout });
  1087→      const statusCode = response ? response.status() : 200;
  1088→
  1089→      if (waitForSelector) {
  1090→        try { await lease.page.waitForSelector(waitForSelector, { timeout: 10_000 }); } catch {}
  1091→      }
  1092→
  1093→      // Simulate human behavior for anti-bot evasion
  1094→      if (statusCode < 400) {
  1095→        try {
  1096→          await humanBehavior.simulatePageInteraction(lease.page);
  1097→        } catch (behaviorErr: any) {
  1098→          logger.debug({ error: behaviorErr.message }, 'Human behavior simulation failed (non-critical)');
  1099→        }
  1100→      }
  1101→
  1102→      const html = await lease.page.content();
  1103→      const finalUrl = lease.page.url();
  1104→
  1105→      return { html, statusCode, strategy: 'browser', finalUrl };
  1106→    } catch (err: any) {
  1107→      if (err.message?.includes('Timeout') || err.name === 'TimeoutError') {
  1108→        return { html: '', statusCode: 408, strategy: 'browser' };
  1109→      }
  1110→      return { html: '', statusCode: 502, strategy: 'browser' };
  1111→    } finally {
  1112→      if (lease) await browserPool.release(lease);
  1113→    }
  1114→  }

  1116→  private async fetchViaStealthBrowser(
  1117→    url: string,
  1118→    _domain: string,
  1119→    proxyUrl?: string,
  1120→    waitForSelector?: string,
  1121→    timeout?: number,
  1122→  ): Promise<{ html: string; statusCode: number; strategy: JobStrategy; headers?: Record<string, string>; finalUrl?: string }> {
  1123→    const requestTimeout = timeout || DEFAULT_TIMEOUT;
  1124→    let lease: BrowserLease | undefined;
  1125→
  1126→    try {
  1127→      lease = await browserPool.acquire(proxyUrl, true);
  1128→
  1129→      const response = await lease.page.goto(url, {
  1130→        waitUntil: 'domcontentloaded',
  1131→        timeout: requestTimeout,
  1132→      });

  1219→  private async retryWithCaptchaToken(
  1220→    url: string,
  1221→    _domain: string,
  1222→    proxyUrl: string | undefined,
  1223→    token: string,
  1224→    captchaType: string,
  1225→    waitForSelector?: string,
  1226→    timeout?: number,

  1232→      lease = await browserPool.acquire(proxyUrl, true);

  1234→      const response = await lease.page.goto(url, {
  1235→        waitUntil: 'domcontentloaded',
  1236→        timeout: requestTimeout,
  1237→      });

  1239→

  1250→                for (const [, value] of Object.entries(client)) {
  1251→                  const v = value as any;
  1252→                  if (typeof v?.callback === 'function') {
  1253→                    v.callback(t);
  1254→                  }
  1255→                }
  1256→              }
  1257→            }
  1258→          } catch {}
  1259→        }, token);

  1261→        await lease.page.evaluate((t) => {
  1262→          const el = document.querySelector('[name="h-captcha-response"]') as HTMLTextAreaElement;
  1263→          if (el) el.value = t;
  1264→          if ((window as any).hcaptcha) {
  1265→            (window as any).hcaptcha.setResponse(t);
  1266→          }
  1267→        }, token);

  1269→        await lease.page.evaluate((t) => {
  1270→          const el = document.querySelector('[name="cf-turnstile-response"]') as HTMLTextAreaElement;
  1271→          if (el) el.value = t;
  1272→          if ((window as any).turnstile) {
  1273→            (window as any).turnstile.getResponse = () => t;
  1274→          }
  1275→        }, token);
  1276→      }
  1277→
  1278→      // Wait for navigation/content after CAPTCHA
  1279→      try {
  1280→        await lease.page.waitForFunction(

  1288→      }
  1289→
  1290→      const html = await lease.page.content();
  1291→      return { html, statusCode };
  1292→    } catch (err: any) {
  1293→      return { html: '', statusCode: 502 };
  1294→    } finally {
  1295→      if (lease) await browserPool.release(lease);
  1296→    }

  1346→  private isEmptyHtml(html: string): boolean {
  1347→    if (!html || html.length < 50) return true;
  1348→    const lowerHtml = html.toLowerCase();
  1349→    if (!lowerHtml.includes('<body')) return true;
  1350→    const textOnly = html.replace(/<[^>]*>/g, '').trim();
  1351→    if (textOnly.length < 100) return true;
  1352→    const tagCount = (html.match(/<[a-zA-Z][^>]*>/g) || []).length;
  1353→    if (tagCount < 5) return true;
  1354→    return false;

  1359→   * encounters an error. This preserves the original Redis-counter approach.

  1361→  private async acquireSimpleRateLimitToken(domain: string, safeRps: number): Promise<void> {
  1362→    const key = `ratelimit:${domain}`;
  1363→    const maxRetries = 50;
  1364→    const retryDelayMs = 100;
  1365→
  1366→    for (let attempt = 0; attempt < maxRetries; attempt++) {
  1367→      const current = await redis.incr(key);
  1368→      if (current === 1) await redis.expire(key, RATE_LIMIT_WINDOW_SECONDS);
  1369→      if (current <= Math.max(1, Math.ceil(safeRps))) return;
  1370→      await this.sleep(retryDelayMs);
  1371→    }
  1372→
  1373→    logger.warn({ domain, safeRps }, 'Rate limit token acquisition timed out');
  1374→  }
