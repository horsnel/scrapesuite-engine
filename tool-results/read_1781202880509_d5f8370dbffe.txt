   466→  constructor() {
   467→    this.binariesDir = join(process.cwd(), '.botbrowser', 'binaries');
   468→    this.profilesDir = join(process.cwd(), '.botbrowser', 'profiles');

   472→  // --- Initialization ------------------------------------------------------
   473→
   474→  /**
   475→   * Initialize the stealth browser engine.
   476→   * Scans for cached binaries, creates directories, and loads health data from Redis.
   477→   */
   478→  async initialize(): Promise<void> {
   479→    if (this.initialized) return;
   480→
   481→    logger.info({ platform: this.currentPlatform }, 'Initializing Stealth Browser Engine');
   482→
   483→    // Ensure directories exist
   484→    for (const dir of [this.binariesDir, this.profilesDir]) {
   485→      if (!existsSync(dir)) {
   486→        mkdirSync(dir, { recursive: true });
   487→        logger.info({ dir }, 'Created BotBrowser directory');
   488→      }
   489→    }
   490→
   491→    // Scan for existing cached binaries
   492→    await this.scanCachedBinaries();
   493→
   494→    // Load health data from Redis
   495→    await this.loadHealthData();
   496→
   497→    this.initialized = true;
   498→    logger.info(

   506→  /**
   507→   * Launch a stealth browser with BotBrowser binary (or Nodriver fallback).
   508→   * Returns a Playwright Browser instance launched with sanitized arguments.
   509→   */
   510→  async launchStealthBrowser(options: StealthLaunchOptions = {}): Promise<StealthBrowserResult> {
   511→    if (!this.initialized) await this.initialize();
   512→
   513→    const {
   514→      chromeVersion = '130',
   515→      useBotBrowser = true,
   516→      fallbackToNodriver = true,
   517→      proxyServer,
   518→      extraArgs = [],
   519→      headless = false,
   520→      disableGpu = true,
   521→      windowSize = { width: 1920, height: 1080 },
   522→      timeout = 30000,
   523→      locale = 'en-US',
   524→      timezoneId = 'America/New_York',
   525→    } = options;
   526→
   527→    const sessionId = this.generateSessionId();
   528→    const userDataDir = options.userDataDir || this.createIsolatedProfileDir(sessionId);
   529→
   530→    // Build sanitized launch arguments
   531→    const launchArgs = this.buildSanitizedArgs({
   532→      windowSize,
   533→      disableGpu,
   534→      headless,
   535→      proxyServer,
   536→      extraArgs,
   537→      locale,
   538→      timezoneId,
   539→    });
   540→
   541→    let binary: BotBrowserBinary | null = null;
   542→    let mode: 'botbrowser' | 'nodriver' = 'nodriver';
   543→    let browser: Browser | null = null;
   544→    let cdpSurfaceReduced = false;
   545→
   546→    // Attempt BotBrowser launch first
   547→    if (useBotBrowser) {
   548→      try {
   549→        binary = await this.getBinaryForVersion(chromeVersion);
   550→        if (binary) {
   551→          logger.info({ version: chromeVersion, binaryPath: binary.binaryPath }, 'Launching with BotBrowser binary');
   552→
   553→          const { chromium } = await import('playwright');
   554→          browser = await chromium.launch({
   555→            executablePath: binary.binaryPath,
   556→            headless: headless === 'new' ? false : headless,
   557→            args: launchArgs,
   558→            timeout,
   559→          });
   560→
   561→          mode = 'botbrowser';
   562→          binary.launchCount++;
   563→
   564→          // Apply minimal CDP session management
   565→          cdpSurfaceReduced = await this.applyMinimalCdpSurface(browser);
   566→
   567→          // Cache the binary info
   568→          await cacheSet(`botbrowser:binary:${chromeVersion}:${this.currentPlatform}`, binary, 86400);
   569→
   570→          logger.info({ version: chromeVersion, mode: 'botbrowser' }, 'BotBrowser launched successfully');
   571→        } else if (!fallbackToNodriver) {
   572→          throw new Error(`BotBrowser binary for Chrome ${chromeVersion} not found and fallback disabled`);
   573→        }
   574→      } catch (err: any) {
   575→        logger.warn({ err: err.message, version: chromeVersion }, 'BotBrowser launch failed');
   576→        if (!fallbackToNodriver) throw err;
   577→        this.recordCrash(chromeVersion);
   578→      }
   579→    }
   580→
   581→    // Nodriver fallback -- launch with system Chromium but extreme sanitization
   582→    if (!browser && (!useBotBrowser || fallbackToNodriver)) {
   583→      mode = 'nodriver';
   584→      logger.info({ version: chromeVersion }, 'Launching in Nodriver fallback mode');
   585→
   586→      const { chromium } = await import('playwright');
   587→      browser = await chromium.launch({
   588→        headless: headless === 'new' ? false : headless,
   589→        args: [...launchArgs, '--disable-blink-features=AutomationControlled'],
   590→        timeout,
   591→      });
   592→
   593→      // Apply CDP surface reduction for Nodriver mode
   594→      cdpSurfaceReduced = await this.applyNodriverCdpReduction(browser);
   595→    }
   596→
   597→    if (!browser) {
   598→      throw new Error('Failed to launch stealth browser -- no browser instance created');
   599→    }
   600→
   601→    // Create the isolated context
   602→    const context = await this.createStealthContext(browser, {
   603→      userDataDir,
   604→      locale,
   605→      timezoneId,
   606→      proxyServer,
   607→      sessionId,
   608→    });
   609→
   610→    // Track session
   611→    this.activeSessions.set(sessionId, { browser, context, launchedAt: Date.now() });
   612→
   613→    logger.info({
   614→      sessionId,
   615→      mode,
   616→      chromeVersion,
   617→      argsCount: launchArgs.length,
   618→      userDataDir,
   619→    }, 'Stealth browser session launched');
   620→
   621→    return {
   622→      browser,
   623→      context,
   624→      binary,
   625→      mode,
   626→      launchArgs,
   627→      userDataDir,
   628→      sessionId,
   629→      initScriptApplied: true,
   630→      cdpSurfaceReduced,
   631→    };
   632→  }

   698→  /**
   699→   * Get a BotBrowser binary for the specified Chrome version.
   700→   * Downloads if not cached, verifies integrity.
   701→   */
   702→  async getBinaryForVersion(version: ChromeVersion): Promise<BotBrowserBinary | null> {
   703→    const cacheKey = `${version}:${this.currentPlatform}`;
   704→
   705→    // Check in-memory cache
   706→    const cached = this.cachedBinaries.get(cacheKey);
   707→    if (cached && cached.verified) {
   708→      // Quick existence check
   709→      if (existsSync(cached.binaryPath)) {
   710→        return cached;
   711→      }
   712→      logger.warn({ version, path: cached.binaryPath }, 'Cached binary no longer exists, re-downloading');
   713→      this.cachedBinaries.delete(cacheKey);
   714→    }
   715→
   716→    // Check Redis cache
   717→    try {
   718→      const redisBinary = await cacheGet<BotBrowserBinary>(`botbrowser:binary:${version}:${this.currentPlatform}`);
   719→      if (redisBinary && existsSync(redisBinary.binaryPath)) {
   720→        this.cachedBinaries.set(cacheKey, redisBinary);
   721→        return redisBinary;
   722→      }
   723→    } catch { /* Redis unavailable, continue */ }
   724→
   725→    // Download the binary
   726→    try {
   727→      const binary = await this.downloadBinary(version);
   728→      if (binary) {
   729→        this.cachedBinaries.set(cacheKey, binary);
   730→        return binary;
   731→      }
   732→    } catch (err: any) {
   733→      logger.error({ err: err.message, version }, 'Failed to download BotBrowser binary');
   734→    }
   735→
   736→    return null;
   737→  }

   739→  /**
   740→   * Download a BotBrowser binary for the specified version and platform.
   741→   * Includes integrity verification via SHA256.
   742→   */
   743→  async downloadBinary(version: ChromeVersion): Promise<BotBrowserBinary | null> {
   744→    const versionInfo = CHROME_VERSIONS.find(v => v.version === version);
   745→    if (!versionInfo) {
   746→      logger.error({ version }, 'Unknown Chrome version');
   747→      return null;
   748→    }
   749→
   750→    const platform = this.currentPlatform;
   751→    const downloadUrl = versionInfo.downloadUrls[platform];
   752→    const expectedSha256 = versionInfo.sha256[platform];
   753→
   754→    const versionDir = join(this.binariesDir, `chrome-${version}`);
   755→    if (!existsSync(versionDir)) {
   756→      mkdirSync(versionDir, { recursive: true });
   757→    }
   758→
   759→    const archivePath = join(versionDir, `botbrowser_${version}.zip`);
   760→    const binaryPath = platform === 'win'
   761→      ? join(versionDir, 'chrome.exe')
   762→      : join(versionDir, 'chrome');
   763→
   764→    // Skip if already downloaded and verified
   765→    if (existsSync(binaryPath)) {
   766→      const verified = await this.verifyBinaryIntegrity(binaryPath, expectedSha256);
   767→      if (verified) {
   768→        const stat = statSync(binaryPath);
   769→        const binary: BotBrowserBinary = {
   770→          version,
   771→          platform,
   772→          binaryPath,
   773→          profileDir: versionDir,
   774→          sha256: expectedSha256,
   775→          sizeBytes: stat.size,
   776→          downloadedAt: Date.now(),
   777→          lastVerifiedAt: Date.now(),
   778→          verified: true,
   779→          crashCount: 0,
   780→          launchCount: 0,
   781→        };
   782→        logger.info({ version, platform, size: stat.size }, 'BotBrowser binary already exists and verified');
   783→        return binary;
   784→      }
   785→      logger.warn({ version, path: binaryPath }, 'Existing binary failed integrity check, re-downloading');
   786→    }
   787→
   788→    logger.info({ version, platform, url: downloadUrl }, 'Downloading BotBrowser binary');
   789→
   790→    try {
   791→      // Download the archive
   792→      const response = await fetch(downloadUrl);
   793→      if (!response.ok) {
   794→        throw new Error(`Download failed: HTTP ${response.status}`);
   795→      }
   796→
   797→      const body = response.body;
   798→      if (!body) {
   799→        throw new Error('Download failed: empty response body');
   800→      }
   801→
   802→      // Stream to file
   803→      const fileStream = createWriteStream(archivePath);
   804→      // Use ReadableStream from fetch response
   805→      const reader = body.getReader();
   806→      const writer = fileStream;
   807→
   808→      try {
   809→        while (true) {
   810→          const { done, value } = await reader.read();
   811→          if (done) break;
   812→          writer.write(value);
   813→        }
   814→      } finally {
   815→        writer.end();
   816→      }
   817→
   818→      logger.info({ version, archivePath }, 'Download complete, extracting...');
   819→
   820→      // Extract the archive (using system unzip/tar)
   821→      const { execSync } = await import('child_process');
   822→      if (platform === 'win') {
   823→        execSync(`powershell -command "Expand-Archive -Path '${archivePath}' -DestinationPath '${versionDir}' -Force"`, { timeout: 120000 });
   824→      } else {
   825→        execSync(`unzip -o -q '${archivePath}' -d '${versionDir}'`, { timeout: 120000 });
   826→      }
   827→
   828→      // Make binary executable on Linux/Mac
   829→      if (platform !== 'win' && existsSync(binaryPath)) {
   830→        execSync(`chmod +x '${binaryPath}'`);
   831→      }
   832→
   833→      // Verify integrity
   834→      const verified = await this.verifyBinaryIntegrity(binaryPath, expectedSha256);
   835→      if (!verified) {
   836→        logger.error({ version, path: binaryPath }, 'Binary integrity verification failed after download');
   837→        rmSync(versionDir, { recursive: true, force: true });
   838→        return null;
   839→      }
   840→
   841→      const stat = statSync(binaryPath);
   842→      const binary: BotBrowserBinary = {
   843→        version,
   844→        platform,
   845→        binaryPath,
   846→        profileDir: versionDir,
   847→        sha256: expectedSha256,
   848→        sizeBytes: stat.size,
   849→        downloadedAt: Date.now(),
   850→        lastVerifiedAt: Date.now(),
   851→        verified: true,
   852→        crashCount: 0,
   853→        launchCount: 0,
   854→      };
   855→
   856→      // Cache to Redis
   857→      await cacheSet(`botbrowser:binary:${version}:${platform}`, binary, 86400 * 7);
   858→
   859→      logger.info({ version, platform, size: stat.size }, 'BotBrowser binary downloaded and verified');
   860→      return binary;
   861→    } catch (err: any) {
   862→      logger.error({ err: err.message, version, platform }, 'Failed to download BotBrowser binary');
   863→      // Clean up partial download
   864→      try { rmSync(versionDir, { recursive: true, force: true }); } catch {}
   865→      return null;
   866→    }
   867→  }

   869→  /**
   870→   * Verify binary integrity using SHA256 hash comparison.
   871→   */
   872→  async verifyBinaryIntegrity(binaryPath: string, expectedSha256: string): Promise<boolean> {
   873→    if (!existsSync(binaryPath)) {
   874→      logger.warn({ path: binaryPath }, 'Binary path does not exist for integrity check');
   875→      return false;
   876→    }
   877→
   878→    try {
   879→      const hash = createHash('sha256');
   880→      const stream = createReadStream(binaryPath);
   881→
   882→      return new Promise<boolean>((resolve) => {
   883→        stream.on('data', (data) => hash.update(data));
   884→        stream.on('end', () => {
   885→          const actualSha256 = hash.digest('hex');
   886→          const match = actualSha256 === expectedSha256;
   887→          if (!match) {
   888→            logger.warn({ path: binaryPath, expected: expectedSha256.substring(0, 16), actual: actualSha256.substring(0, 16) }, 'SHA256 mismatch');
   889→          }
   890→          resolve(match);
   891→        });
   892→        stream.on('error', (err) => {
   893→          logger.error({ err: err.message, path: binaryPath }, 'Error reading binary for integrity check');
   894→          resolve(false);
   895→        });
   896→      });
   897→    } catch (err: any) {
   898→      logger.error({ err: err.message, path: binaryPath }, 'Integrity verification failed');
   899→      return false;
   900→    }
   901→  }

   905→  /**
   906→   * Get the health state of a browser binary.
   907→   * Tracks crash rates, memory leaks, and degradation.
   908→   */
   909→  getHealth(version: ChromeVersion): BrowserHealthState | null {
   910→    const cacheKey = `${version}:${this.currentPlatform}`;
   911→    const binary = this.cachedBinaries.get(cacheKey);
   912→    if (!binary) return null;
   913→
   914→    const data = this.healthData.get(cacheKey) || {
   915→      crashes: 0, launches: 0, launchTimes: [], lastCrashAt: null as number | null, consecutiveCrashes: 0,
   916→    };
   917→
   918→    const crashRate = data.launches > 0 ? data.crashes / data.launches : 0;
   919→    const avgLaunchTimeMs = data.launchTimes.length > 0
   920→      ? data.launchTimes.reduce((a, b) => a + b, 0) / data.launchTimes.length
   921→      : 0;
   922→
   923→    // Detect memory leak pattern -- increasing launch times over recent sessions
   924→    let memoryLeakDetected = false;
   925→    if (data.launchTimes.length >= 5) {
   926→      const recent = data.launchTimes.slice(-5);
   927→      const oldest = recent[0];
   928→      const newest = recent[recent.length - 1];
   929→      if (newest > oldest * 2) {
   930→        memoryLeakDetected = true;
   931→      }
   932→    }
   933→
   934→    const degraded = crashRate > 0.3 || data.consecutiveCrashes >= 3 || memoryLeakDetected;
   935→
   936→    let recommendation: BrowserHealthState['recommendation'] = 'healthy';
   937→    if (degraded) recommendation = 'replace';
   938→    else if (crashRate > 0.15 || data.consecutiveCrashes >= 2) recommendation = 'restart';
   939→    else if (crashRate > 0.05 || memoryLeakDetected) recommendation = 'monitor';
   940→
   941→    return {
   942→      binary,
   943→      crashRate,
   944→      avgLaunchTimeMs,
   945→      memoryLeakDetected,
   946→      lastCrashAt: data.lastCrashAt,
   947→      consecutiveCrashes: data.consecutiveCrashes,
   948→      degraded,
   949→      recommendation,
   950→    };
   951→  }

  1061→  /**
  1062→   * Check if a Chrome launch argument is safe (not an automation leak).
  1063→   */
  1064→  private isArgSafe(arg: string): boolean {
  1065→    const stripped = AUTOMATION_FLAGS_TO_STRIP.map(flag => {
  1066→      // Handle flags with = values -- check prefix
  1067→      const eqIdx = flag.indexOf('=');
  1068→      return eqIdx > -1 ? flag.substring(0, eqIdx) : flag;
  1069→    });
  1070→
  1071→    const argPrefix = arg.includes('=') ? arg.substring(0, arg.indexOf('=')) : arg;
  1072→
  1073→    for (const unsafe of stripped) {
  1074→      if (argPrefix === unsafe || arg.startsWith(unsafe + '=')) {
  1075→        return false;
  1076→      }
  1077→    }
  1078→
  1079→    return true;
  1080→  }

  1082→  /**

  1084→   * Only connects what's needed -- avoids detectable CDP commands.
  1085→   */
  1086→  private async applyMinimalCdpSurface(browser: Browser): Promise<boolean> {
  1087→    try {
  1088→      const contexts = browser.contexts();
  1089→      if (contexts.length === 0) return false;
  1090→
  1091→      const pages = contexts[0].pages();
  1092→      if (pages.length === 0) return false;

  1096→      return true;

  1099→      return false;
  1100→    }

  1103→  /**
  1104→   * Apply Nodriver-style CDP reduction for when BotBrowser binary is unavailable.
  1105→   * The key principle: avoid Runtime.enable entirely. This is what Kasada detects.
  1106→   */
  1107→  private async applyNodriverCdpReduction(browser: Browser): Promise<boolean> {
  1108→    try {
  1109→      const contexts = browser.contexts();
  1110→      if (contexts.length === 0) return false;
  1111→
  1112→      const pages = contexts[0].pages();

  1162→   * Scan the binaries directory for already-cached BotBrowser binaries.
  1163→   */
  1164→  private async scanCachedBinaries(): Promise<void> {
  1165→    if (!existsSync(this.binariesDir)) return;
  1166→
  1167→    const entries = readdirSync(this.binariesDir, { withFileTypes: true });
  1168→    for (const entry of entries) {
  1169→      if (!entry.isDirectory()) continue;
  1170→
  1171→      const match = entry.name.match(/^chrome-(126|127|128|129|130)$/);
  1172→      if (!match) continue;
  1173→
  1174→      const version = match[1] as ChromeVersion;
  1175→      const versionDir = join(this.binariesDir, entry.name);
  1176→      const binaryName = this.currentPlatform === 'win' ? 'chrome.exe' : 'chrome';
  1177→      const binaryPath = join(versionDir, binaryName);
  1178→
  1179→      if (existsSync(binaryPath)) {
  1180→        const stat = statSync(binaryPath);
  1181→        const versionInfo = CHROME_VERSIONS.find(v => v.version === version);
  1182→        const expectedSha256 = versionInfo?.sha256[this.currentPlatform] || '';
  1183→
  1184→        const binary: BotBrowserBinary = {
  1185→          version,
  1186→          platform: this.currentPlatform,
  1187→          binaryPath,
  1188→          profileDir: versionDir,
  1189→          sha256: expectedSha256,
  1190→          sizeBytes: stat.size,
  1191→          downloadedAt: stat.mtimeMs,
  1192→          lastVerifiedAt: 0,
  1193→          verified: false, // Will be verified on first use
  1194→          crashCount: 0,
  1195→          launchCount: 0,
  1196→        };
  1197→
  1198→        this.cachedBinaries.set(`${version}:${this.currentPlatform}`, binary);
  1199→        logger.debug({ version, size: stat.size }, 'Found cached BotBrowser binary');
  1200→      }
  1201→    }
  1202→  }
