   709→  /**
   710→   * Get the best proxy for a given domain and tier.
   711→   * Supports: ASN targeting, city-level targeting, sticky sessions,
   712→   * weighted random, domain-optimized selection, adaptive strategy,
   713→   * cost optimization, and reputation-weighted selection.
   714→   *
   715→   * This is the BACKWARD-COMPATIBLE entry point.
   716→   */
   717→  async getProxy(
   718→    domain: string,
   719→    tier: ProxyTier = 'residential',
   720→    country?: string,
   721→    strategy: RotationStrategy = 'least-failures',
   722→    options?: {
   723→      city?: string;
   724→      asn?: string;
   725→      sessionId?: string;
   726→      requireStealth?: boolean;
   727→      maxCostPerGb?: number;
   728→      preferredProvider?: ProxyProvider;
   729→    },
   730→  ): Promise<ProxySelection | null> {
   731→    try {
   732→      // -- Step 1: Check sticky session by ID ------------------------------
   733→      if (options?.sessionId) {
   734→        const stickyResult = await this.checkStickySession(
   735→          options.sessionId, tier, domain,
   736→        );
   737→        if (stickyResult) return stickyResult;
   738→      }
   739→
   740→      // -- Step 2: Check domain-optimized proxy ----------------------------
   741→      if (!options?.sessionId) {
   742→        const domainResult = await this.checkDomainOptimizedProxy(domain, tier);
   743→        if (domainResult) return domainResult;
   744→      }
   745→
   746→      // -- Step 3: Check domain intelligence for adaptive selection --------
   747→      const intel = this.getDomainIntelligence(domain);
   748→      const effectiveStrategy = this.resolveStrategy(domain, strategy, intel);
   749→
   750→      // -- Step 4: Determine if stealth or CAPTCHA routing is needed -------
   751→      const needsStealth = options?.requireStealth || intel?.requiresStealth || this.isStealthDomain(domain);
   752→      const needsCaptcha = intel?.requiresCaptcha || false;
   753→
   754→      // -- Step 5: Get available proxies with advanced filtering -----------
   755→      const effectiveTier = intel?.preferredTier || tier;
   756→      const effectiveCountry = intel?.preferredCountry || country;
   757→
   758→      const proxies = await this.getAvailableProxies(
   759→        effectiveTier, effectiveCountry, options?.city, options?.asn,
   760→      );
   761→
   762→      // -- Step 6: If stealth required, prefer residential/mobile ----------
   763→      if (needsStealth && proxies.length > 0) {
   764→        const stealthProxies = proxies.filter(
   765→          (p: any) => p.tier === 'residential' || p.tier === 'mobile' || p.tier === 'isp',
   766→        );
   767→        if (stealthProxies.length > 0) {
   768→          const selected = this.selectProxy(stealthProxies, domain, effectiveStrategy);
   769→          if (selected) {
   770→            await this.finalizeProxySelection(selected, domain, options);
   771→            return this.toSelection(selected);
   772→          }
   773→        }
   774→      }
   775→
   776→      // -- Step 7: If no internal proxies, try external sources ------------
   777→      if (proxies.length === 0) {
   778→        logger.warn(
   779→          { tier: effectiveTier, country: effectiveCountry, domain, city: options?.city, asn: options?.asn },
   780→          'No available proxies for criteria -- trying external sources',
   781→        );
   782→
   783→        // Try multiple external sources in parallel
   784→        const externalResult = await this.tryExternalSourcesWithFailover(
   785→          effectiveTier, effectiveCountry, options?.city, options?.asn, domain,
   786→        );
   787→        if (externalResult) return externalResult;
   788→      }
   789→
   790→      // -- Step 8: Select proxy based on resolved strategy -----------------
   791→      const selected = this.selectProxy(proxies, domain, effectiveStrategy);
   792→      if (!selected) return null;
   793→
   794→      // -- Step 9: Finalize selection --------------------------------------
   795→      await this.finalizeProxySelection(selected, domain, options);
   796→
   797→      return this.toSelection(selected);
   798→    } catch (err: any) {
   799→      logger.error(
   800→        { domain, tier, country, strategy, error: err.message },
   801→        'Proxy selection failed -- degrading gracefully',
   802→      );
   803→
   804→      // Emergency fallback: try to get any proxy at all
   805→      return this.emergencyFallback(domain, tier, country);
   806→    }
   807→  }

  1194→          statusCode,
  1195→          latencyMs,
  1196→        },
  1197→      });
  1198→
  1199→      // -- Update proxy health metrics -------------------------------------
  1200→      const proxy = await db.proxy.findUnique({ where: { id: proxyId } });
  1201→      if (!proxy) return;
  1202→
  1203→      const newConsecutiveFailures = success ? 0 : proxy.consecutiveFailures + 1;
  1204→
  1205→      // EMA success rate (alpha=0.1)
  1206→      const alpha = 0.1;
  1207→      const newSuccessRate = success
  1208→        ? proxy.successRate * (1 - alpha) + 1 * alpha
  1209→        : proxy.successRate * (1 - alpha);
  1210→
  1211→      // Update p95 latency (approximation)
  1212→      const newP95Latency = latencyMs > proxy.p95Latency
  1213→        ? Math.round(proxy.p95Latency * 0.9 + latencyMs * 0.1)
  1214→        : proxy.p95Latency;
  1215→
  1216→      // Decide whether to retire the proxy
  1217→      const shouldRetire = newConsecutiveFailures >= MAX_CONSECUTIVE_FAILURES
  1218→        || newSuccessRate < RETIRE_THRESHOLD;
  1219→
  1220→      await db.proxy.update({
  1221→        where: { id: proxyId },
  1222→        data: {
  1223→          successRate: Math.round(newSuccessRate * 1000) / 1000,
  1224→          p95Latency: newP95Latency,
  1225→          failures: success ? proxy.failures : proxy.failures + 1,
  1226→          consecutiveFailures: newConsecutiveFailures,
  1227→          retired: shouldRetire,
  1228→          lastChecked: new Date(),
  1229→        },
  1230→      });
  1231→
  1232→      if (shouldRetire) {
  1233→        logger.warn(
  1234→          { proxyId, domain, successRate: newSuccessRate, consecutiveFailures: newConsecutiveFailures },
  1235→          'Proxy retired due to poor performance',
  1236→        );
  1237→
  1238→        // Remove from domain optimization map
  1239→        if (this.domainProxyMap.get(domain) === proxyId) {
  1240→          this.domainProxyMap.delete(domain);
  1241→        }
  1242→
  1243→        // Report to containment shield
  1244→        this.reportRetiredProxy(proxyId, domain).catch(() => {});
  1245→      }
  1246→
  1247→      // On failure, clear domain-optimized proxy so next request uses a different one
  1248→      if (!success && this.domainProxyMap.get(domain) === proxyId) {
  1249→        this.domainProxyMap.delete(domain);
  1250→      }
  1251→
  1252→      // Invalidate cache
  1253→      await redis.del(`proxies:${proxy.tier}`);
  1254→    } catch (error: any) {
  1255→      logger.error({ error: error.message, proxyId: outcome.proxyId }, 'Failed to record proxy outcome');
  1256→    }
  1257→  }
