   130→  for (const variant of REGIONAL_VARIANTS) {
   131→    const regional: TlsProfile = {
   132→      ...base,
   133→      name: `${base.name}${variant.suffix}`,
   134→      defaultHeaders: { ...base.defaultHeaders, 'Accept-Language': variant.lang },
   135→      quality: base.quality - 5, // Slightly lower since less common
   136→    };
   137→    TLS_PROFILES.push(regional);
   138→  }
   139→}

   141→// ===============================================================================
   142→// TLS FINGERPRINT ENGINE
   143→// ===============================================================================
   144→
   145→export class TlsFingerprintEngine {
   146→  private gotScrapingAvailable = false;
   147→  private assignments = new Map<string, FingerprintAssignment>();
   148→  private domainProfileMap = new Map<string, string>();
   149→  private blockCounts = new Map<string, number>();
   150→  private successCounts = new Map<string, number>();
   151→  private rotationStrategy: RotationStrategy = 'adaptive';
   152→  private rotationInterval = 600000; // 10 minutes
   153→  private maxRequestsPerProfile = 100;
   154→
   155→  constructor() {
   156→    try {
   157→      const fs = require('fs'); const path = require('path');
   158→      this.gotScrapingAvailable = fs.existsSync(path.join(process.cwd(), 'node_modules', 'got-scraping', 'package.json'));
   159→      if (this.gotScrapingAvailable) logger.info('got-scraping available -- TLS fingerprint engine enabled');
   160→      else logger.warn('got-scraping not available -- TLS fingerprint engine in fallback mode');
   161→    } catch { this.gotScrapingAvailable = false; }
   162→  }
   163→
   164→  get isAvailable(): boolean { return this.gotScrapingAvailable; }
   165→
   166→  get profileCount(): number { return TLS_PROFILES.length; }
   167→
   168→  /**
   169→   * Fetch a URL with managed TLS fingerprint.
   170→   */
   171→  async fetch(url: string, proxyUrl?: string, options?: TlsFetchOptions, fallbackFn?: (url: string, proxyUrl?: string, opts?: any) => Promise<{ text: string; status: number; headers: Record<string, string>; url: string; ok: boolean }>): Promise<TlsFetchResult> {
   172→    const domain = new URL(url).hostname;
   173→    const profile = this.selectProfile(domain, options);
   174→
   175→    // Track assignment
   176→    this.trackAssignment(profile, domain);
   177→
   178→    if (!this.gotScrapingAvailable) {
   179→      if (fallbackFn) {
   180→        const result = await fallbackFn(url, proxyUrl, options);
   181→        return { ...result, profileUsed: profile.name };
   182→      }
   183→      throw new Error('got-scraping not available and no fallback provided');
   184→    }
   185→
   186→    try {
   187→      const { gotScraping } = await import('got-scraping');
   188→      const mergedHeaders = { ...profile.defaultHeaders, 'User-Agent': profile.userAgent, ...options?.headers };
   189→
   190→      const gotOptions: any = {
   191→        url, method: options?.method || 'GET', headers: mergedHeaders,
   192→        timeout: { request: options?.timeout || 30000 },
   193→        followRedirect: true, maxRedirects: 5,
   194→        http2: profile.httpVersion === 2,
   195→        headerGeneratorOptions: {
   196→          browsers: [{ name: profile.browser === 'edge' ? 'chrome' : profile.browser === 'android-chrome' ? 'chrome' : profile.browser === 'ios-safari' ? 'safari' : profile.browser === 'samsung-internet' ? 'chrome' : profile.browser }],
   197→          devices: [profile.mobile ? 'mobile' : 'desktop'],
   198→          operatingSystems: profile.platform.includes('Win') ? ['windows'] : profile.platform.includes('Mac') ? ['macos'] : profile.platform.includes('iPhone') ? ['ios'] : ['linux'],
   199→          locales: ['en-US', 'en'],
   200→        },
   201→      };
   202→
   203→      if (proxyUrl) gotOptions.proxyUrl = proxyUrl;
   204→      if (options?.body) gotOptions.body = options.body;
   205→
   206→      const response = await gotScraping(gotOptions);
   207→      const responseHeaders: Record<string, string> = {};
   208→      if (response.headers) { for (const [key, value] of Object.entries(response.headers)) { if (value) responseHeaders[key] = Array.isArray(value) ? value.join(', ') : String(value); } }
   209→
   210→      this.recordSuccess(profile.name);
   211→      return { text: response.body, status: response.statusCode, headers: responseHeaders, url: response.url || url, ok: response.statusCode >= 200 && response.statusCode < 400, profileUsed: profile.name, ja3Used: profile.ja3Hash };
   212→    } catch (err: any) {
   213→      if (err.response) {
   214→        const isBlock = [403, 429, 503].includes(err.response.statusCode);
   215→        if (isBlock) this.recordBlock(profile.name);
   216→        const responseHeaders: Record<string, string> = {};
   217→        if (err.response.headers) { for (const [key, value] of Object.entries(err.response.headers)) { if (value) responseHeaders[key] = Array.isArray(value) ? value.join(', ') : String(value); } }
   218→        return { text: err.response.body || '', status: err.response.statusCode || 502, headers: responseHeaders, url: err.response.url || url, ok: false, profileUsed: profile.name, ja3Used: profile.ja3Hash };
   219→      }
   220→
   221→      if (fallbackFn) {
   222→        logger.debug({ url, error: err.message }, 'got-scraping failed -- using fallback');
   223→        const result = await fallbackFn(url, proxyUrl, options);
   224→        return { ...result, profileUsed: `${profile.name}-fallback` };
   225→      }
   226→      throw err;
   227→    }
   228→  }
   229→
   230→  /**
   231→   * Select the best TLS profile for a given domain.
   232→   */
   233→  selectProfile(domain: string, options?: TlsFetchOptions): TlsProfile {
   234→    // If specific profile requested
   235→    if (options?.profile) {
   236→      const specific = options.profileVersion
   237→        ? TLS_PROFILES.find(p => p.browser === options.profile && p.version === options.profileVersion)
   238→        : TLS_PROFILES.find(p => p.browser === options.profile);
   239→      if (specific) return specific;
   240→    }
   241→
   242→    // Domain-based assignment (session stickiness)
   243→    const strategy = options?.rotationStrategy || this.rotationStrategy;
   244→    if (strategy === 'per-domain' || strategy === 'per-session' || strategy === 'adaptive') {
   245→      const assigned = this.domainProfileMap.get(domain);
   246→      if (assigned) {
   247→        const assignment = this.assignments.get(assigned);
   248→        if (assignment && this.shouldKeepAssignment(assignment)) {
   249→          return assignment.profile;
   250→        }
   251→        // Assignment expired or blocked -- rotate
   252→        this.domainProfileMap.delete(domain);
   253→        this.assignments.delete(assigned);
   254→      }
   255→    }
   256→
   257→    // Select new profile using weighted quality scoring
   258→    const candidates = TLS_PROFILES.filter(p => {
   259→      const blocks = this.blockCounts.get(p.name) || 0;
   260→      return blocks < 5; // Skip profiles with too many blocks
   261→    });
   262→
   263→    if (candidates.length === 0) {
   264→      // All profiles blocked -- reset and use best quality
   265→      this.blockCounts.clear();
   266→      return this.getBestQualityProfile();
   267→    }
   268→
   269→    // Weighted selection: higher quality = more likely
   270→    const weights = candidates.map(p => p.quality * (1 - (this.blockCounts.get(p.name) || 0) * 0.1));
   271→    const totalWeight = weights.reduce((a, b) => a + Math.max(b, 1), 0);
   272→    let rand = Math.random() * totalWeight;
   273→    for (let i = 0; i < candidates.length; i++) {
   274→      rand -= Math.max(weights[i], 1);
   275→      if (rand <= 0) return candidates[i];
   276→    }
   277→
   278→    return candidates[0];
   279→  }
   280→
   281→  private shouldKeepAssignment(assignment: FingerprintAssignment): boolean {
   282→    const age = Date.now() - assignment.assignedAt;
   283→    if (age > this.rotationInterval) return false;
   284→    if (assignment.requestCount > this.maxRequestsPerProfile) return false;
   285→    if ((this.blockCounts.get(assignment.profile.name) || 0) > 3) return false;
   286→    return true;
   287→  }
   288→
   289→  private getBestQualityProfile(): TlsProfile {
   290→    return TLS_PROFILES.reduce((best, p) => p.quality > best.quality ? p : best, TLS_PROFILES[0]);
   291→  }
   292→
   293→  private trackAssignment(profile: TlsProfile, domain: string): void {
   294→    const key = `${profile.name}-${domain}`;
   295→    let assignment = this.assignments.get(key);
   296→    if (!assignment) {
   297→      assignment = { profile, assignedAt: Date.now(), requestCount: 0, lastUsed: Date.now(), successCount: 0, blockCount: 0, domains: new Set([domain]) };
   298→      this.assignments.set(key, assignment);
   299→    }
   300→    assignment.requestCount++;
   301→    assignment.lastUsed = Date.now();
   302→    this.domainProfileMap.set(domain, key);
   303→  }
   304→
   305→  recordSuccess(profileName: string): void {
   306→    this.successCounts.set(profileName, (this.successCounts.get(profileName) || 0) + 1);
   307→  }
   308→
   309→  recordBlock(profileName: string): void {
   310→    this.blockCounts.set(profileName, (this.blockCounts.get(profileName) || 0) + 1);
   311→    logger.debug({ profile: profileName, totalBlocks: this.blockCounts.get(profileName) }, 'Profile blocked -- considering rotation');
   312→  }
   313→
   314→  /**
   315→   * Get a random TLS profile.
   316→   */
   317→  getRandomProfile(): TlsProfile {
   318→    return TLS_PROFILES[Math.floor(Math.random() * TLS_PROFILES.length)];
   319→  }
   320→
   321→  getProfile(name: string): TlsProfile | undefined {
   322→    return TLS_PROFILES.find(p => p.name === name);
   323→  }
   324→
   325→  getAvailableProfiles(): string[] {
   326→    return TLS_PROFILES.map(p => p.name);
   327→  }
   328→
   329→  getUserAgent(profile?: TlsBrowserProfile): string {
   330→    const p = profile ? TLS_PROFILES.find(pr => pr.browser === profile) : this.getRandomProfile();
   331→    return p?.userAgent || TLS_PROFILES[0].userAgent;
   332→  }
   333→
   334→  /**
   335→   * Get fingerprint engine statistics.
   336→   */
   337→  getStats(): Record<string, any> {
   338→    const topBlocked = [...this.blockCounts.entries()].sort((a, b) => b[1] - a[1]).slice(0, 10);
   339→    const topSuccess = [...this.successCounts.entries()].sort((a, b) => b[1] - a[1]).slice(0, 10);
   340→    return {
   341→      totalProfiles: TLS_PROFILES.length,
   342→      activeAssignments: this.assignments.size,
   343→      domainMappings: this.domainProfileMap.size,
   344→      rotationStrategy: this.rotationStrategy,
   345→      gotScrapingAvailable: this.gotScrapingAvailable,
   346→      topBlockedProfiles: Object.fromEntries(topBlocked),
   347→      topSuccessfulProfiles: Object.fromEntries(topSuccess),
   348→    };
   349→  }
   350→
   351→  setRotationStrategy(strategy: RotationStrategy): void { this.rotationStrategy = strategy; }
   352→  setRotationInterval(ms: number): void { this.rotationInterval = ms; }
   353→  setMaxRequestsPerProfile(n: number): void { this.maxRequestsPerProfile = n; }
   354→
   355→  /**
   356→   * Reset all block counts -- useful after cooldown periods.
   357→   */
   358→  resetBlockCounts(): void {
   359→    this.blockCounts.clear();
   360→    logger.info('All TLS profile block counts reset');
   361→  }
   362→}
