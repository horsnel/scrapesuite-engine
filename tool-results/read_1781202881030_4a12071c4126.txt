   660→    try {
   661→      // CDP is available when running Chromium via Playwright
   662→      // Check if we can create a CDP session
   663→      this.cdpAvailable = true; // Will be verified on first use

   672→  /**
   673→   * Apply CDP-level fingerprint injection to a Playwright page.
   674→   * This is the main entry point. Returns detailed results about what was applied.
   675→   */
   676→  async inject(page: Page, domain: string, options?: {
   677→    sessionId?: string;
   678→    profile?: CoherentProfile;
   679→    injectionLevel?: CdpInjectionLevel;
   680→    config?: Partial<CdpOverrideConfig>;
   681→  }): Promise<CdpInjectionResult> {
   682→    const errors: string[] = [];
   683→    const overridesApplied: string[] = [];
   684→    const effectiveConfig = { ...this.config, ...options?.config };
   685→
   686→    // Get or create profile for this domain
   687→    const profile = this.getOrCreateProfile(domain, options?.sessionId, options?.profile);
   688→    const coherentProfile = profile.coherentProfile;
   689→    let level = options?.injectionLevel || 'full';
   690→    let fallback: CdpFallbackMode = 'none';
   691→    let cdpSessionCreated = false;
   692→
   693→    // Step 1: Get CDP session from the page
   694→    let cdpSession: CDPSession | null = null;
   695→    try {
   696→      cdpSession = await page.context().newCDPSession(page);
   697→      cdpSessionCreated = true;
   698→      this.cdpAvailable = true;
   699→    } catch (err: any) {
   700→      errors.push(`CDP session creation failed: ${err.message}`);
   701→      logger.debug({ error: err.message, domain }, 'CDP session creation failed -- falling back to addInitScript');
   702→      level = 'partial';
   703→      fallback = 'addInitScript';
   704→    }
   705→
   706→    // Step 2: Apply CDP-level overrides (if session available)
   707→    if (cdpSession) {
   708→      try {
   709→        // User-Agent override at CDP level
   710→        if (effectiveConfig.userAgent) {
   711→          await cdpSession.send('Network.setUserAgentOverride', {
   712→            userAgent: coherentProfile.userAgent,
   713→            platform: coherentProfile.platform,
   714→            acceptLanguage: `${coherentProfile.locale},en;q=0.9`,
   715→          });
   716→          overridesApplied.push('userAgent-cdp');
   717→        }
   718→      } catch (err: any) {
   719→        errors.push(`UserAgent CDP override failed: ${err.message}`);
   720→      }
   721→
   722→      try {
   723→        // Device metrics override at CDP level
   724→        if (effectiveConfig.deviceMetrics) {
   725→          const isMobile = coherentProfile.os === 'android' || coherentProfile.os === 'ios';
   726→          const deviceScaleFactor = isMobile
   727→            ? (coherentProfile.screenResolution.width / coherentProfile.viewport.width)
   728→            : (coherentProfile.screenResolution.width > 1920 ? 2 : 1);
   729→
   730→          await cdpSession.send('Emulation.setDeviceMetricsOverride', {
   731→            width: coherentProfile.viewport.width,
   732→            height: coherentProfile.viewport.height,
   733→            deviceScaleFactor: Math.round(deviceScaleFactor * 100) / 100,
   734→            mobile: isMobile,
   735→            screenWidth: coherentProfile.screenResolution.width,
   736→            screenHeight: coherentProfile.screenResolution.height,
   737→            dontSetVisibleSize: true,
   738→          } as any);
   739→          overridesApplied.push('deviceMetrics-cdp');
   740→        }
   741→      } catch (err: any) {
   742→        errors.push(`DeviceMetrics CDP override failed: ${err.message}`);
   743→      }
   744→
   745→      try {
   746→        // Timezone override at CDP level
   747→        if (effectiveConfig.timezone) {
   748→          await cdpSession.send('Emulation.setTimezoneOverride', {
   749→            timezoneId: coherentProfile.timezone,
   750→          });
   751→          overridesApplied.push('timezone-cdp');
   752→        }
   753→      } catch (err: any) {
   754→        errors.push(`Timezone CDP override failed: ${err.message}`);
   755→      }
   756→
   757→      try {
   758→        // Geolocation override at CDP level
   759→        if (effectiveConfig.geolocation) {
   760→          const geo = TIMEZONE_GEO[coherentProfile.timezone];
   761→          if (geo) {
   762→            await cdpSession.send('Emulation.setGeolocationOverride', {
   763→              latitude: geo.latitude + (Math.random() - 0.5) * 0.01,
   764→              longitude: geo.longitude + (Math.random() - 0.5) * 0.01,
   765→              accuracy: geo.accuracy,
   766→            });
   767→            overridesApplied.push('geolocation-cdp');
   768→          }
   769→        }
   770→      } catch (err: any) {
   771→        errors.push(`Geolocation CDP override failed: ${err.message}`);
   772→      }
   773→
   774→      try {
   775→        // Locale override at CDP level
   776→        if (effectiveConfig.language) {
   777→          await cdpSession.send('Emulation.setLocaleOverride', {
   778→            locale: coherentProfile.locale,
   779→          });
   780→          overridesApplied.push('locale-cdp');
   781→        }
   782→      } catch (err: any) {
   783→        errors.push(`Locale CDP override failed: ${err.message}`);
   784→      }
   785→
   786→      try {
   787→        // Network-level extra HTTP headers (sec-ch-ua, etc.)
   788→        if (effectiveConfig.httpHeaders) {
   789→          const secChHeaders = buildSecChUaHeaders(coherentProfile);
   790→          const extraHeaders: Record<string, string> = {
   791→            ...secChHeaders,
   792→            'Accept-Language': `${coherentProfile.locale},en;q=0.9`,
   793→            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8',
   794→            'Accept-Encoding': 'gzip, deflate, br',
   795→            'Sec-Fetch-Dest': 'document',
   796→            'Sec-Fetch-Mode': 'navigate',
   797→            'Sec-Fetch-Site': 'none',
   798→            'Sec-Fetch-User': '?1',
   799→            'Upgrade-Insecure-Requests': '1',
   800→          };
   801→
   802→          await cdpSession.send('Network.setExtraHTTPHeaders', {
   803→            headers: extraHeaders,
   804→          });
   805→          overridesApplied.push('httpHeaders-cdp');
   806→        }
   807→      } catch (err: any) {
   808→        errors.push(`HTTP headers CDP override failed: ${err.message}`);
   809→      }
   810→
   811→      try {
   812→        // Touch emulation at CDP level
   813→        if (coherentProfile.touchSupport) {
   814→          await cdpSession.send('Emulation.setTouchEmulationEnabled', {
   815→            enabled: true,
   816→            maxTouchPoints: coherentProfile.maxTouchPoints,
   817→          });
   818→          overridesApplied.push('touchEmulation-cdp');
   819→        }
   820→      } catch (err: any) {
   821→        errors.push(`Touch emulation CDP override failed: ${err.message}`);
   822→      }
   823→
   824→      // Store CDP session for later use (network interception, etc.)
   825→      const sessionState: CdpSessionState = {
   826→        sessionId: profile.sessionId,
   827→        cdpSession,
   828→        page,
   829→        context: page.context(),
   830→        profile,
   831→        createdAt: Date.now(),
   832→        lastActivity: Date.now(),
   833→        active: true,
   834→        interceptionEnabled: false,
   835→        scriptsInjected: false,
   836→      };
   837→      this.sessions.set(profile.sessionId, sessionState);
   838→    }
   839→
   840→    // Step 3: Inject V8-level scripts via Page.addScriptToEvaluateOnNewDocument
   841→    // This runs BEFORE any page JavaScript and is harder to detect than addInitScript
   842→    try {
   843→      const v8Script = generateV8InjectionScript(coherentProfile, effectiveConfig);
   844→      await page.addInitScript(v8Script);
   845→      overridesApplied.push('v8InjectionScript');
   846→    } catch (err: any) {
   847→      errors.push(`V8 injection script failed: ${err.message}`);
   848→    }
   849→
   850→    // Step 4: If CDP wasn't available, fall back to context-level overrides
   851→    if (level === 'partial' && fallback === 'addInitScript') {
   852→      try {
   853→        const context = page.context();
   854→        await fingerprintConsistencyEngine.applyToContext(context, coherentProfile);
   855→        overridesApplied.push('fingerprintConsistency-fallback');
   856→      } catch (err: any) {
   857→        errors.push(`Fingerprint consistency fallback failed: ${err.message}`);
   858→        fallback = 'basic';
   859→      }
   860→    }
   861→
   862→    // Step 5: Enable network request interception if configured
   863→    if (cdpSession && effectiveConfig.networkInterception) {
   864→      try {
   865→        await this.enableNetworkInterception(cdpSession, coherentProfile);
   866→        overridesApplied.push('networkInterception-cdp');
   867→      } catch (err: any) {
   868→        errors.push(`Network interception CDP setup failed: ${err.message}`);
   869→      }
   870→    }
   871→
   872→    // Update profile tracking
   873→    profile.lastUsed = Date.now();
   874→    profile.requestCount++;
   875→    profile.cdpSessionActive = cdpSessionCreated;
   876→
   877→    logger.info({
   878→      domain,
   879→      profileId: profile.id,
   880→      level,
   881→      fallback,
   882→      overridesApplied: overridesApplied.length,
   883→      errors: errors.length,
   884→      cdpSessionCreated,
   885→    }, 'CDP injection applied');
   886→
   887→    return {
   888→      success: overridesApplied.length > 0,
   889→      level,
   890→      fallbackUsed: fallback,
   891→      profileId: profile.id,
   892→      overridesApplied,
   893→      errors,
   894→      cdpSessionCreated,
   895→    };
   896→  }

   898→  /**
   899→   * Enable network request interception at CDP level.
   900→   * This allows us to modify request headers on-the-fly for header order spoofing
   901→   * and response header modification for anti-detection.
   902→   */
   903→  private async enableNetworkInterception(cdpSession: CDPSession, profile: CoherentProfile): Promise<void> {
   904→    await cdpSession.send('Fetch.enable', {
   905→      patterns: [
   906→        { urlPattern: '*', requestStage: 'Request' },
   907→      ],
   908→      handleAuthRequests: false,
   909→    });
   910→
   911→    cdpSession.on('Fetch.requestPaused', async (event: any) => {
   912→      try {
   913→        const { requestId, request } = event;
   914→
   915→        // Modify headers for header order consistency
   916→        const modifiedHeaders: Record<string, string> = { ...request.headers };
   917→
   918→        // Ensure sec-ch-ua headers are present and consistent
   919→        const secChHeaders = buildSecChUaHeaders(profile);
   920→        for (const [key, value] of Object.entries(secChHeaders)) {
   921→          modifiedHeaders[key.toLowerCase()] = value;
   922→        }
   923→
   924→        // Ensure Accept-Language is consistent
   925→        modifiedHeaders['accept-language'] = `${profile.locale},en;q=0.9`;
   926→
   927→        // Remove any automation-related headers
   928→        delete modifiedHeaders['x-playwright'];
   929→        delete modifiedHeaders['x-puppeteer'];
   930→
   931→        await cdpSession.send('Fetch.continueRequest', {
   932→          requestId,
   933→          headers: Object.entries(modifiedHeaders).map(([name, value]) => ({ name, value })),
   934→        });
   935→      } catch (err: any) {
   936→        // Silently continue if interception fails -- don't break the request
   937→        try {
   938→          await cdpSession.send('Fetch.continueRequest', { requestId: event.requestId });
   939→        } catch {}
   940→      }
   941→    });
   942→  }

   946→  private getOrCreateProfile(domain: string, sessionId?: string, profile?: CoherentProfile): CdpProfile {
   947→    // Check for existing domain assignment
   948→    const existing = this.domainProfiles.get(domain);
   949→    if (existing && existing.blockCount < 5) {
   950→      return existing;
   951→    }

   954→    const coherentProfile = profile || fingerprintConsistencyEngine.getProfile(domain, sessionId);
   955→    const id = `cdp-${domain}-${Date.now()}`;
   956→    const sid = sessionId || `cdp-sess-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   957→
   958→    const cdpProfile: CdpProfile = {
   959→      id,
   960→      coherentProfile,
   961→      injectionLevel: 'full',
   962→      fallbackMode: 'none',
   963→      domain,
   964→      sessionId: sid,
   965→      createdAt: Date.now(),
   966→      lastUsed: Date.now(),
   967→      requestCount: 0,
   968→      blockCount: 0,
   969→      successCount: 0,
   970→      cdpSessionActive: false,
   971→    };
   972→
   973→    this.domainProfiles.set(domain, cdpProfile);

  1023→   */
  1024→  rotateProfile(domain: string): CdpProfile {
  1025→    this.domainProfiles.delete(domain);
  1026→    return this.getOrCreateProfile(domain);
  1027→  }

  1029→  // --- Cleanup ----------------------------------------------------------------
  1030→
  1031→  private startCleanup(): void {
  1032→    this.cleanupInterval = setInterval(() => this.cleanup(), 60000);
  1033→  }
  1034→
  1035→  private cleanup(): void {
  1036→    const now = Date.now();
  1037→
  1038→    // Clean up expired sessions
  1039→    for (const [sessionId, state] of this.sessions) {
  1040→      if (now - state.lastActivity > this.sessionTtl || !state.active) {
  1041→        try {
  1042→          state.cdpSession?.detach();
  1043→        } catch {}
  1044→        this.sessions.delete(sessionId);
  1045→      }
  1046→    }
  1047→
  1048→    // Clean up expired domain profiles
  1049→    for (const [domain, profile] of this.domainProfiles) {
  1050→      if (now - profile.lastUsed > this.sessionTtl) {
  1051→        this.domainProfiles.delete(domain);
  1052→      }
  1053→    }
  1054→
  1055→    // Limit total sessions
  1056→    if (this.sessions.size > this.maxSessions) {
  1057→      const sorted = [...this.sessions.entries()]
  1058→        .sort((a, b) => a[1].lastActivity - b[1].lastActivity);
  1059→      const toRemove = sorted.slice(0, this.sessions.size - this.maxSessions);
  1060→      for (const [sessionId, state] of toRemove) {
  1061→        try {
  1062→          state.cdpSession?.detach();
  1063→        } catch {}
  1064→        this.sessions.delete(sessionId);
  1065→      }
  1066→    }
  1067→  }
  1068→
  1069→  /**
  1070→   * Destroy the engine and clean up all sessions.
  1071→   */
  1072→  async destroy(): Promise<void> {
  1073→    if (this.cleanupInterval) {
  1074→      clearInterval(this.cleanupInterval);
  1075→      this.cleanupInterval = null;
  1076→    }
  1077→
  1078→    for (const [sessionId, state] of this.sessions) {
  1079→      try {
  1080→        state.active = false;
  1081→        await state.cdpSession?.detach();
  1082→      } catch {}
  1083→    }
  1084→    this.sessions.clear();
  1085→    this.domainProfiles.clear();
  1086→    logger.info('CDP Injection Engine destroyed');
  1087→  }
