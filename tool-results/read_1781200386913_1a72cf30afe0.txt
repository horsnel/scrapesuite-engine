    83→    83→    83→  /**
    84→    84→    84→   * Generate a complete Reddit browsing session with realistic interactions.
    85→    85→    85→   *
    86→    86→    86→   * A session includes:
    87→    87→    87→   *   - Entry point (usually a subreddit listing or the front page)
    88→    88→    88→   *   - Scrolling through posts with reading pauses
    89→    89→    89→   *   - Clicking into posts to read content/comments
    90→    90→    90→   *   - Voting on some posts and comments
    91→    91→    91→   *   - Navigating between sections (hot → comments → back)
    92→    92→    92→   *   - Optional search behavior
    93→    93→    93→   *   - Break periods for realism
    94→    94→    94→   *
    95→    95→    95→   * @param config - Optional override for interaction config
    96→    96→    96→   * @returns A complete browsing session descriptor

    98→    98→    98→  generateBrowsingSession(config?: Partial<RedditInteractionConfig>): RedditBrowsingSession {
    99→    99→    99→    const effectiveConfig = config
   100→   100→   100→      ? { ...this.config, ...config }

   102→   102→   102→
   103→   103→   103→    const sessionId = this.generateSessionId();

   128→   128→   128→    const session: RedditBrowsingSession = {

   170→   170→   170→
   171→   171→   171→  /**
   172→   172→   172→   * Generate a scroll pattern for a specific Reddit section.
   173→   173→   173→   *
   174→   174→   174→   * Different sections have different natural scroll behaviors:
   175→   175→   175→   *   - Listing pages: Continuous scroll with periodic pauses
   176→   176→   176→   *   - Comment threads: Scroll-and-expand with depth reading
   177→   177→   177→   *   - Search results: Faster scanning, fewer pauses
   178→   178→   178→   *   - User profiles: Variable based on content type
   179→   179→   179→   *
   180→   180→   180→   * @param section - The Reddit section to generate scroll patterns for
   181→   181→   181→   * @returns Array of scroll actions with timing
   182→   182→   182→   */
   183→   183→   183→  generateScrollPattern(section: RedditScrapeTarget): Array<{
   184→   184→   184→    direction: 'down' | 'up';
   185→   185→   185→    distancePx: number;
   186→   186→   186→    delayAfterMs: number;
   187→   187→   187→    isPause: boolean;
   188→   188→   188→  }> {
   189→   189→   189→    const patterns: Array<{
   190→   190→   190→      direction: 'down' | 'up';
   191→   191→   191→      distancePx: number;
   192→   192→   192→      delayAfterMs: number;
   193→   193→   193→      isPause: boolean;
   194→   194→   194→    }> = [];
   195→   195→   195→
   196→   196→   196→    let totalDistance = 0;
   197→   197→   197→    const maxScrollDistance = this.getMaxScrollDistance(section);
   198→   198→   198→    const scrollCount = this.randomBetween(5, Math.min(25, MAX_SIMULATION_ITEMS));
   199→   199→   199→
   200→   200→   200→    for (let i = 0; i < scrollCount && totalDistance < maxScrollDistance; i++) {
   201→   201→   201→      // Direction — mostly down, occasional up
   202→   202→   202→      const direction: 'down' | 'up' = Math.random() < this.config.scrollPatterns.scrollBackProbability && i > 2
   203→   203→   203→        ? 'up'
   204→   204→   204→        : 'down';
   205→   205→   205→
   206→   206→   206→      // Distance — variable speed
   207→   207→   207→      const distance = this.randomBetween(
   208→   208→   208→        this.config.scrollPatterns.scrollSpeedRange.min,
   209→   209→   209→        this.config.scrollPatterns.scrollSpeedRange.max
   210→   210→   210→      );
   211→   211→   211→
   212→   212→   212→      // Delay after scroll
   213→   213→   213→      const baseDelay = this.randomBetween(
   214→   214→   214→        this.config.scrollPatterns.scrollIntervalRange.min,
   215→   215→   215→        this.config.scrollPatterns.scrollIntervalRange.max
   216→   216→   216→      );
   217→   217→   217→
   218→   218→   218→      // Reading pause — occurs at natural break points
   219→   219→   219→      const isPause = Math.random() < this.config.scrollPatterns.pauseProbability;
   220→   220→   220→      const pauseExtra = isPause
   221→   221→   221→        ? this.randomBetween(
   222→   222→   222→            this.config.scrollPatterns.pauseDurationRange.min,
   223→   223→   223→            this.config.scrollPatterns.pauseDurationRange.max
   224→   224→   224→          )
   225→   225→   225→        : 0;
   226→   226→   226→
   227→   227→   227→      patterns.push({
   228→   228→   228→        direction,
   229→   229→   229→        distancePx: direction === 'down' ? distance : Math.min(distance, totalDistance),
   230→   230→   230→        delayAfterMs: baseDelay + pauseExtra,
   231→   231→   231→        isPause,
   232→   232→   232→      });
   233→   233→   233→
   234→   234→   234→      totalDistance += direction === 'down' ? distance : -distance;
   235→   235→   235→      totalDistance = Math.max(0, totalDistance);
   236→   236→   236→    }
   237→   237→   237→
   238→   238→   238→    // Section-specific modifications
   239→   239→   239→    if (section === 'comments') {
   240→   240→   240→      // Comment sections have longer pauses (reading)
   241→   241→   241→      for (const pattern of patterns) {
   242→   242→   242→        if (pattern.isPause) {
   243→   243→   243→          pattern.delayAfterMs *= 1.5; // 50% longer pauses for comments
   244→   244→   244→        }
   245→   245→   245→      }
   246→   246→   246→    } else if (section === 'search') {
   247→   247→   247→      // Search results get faster scanning
   248→   248→   248→      for (const pattern of patterns) {
   249→   249→   249→        pattern.delayAfterMs = Math.round(pattern.delayAfterMs * 0.7);
   250→   250→   250→      }
   251→   251→   251→    }
   252→   252→   252→
   253→   253→   253→    this.stats.totalScrolls += patterns.length;
   254→   254→   254→
   255→   255→   255→    logger.debug({
   256→   256→   256→      section,
   257→   257→   257→      scrollCount: patterns.length,
   258→   258→   258→      pauseCount: patterns.filter(p => p.isPause).length,
   259→   259→   259→      totalDistance,
   260→   260→   260→    }, 'Generated scroll pattern');
   261→   261→   261→
   262→   262→   262→    return patterns;
   263→   263→   263→  }
   264→   264→   264→
   265→   265→   265→  // ===========================================================================
   266→   266→   266→  // VOTE PATTERN GENERATION
   267→   267→   267→  // ===========================================================================
   268→   268→   268→
   269→   269→   269→  /**
   270→   270→   270→   * Generate realistic upvote/downvote patterns for a set of posts.
   271→   271→   271→   *
   272→   272→   272→   * Real Reddit user behavior shows:
   273→   273→   273→   *   - ~80% of votes are upvotes (positive bias)
   274→   274→   274→   *   - ~10% are downvotes (rare but necessary for realism)
   275→   275→   275→   *   - ~10% skip voting entirely
   276→   276→   276→   *   - Vote timing correlates with content reading time
   277→   277→   277→   *   - Occasional vote undos (rare, ~2%)
   278→   278→   278→   *   - Comment votes are less frequent than post votes
   279→   279→   279→   *
   280→   280→   280→   * @param postCount - Number of posts to generate vote decisions for
   281→   281→   281→   * @returns Array of vote decisions
   282→   282→   282→   */
   283→   283→   283→  generateVotePattern(postCount: number): Array<{
   284→   284→   284→    direction: 'up' | 'down' | 'skip' | 'undo';
   285→   285→   285→    delayMs: number;
   286→   286→   286→    isCommentVote: boolean;
   287→   287→   287→  }> {
   288→   288→   288→    const votes: Array<{
   289→   289→   289→      direction: 'up' | 'down' | 'skip' | 'undo';
   290→   290→   290→      delayMs: number;
   291→   291→   291→      isCommentVote: boolean;
   292→   292→   292→    }> = [];
   293→   293→   293→
   294→   294→   294→    for (let i = 0; i < Math.min(postCount, MAX_SIMULATION_ITEMS); i++) {
   295→   295→   295→      const rand = Math.random();
   296→   296→   296→
   297→   297→   297→      // Determine vote direction based on configured probabilities
   298→   298→   298→      let direction: 'up' | 'down' | 'skip';
   299→   299→   299→      if (rand < this.config.voteBehavior.upvoteProbability) {
   300→   300→   300→        direction = 'up';
   301→   301→   301→      } else if (rand < this.config.voteBehavior.upvoteProbability + this.config.voteBehavior.downvoteProbability) {
   302→   302→   302→        direction = 'down';
   303→   303→   303→      } else {
   304→   304→   304→        direction = 'skip';
   305→   305→   305→      }
   306→   306→   306→
   307→   307→   307→      // Reading time before voting (correlated with content length)
   308→   308→   308→      const viewTime = this.randomBetween(
   309→   309→   309→        this.config.voteBehavior.minViewTimeBeforeVote,
   310→   310→   310→        this.config.voteBehavior.maxViewTimeBeforeVote
   311→   311→   311→      );
   312→   312→   312→
   313→   313→   313→      // Determine if this is a comment vote (less common)
   314→   314→   314→      const isCommentVote = Math.random() < this.config.voteBehavior.commentVoteProbability;
   315→   315→   315→
   316→   316→   316→      votes.push({
   317→   317→   317→        direction,
   318→   318→   318→        delayMs: viewTime,
   319→   319→   319→        isCommentVote,
   320→   320→   320→      });
   321→   321→   321→
   322→   322→   322→      // Occasional undo vote
   323→   323→   323→      if (direction !== 'skip' && Math.random() < this.config.voteBehavior.undoVoteProbability) {
   324→   324→   324→        votes.push({
   325→   325→   325→          direction: 'undo',
   326→   326→   326→          delayMs: this.randomBetween(3000, 10000), // Undo after a delay
   327→   327→   327→          isCommentVote,
   328→   328→   328→        });
   329→   329→   329→      }
   330→   330→   330→    }
   331→   331→   331→
   332→   332→   332→    this.stats.totalVotes += votes.filter(v => v.direction !== 'skip').length;
   333→   333→   333→
   334→   334→   334→    logger.debug({
   335→   335→   335→      postCount,
   336→   336→   336→      upvotes: votes.filter(v => v.direction === 'up').length,
   337→   337→   337→      downvotes: votes.filter(v => v.direction === 'down').length,
   338→   338→   338→      skips: votes.filter(v => v.direction === 'skip').length,
   339→   339→   339→      undos: votes.filter(v => v.direction === 'undo').length,
   340→   340→   340→    }, 'Generated vote pattern');
   341→   341→   341→
   342→   342→   342→    return votes;
   343→   343→   343→  }
   344→   344→   344→
   345→   345→   345→  // ===========================================================================
   346→   346→   346→  // COMMENT READING PATTERN GENERATION
   347→   347→   347→  // ===========================================================================
   348→   348→   348→
   349→   349→   349→  /**
   350→   350→   350→   * Generate comment reading behavior for a Reddit comment thread.
   351→   351→   351→   *
   352→   352→   352→   * Real comment reading behavior:
   353→   353→   353→   *   - Top-level comments get the most attention
   354→   354→   354→   *   - Deeper comments get progressively less reading time
   355→   355→   355→   *   - Some collapsed comments get expanded (curiosity)
   356→   356→   356→   *   - Long reply chains may be partially read
   357→   357→   357→   *   - Reading time correlates with word count
   358→   358→   358→   *
   359→   359→   359→   * @param commentCount - Total number of comments in the thread
   360→   360→   360→   * @returns Array of comment reading actions with timing
   361→   361→   361→   */
   362→   362→   362→  generateCommentReadingPattern(commentCount: number): Array<{
   363→   363→   363→    depth: number;
   364→   364→   364→    readTimeMs: number;
   365→   365→   365→    isExpanded: boolean;
   366→   366→   366→    isSkipped: boolean;
   367→   367→   367→  }> {
   368→   368→   368→    const actions: Array<{
   369→   369→   369→      depth: number;
   370→   370→   370→      readTimeMs: number;
   371→   371→   371→      isExpanded: boolean;
   372→   372→   372→      isSkipped: boolean;
   373→   373→   373→    }> = [];
   374→   374→   374→
   375→   375→   375→    const effectiveCommentCount = Math.min(commentCount, MAX_SIMULATION_ITEMS);
   376→   376→   376→
   377→   377→   377→    for (let i = 0; i < effectiveCommentCount; i++) {
   378→   378→   378→      // Determine depth — weighted toward top-level comments
   379→   379→   379→      const depth = this.weightedDepth();
   380→   380→   380→
   381→   381→   381→      // Skip probability increases with depth
   382→   382→   382→      const skipProbability = Math.min(0.6, depth * 0.12);
   383→   383→   383→      const isSkipped = Math.random() < skipProbability;
   384→   384→   384→
   385→   385→   385→      if (isSkipped) {
   386→   386→   386→        actions.push({
   387→   387→   387→          depth,
   388→   388→   388→          readTimeMs: 0,
   389→   389→   389→          isExpanded: false,
   390→   390→   390→          isSkipped: true,
   391→   391→   391→        });
   392→   392→   392→        continue;
   393→   393→   393→      }
   394→   394→   394→
   395→   395→   395→      // Reading time based on comment length and depth
   396→   396→   396→      const wordCount = this.randomBetween(
   397→   397→   397→        Math.max(5, AVG_COMMENT_WORDS - 20),
   398→   398→   398→        AVG_COMMENT_WORDS + 40
   399→   399→   399→      );
   400→   400→   400→      const baseReadTimeMs = Math.max(
   401→   401→   401→        MIN_READING_TIME_MS,
   402→   402→   402→        (wordCount / this.config.commentReading.readingSpeedWpm) * 60000
   403→   403→   403→      );
   404→   404→   404→      const depthMultiplier = Math.pow(
   405→   405→   405→        this.config.commentReading.depthTimeMultiplier,
   406→   406→   406→        depth
   407→   407→   407→      );
   408→   408→   408→      const readTimeMs = Math.round(baseReadTimeMs * depthMultiplier);
   409→   409→   409→
   410→   410→   410→      // Expand collapsed comments occasionally
   411→   411→   411→      const isExpanded = depth > 1
   412→   412→   412→        ? Math.random() < this.config.commentReading.expandCollapsedProbability
   413→   413→   413→        : false;
   414→   414→   414→
   415→   415→   415→      actions.push({
   416→   416→   416→        depth,
   417→   417→   417→        readTimeMs,
   418→   418→   418→        isExpanded,
   419→   419→   419→        isSkipped: false,
   420→   420→   420→      });
   421→   421→   421→
   422→   422→   422→      // Simulate reading a reply chain
   423→   423→   423→      if (
   424→   424→   424→        depth === 0 &&
   425→   425→   425→        Math.random() < this.config.commentReading.replyChainReadProbability
   426→   426→   426→      ) {
   427→   427→   427→        // Add 1-3 replies to the chain
   428→   428→   428→        const chainLength = this.randomBetween(1, 3);
   429→   429→   429→        for (let j = 1; j <= chainLength && (i + j) < effectiveCommentCount; j++) {
   430→   430→   430→          const chainWords = this.randomBetween(10, AVG_COMMENT_WORDS + 20);
   431→   431→   431→          const chainReadTime = Math.max(
   432→   432→   432→            MIN_READING_TIME_MS,
   433→   433→   433→            (chainWords / this.config.commentReading.readingSpeedWpm) * 60000
   434→   434→   434→              * Math.pow(this.config.commentReading.depthTimeMultiplier, j)
   435→   435→   435→          );
   436→   436→   436→
   437→   437→   437→          actions.push({
   438→   438→   438→            depth: j,
   439→   439→   439→            readTimeMs: Math.round(chainReadTime),
   440→   440→   440→            isExpanded: j > 1 && Math.random() < this.config.commentReading.expandCollapsedProbability,
   441→   441→   441→            isSkipped: false,
   442→   442→   442→          });
   443→   443→   443→        }
   444→   444→   444→      }
   445→   445→   445→    }
   446→   446→   446→
   447→   447→   447→    this.stats.totalCommentReads += actions.filter(a => !a.isSkipped).length;
   448→   448→   448→
   449→   449→   449→    logger.debug({
   450→   450→   450→      commentCount,
   451→   451→   451→      readComments: actions.filter(a => !a.isSkipped).length,
   452→   452→   452→      skippedComments: actions.filter(a => a.isSkipped).length,
   453→   453→   453→      expandedComments: actions.filter(a => a.isExpanded).length,
   454→   454→   454→      totalReadTimeMs: actions.reduce((sum, a) => sum + a.readTimeMs, 0),
   455→   455→   455→    }, 'Generated comment reading pattern');
   456→   456→   456→
   457→   457→   457→    return actions;
   458→   458→   458→  }
