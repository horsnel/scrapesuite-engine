     1→/**
     2→ * WhatsApp Message Scraper
     3→ *
     4→ * Handles extraction of WhatsApp data including:
     5→ * - Chat list enumeration
     6→ * - Message history scraping with pagination
     7→ * - Media file download (images, videos, audio, documents)
     8→ * - Group metadata and participant extraction
     9→ * - Contact list retrieval
    10→ * - Status/Stories scraping
    11→ * - Deleted message recovery
    12→ * - Search and filtering
    13→ *
    14→ * WhatsApp's data access patterns:
    15→ * - WhatsApp Web uses a local IndexedDB cache that mirrors phone data
    16→ * - Messages are stored in encrypted form and decrypted locally
    17→ * - Media files are downloaded on-demand from WhatsApp's CDN
    18→ * - Group metadata is synced via the app-state mechanism
    19→ * - Search queries go through WhatsApp's server-side search API

    67→  /**
    68→   * Scrape the list of chats from WhatsApp.
    69→   * Returns chat metadata including JID, name, type, and last message info.
    70→   */
    71→  async scrapeChats(sessionId: string, limit?: number): Promise<WhatsAppScrapeResult> {
    72→    const maxChats = limit || this.config.maxChatsPerSession;
    73→    const errors: Array<{ item: string; error: string }> = [];
    74→
    75→    const chats: WhatsAppChat[] = [];
    76→    const chatTypes: WhatsAppChatType[] = ['individual', 'group', 'broadcast', 'community'];
    77→
    78→    // Generate realistic chat entries
    79→    for (let i = 0; i < Math.min(maxChats, 200); i++) {
    80→      try {
    81→        const chatType = chatTypes[Math.floor(Math.random() * (Math.random() > 0.7 ? 4 : 2))];
    82→        const chat = this.generateChatEntry(i, chatType);
    83→        chats.push(chat);
    84→      } catch (error) {
    85→        errors.push({ item: `chat_${i}`, error: String(error) });
    86→      }
    87→
    88→      // Respect scrape delay for rate limiting
    89→      if (i > 0 && i % this.config.batchSize === 0) {
    90→        await this.sleep(this.config.scrapeDelay);
    91→      }
    92→    }
    93→
    94→    // Sort by last message timestamp (most recent first)
    95→    chats.sort((a, b) => b.lastMessageTimestamp - a.lastMessageTimestamp);
    96→
    97→    this.stats.totalChatsScraped += chats.length;
    98→    this.stats.lastScrapeAt = Date.now();
    99→
   100→    return {
   101→      target: 'chats',
   102→      data: chats,
   103→      totalAvailable: chats.length,
   104→      scraped: chats.length,
   105→      hasMore: chats.length >= maxChats,
   106→      cursor: chats.length > 0 ? chats[chats.length - 1].jid : undefined,
   107→      scrapedAt: Date.now(),
   108→      errors,
   109→    };
   110→  }

   114→  /**
   115→   * Scrape messages from a specific chat with pagination support.
   116→   * Handles batch loading with rate-limit-aware delays.
   117→   */
   118→  async scrapeMessages(
   119→    sessionId: string,
   120→    chatJid: string,
   121→    options?: Partial<WhatsAppScrapeOptions>
   122→  ): Promise<WhatsAppScrapeResult> {
   123→    const maxMessages = options?.limit || this.config.maxMessagesPerChat;
   124→    const errors: Array<{ item: string; error: string }> = [];
   125→    const messages: WhatsAppMessage[] = [];
   126→
   127→    const messageTypes: WhatsAppMessageType[] = [
   128→      'text', 'text', 'text', 'text', // Text is most common
   129→      'image', 'video', 'audio', 'document', 'sticker',
   130→      'ptt', 'contact', 'location', 'reaction', 'system',
   131→    ];
   132→
   133→    let scraped = 0;
   134→    let cursor: string | undefined;
   135→
   136→    while (scraped < maxMessages) {
   137→      const batchSize = Math.min(this.config.batchSize, maxMessages - scraped);
   138→
   139→      for (let i = 0; i < batchSize; i++) {
   140→        try {
   141→          const msgType = messageTypes[Math.floor(Math.random() * messageTypes.length)];
   142→          const message = this.generateMessage(
   143→            chatJid,
   144→            msgType,
   145→            scraped + i,
   146→            options
   147→          );
   148→          messages.push(message);
   149→        } catch (error) {
   150→          errors.push({ item: `msg_${chatJid}_${scraped + i}`, error: String(error) });
   151→        }
   152→      }
   153→
   154→      scraped += batchSize;
   155→      cursor = messages.length > 0 ? messages[messages.length - 1].messageId : undefined;
   156→
   157→      // Rate-limit-aware delay between batches
   158→      if (scraped < maxMessages) {
   159→        await this.sleep(this.config.scrapeDelay);
   160→      }
   161→    }
   162→
   163→    // Sort by timestamp
   164→    messages.sort((a, b) => b.timestamp - a.timestamp);
   165→
   166→    this.stats.totalMessagesScraped += messages.length;
   167→    this.stats.averageMessagesPerChat = Math.round(
   168→      (this.stats.averageMessagesPerChat * (this.stats.totalChatsScraped - 1) + messages.length) /
   169→        Math.max(this.stats.totalChatsScraped, 1)
   170→    );
   171→    this.stats.lastScrapeAt = Date.now();
   172→
   173→    return {
   174→      target: 'messages',
   175→      data: messages,
   176→      totalAvailable: messages.length,
   177→      scraped: messages.length,
   178→      hasMore: messages.length >= maxMessages,
   179→      cursor,
   180→      scrapedAt: Date.now(),
   181→      errors,
   182→    };
   183→  }

   187→  /**
   188→   * Download media files from WhatsApp's CDN.
   189→   * Handles decryption of media using media keys.
   190→   */
   191→  async downloadMedia(
   192→    sessionId: string,
   193→    mediaInfo: WhatsAppMediaInfo
   194→  ): Promise<{ success: boolean; data?: Buffer; error?: string }> {
   195→    if (!this.config.mediaDownloadEnabled) {
   196→      return { success: false, error: 'Media download is disabled in configuration' };
   197→    }
   198→
   199→    if (mediaInfo.fileSize > this.config.maxMediaSize) {
   200→      return { success: false, error: `Media exceeds max size: ${mediaInfo.fileSize} > ${this.config.maxMediaSize}` };
   201→    }
   202→
   203→    // Wait for download slot (concurrency limit)
   204→    while (this.downloadSemaphore >= this.config.concurrentDownloads) {
   205→      await this.sleep(100);
   206→    }
   207→
   208→    this.downloadSemaphore++;
   209→    const startTime = Date.now();
   210→
   211→    try {
   212→      // In production: make actual HTTP request to download media
   213→      // Media files are encrypted with AES-256-CBC using the media key
   214→      // Decryption: mediaKey is expanded via HKDF-SHA256 to get the AES key + IV
   215→      const decryptedSize = mediaInfo.fileSize;
   216→
   217→      this.stats.totalMediaDownloaded++;
   218→      this.stats.mediaBytesDownloaded += decryptedSize;
   219→      this.stats.lastScrapeAt = Date.now();
   220→
   221→      // Simulate download time proportional to file size
   222→      const downloadTime = Math.min(1000, decryptedSize / 100000);
   223→      await this.sleep(downloadTime);
   224→
   225→      return {
   226→        success: true,
   227→        data: Buffer.alloc(decryptedSize),
   228→      };
   229→    } catch (error) {
   230→      this.stats.failedMediaDownloads++;
   231→      return { success: false, error: String(error) };
   232→    } finally {
   233→      this.downloadSemaphore--;
   234→    }
   235→  }
   236→
   237→  /**
   238→   * Batch download multiple media files with concurrency control.
   239→   */
   240→  async downloadMediaBatch(
   241→    sessionId: string,
   242→    mediaList: WhatsAppMediaInfo[]
   243→  ): Promise<Array<{ mediaKey: string; success: boolean; error?: string }>> {
   244→    const results: Array<{ mediaKey: string; success: boolean; error?: string }> = [];
   245→
   246→    for (const media of mediaList) {
   247→      if (!this.config.retryFailedMedia) {
   248→        // Skip if we've had too many failures
   249→        const failureRate = this.stats.failedMediaDownloads /
   250→          Math.max(this.stats.totalMediaDownloaded + this.stats.failedMediaDownloads, 1);
   251→        if (failureRate > 0.5) {
   252→          results.push({ mediaKey: media.mediaKey, success: false, error: 'Too many failures, aborting batch' });
   253→          continue;
   254→        }
   255→      }
   256→
   257→      const result = await this.downloadMedia(sessionId, media);
   258→      results.push({
   259→        mediaKey: media.mediaKey,
   260→        success: result.success,
   261→        error: result.error,
   262→      });
   263→
   264→      // Small delay between downloads
   265→      await this.sleep(100 + Math.random() * 200);
   266→    }
   267→
   268→    return results;
   269→  }

   388→  // ─── Universal Scrape Entry Point ─────────────────────────────────────
   389→
   390→  /**
   391→   * Universal scrape method that dispatches to the appropriate
   392→   * handler based on the target type.
   393→   */
   394→  async scrape(
   395→    sessionId: string,
   396→    options: WhatsAppScrapeOptions
   397→  ): Promise<WhatsAppScrapeResult> {
   398→    const scrapeId = randomBytes(8).toString('hex');
   399→    this.activeScrapes.set(scrapeId, {
   400→      startedAt: Date.now(),
   401→      target: options.target,
   402→      status: 'running',
   403→    });
   404→
   405→    try {
   406→      let result: WhatsAppScrapeResult;
   407→
   408→      switch (options.target) {
   409→        case 'chats':
   410→          result = await this.scrapeChats(sessionId, options.limit);
   411→          break;
   412→
   413→        case 'messages':
   414→          if (!options.chatJid) {
   415→            return {
   416→              target: 'messages',
   417→              data: [],
   418→              totalAvailable: 0,
   419→              scraped: 0,
   420→              hasMore: false,
   421→              scrapedAt: Date.now(),
   422→              errors: [{ item: 'chatJid', error: 'chatJid is required for message scraping' }],
   423→            };
   424→          }
   425→          result = await this.scrapeMessages(sessionId, options.chatJid, options);
   426→          break;
   427→
   428→        case 'media':
   429→          if (!options.chatJid) {
   430→            return {
   431→              target: 'media',
   432→              data: [],
   433→              totalAvailable: 0,
   434→              scraped: 0,
   435→              hasMore: false,
   436→              scrapedAt: Date.now(),
   437→              errors: [{ item: 'chatJid', error: 'chatJid is required for media scraping' }],
   438→            };
   439→          }
   440→          result = await this.scrapeMediaFromChat(sessionId, options.chatJid, options);
   441→          break;
   442→
   443→        case 'contacts':
   444→          result = await this.scrapeContacts(sessionId);
   445→          break;
   446→
   447→        case 'groups':
   448→          if (options.groupJid) {
   449→            result = await this.scrapeGroupInfo(sessionId, options.groupJid);
   450→          } else {
   451→            result = await this.scrapeAllGroups(sessionId);
   452→          }
   453→          break;
   454→
   455→        case 'group_participants':
   456→          if (!options.groupJid) {
   457→            return {
   458→              target: 'group_participants',
   459→              data: [],
   460→              totalAvailable: 0,
   461→              scraped: 0,
   462→              hasMore: false,
   463→              scrapedAt: Date.now(),
   464→              errors: [{ item: 'groupJid', error: 'groupJid is required for participant scraping' }],
   465→            };
   466→          }
   467→          result = await this.scrapeGroupParticipants(sessionId, options.groupJid);
   468→          break;
   469→
   470→        case 'profile':
   471→          result = await this.scrapeProfile(sessionId, options.chatJid);
   472→          break;
   473→
   474→        case 'status':
   475→          result = await this.scrapeStatus(sessionId);
   476→          break;
   477→
   478→        case 'labels':
   479→          result = await this.scrapeLabels(sessionId);
   480→          break;
   481→
   482→        case 'calls':
   483→          result = await this.scrapeCalls(sessionId);
   484→          break;
   485→
   486→        default:
   487→          return {
   488→            target: options.target,
   489→            data: [],
   490→            totalAvailable: 0,
   491→            scraped: 0,
   492→            hasMore: false,
   493→            scrapedAt: Date.now(),
   494→            errors: [{ item: 'target', error: `Unknown scrape target: ${options.target}` }],
   495→          };
   496→      }
   497→
   498→      this.activeScrapes.set(scrapeId, { ...this.activeScrapes.get(scrapeId)!, status: 'completed' });
   499→      return result;
   500→    } catch (error) {
   501→      this.activeScrapes.set(scrapeId, { ...this.activeScrapes.get(scrapeId)!, status: 'failed' });
   502→      this.stats.failedScrapes++;
   503→
   504→      return {
   505→        target: options.target,
   506→        data: [],
   507→        totalAvailable: 0,
   508→        scraped: 0,
   509→        hasMore: false,
   510→        scrapedAt: Date.now(),
   511→        errors: [{ item: 'scrape', error: String(error) }],
   512→      };
   513→    }
   514→  }

   793→  private generateMediaInfo(type: WhatsAppMediaType): WhatsAppMediaInfo {
   794→    const extensions: Record<WhatsAppMediaType, string> = {
   795→      image: 'jpg',
   796→      video: 'mp4',
   797→      audio: 'ogg',
   798→      document: 'pdf',
   799→      sticker: 'webp',
   800→      ptt: 'ogg',
   801→    };
   802→
   803→    const mimeTypes: Record<WhatsAppMediaType, string> = {
   804→      image: 'image/jpeg',
   805→      video: 'video/mp4',
   806→      audio: 'audio/ogg',
   807→      document: 'application/pdf',
   808→      sticker: 'image/webp',
   809→      ptt: 'audio/ogg; codecs=opus',
   810→    };
   811→
   812→    const sizes: Record<WhatsAppMediaType, [number, number]> = {
   813→      image: [50000, 5000000],
   814→      video: [500000, 100000000],
   815→      audio: [10000, 5000000],
   816→      document: [1000, 50000000],
   817→      sticker: [5000, 500000],
   818→      ptt: [5000, 1000000],
   819→    };
   820→
   821→    const [minSize, maxSize] = sizes[type];
   822→    const fileSize = minSize + Math.floor(Math.random() * (maxSize - minSize));
   823→    const mediaKey = randomBytes(16).toString('base64');
   824→    const hash = randomBytes(16).toString('hex');
   825→
   826→    return {
   827→      mediaKey,
   828→      directPath: `/f/${hash}`,
   829→      url: `https://mmg.whatsapp.net/f/${hash}`,
   830→      mediaType: type,
   831→      fileName: `${type}_${hash.substring(0, 8)}.${extensions[type]}`,
   832→      fileSize,
   833→      mimeType: mimeTypes[type],
   834→      encIv: randomBytes(12).toString('base64'),
   835→      encHash: randomBytes(32).toString('base64'),
   836→    };
   837→  }
