   308→// --- PII Detection & Redaction Engine -----------------------------------------
   309→
   310→export class PIIDetectionEngine {
   311→  private patterns: PIIPattern[] = [];
   312→  private redactionRules: Map<PIICategory, RedactionRule> = new Map();
   313→  private defaultStrategy: RedactionStrategy;
   314→  private enabledCategories: Set<PIICategory>;
   315→
   316→  constructor(config: RedactionConfig) {
   317→    this.defaultStrategy = config.defaultStrategy || RedactionStrategy.REDACTED;
   318→    this.enabledCategories = new Set(config.enabledCategories || Object.values(PIICategory));
   319→
   320→    // Load built-in patterns
   321→    this.patterns = [...DEFAULT_PII_PATTERNS];
   322→
   323→    // Load custom patterns
   324→    if (config.customPatterns?.length) {
   325→      this.patterns.push(...config.customPatterns);
   326→    }
   327→
   328→    // Build redaction rule map
   329→    const rules = config.rules?.length ? config.rules : DEFAULT_REDACTION_RULES;
   330→    for (const rule of rules) {
   331→      this.redactionRules.set(rule.category, rule);
   332→    }
   333→  }
   334→
   335→  /**
   336→   * Detect all PII in a given text string.
   337→   */
   338→  detectPII(text: string): PIIDetectionResult[] {
   339→    const results: PIIDetectionResult[] = [];
   340→
   341→    for (const pattern of this.patterns) {
   342→      if (!this.enabledCategories.has(pattern.category)) continue;
   343→
   344→      // Reset regex state for patterns with global flag
   345→      const regex = new RegExp(pattern.pattern.source, pattern.pattern.flags);
   346→      let match: RegExpExecArray | null;
   347→
   348→      while ((match = regex.exec(text)) !== null) {
   349→        const matchedText = match[0];
   350→        const rule = this.redactionRules.get(pattern.category);
   351→        const strategy = rule?.enabled ? rule.strategy : this.defaultStrategy;
   352→
   353→        results.push({
   354→          category: pattern.category,
   355→          match: matchedText,
   356→          startIndex: match.index,
   357→          endIndex: match.index + matchedText.length,
   358→          confidence: pattern.confidence,
   359→          redacted: this.applyRedaction(matchedText, pattern.category, rule),
   360→          strategy,
   361→        });
   362→      }
   363→    }
   364→
   365→    // Sort by start index for positional processing
   366→    results.sort((a, b) => a.startIndex - b.startIndex);
   367→    return results;
   368→  }
   369→
   370→  /**
   371→   * Redact all PII in a given text string, returning the redacted text.
   372→   */
   373→  redactText(text: string): { redactedText: string; detections: PIIDetectionResult[] } {
   374→    const detections = this.detectPII(text);
   375→
   376→    if (detections.length === 0) {
   377→      return { redactedText: text, detections };
   378→    }
   379→
   380→    // Process from end to start to preserve indices
   381→    let redactedText = text;
   382→    for (let i = detections.length - 1; i >= 0; i--) {
   383→      const det = detections[i];
   384→      redactedText =
   385→        redactedText.slice(0, det.startIndex) + det.redacted + redactedText.slice(det.endIndex);
   386→    }
   387→
   388→    return { redactedText, detections };
   389→  }
   390→
   391→  /**
   392→   * Redact PII in a structured data object (deep scan).
   393→   */
   394→  redactObject<T extends Record<string, any>>(obj: T): { data: T; detections: PIIDetectionResult[] } {
   395→    const allDetections: PIIDetectionResult[] = [];
   396→    const result = this.deepRedact(obj, allDetections) as T;
   397→    return { data: result, detections: allDetections };
   398→  }
   399→
   400→  /**
   401→   * Add a custom PII pattern at runtime.
   402→   */
   403→  addCustomPattern(pattern: PIIPattern): void {
   404→    this.patterns.push({ ...pattern, category: PIICategory.CUSTOM });
   405→    logger.info({ name: pattern.name }, 'Custom PII pattern added');
   406→  }
   407→
   408→  /**
   409→   * Remove a PII category from detection.
   410→   */
   411→  disableCategory(category: PIICategory): void {
   412→    this.enabledCategories.delete(category);
   413→    logger.info({ category }, 'PII category disabled');
   414→  }
   415→
   416→  /**
   417→   * Enable a PII category for detection.
   418→   */
   419→  enableCategory(category: PIICategory): void {
   420→    this.enabledCategories.add(category);
   421→    logger.info({ category }, 'PII category enabled');
   422→  }
   423→
   424→  /**
   425→   * Update redaction rule for a specific PII category.
   426→   */
   427→  updateRedactionRule(category: PIICategory, rule: Partial<RedactionRule>): void {
   428→    const existing = this.redactionRules.get(category);
   429→    if (existing) {
   430→      this.redactionRules.set(category, { ...existing, ...rule });
   431→    } else {
   432→      this.redactionRules.set(category, {
   433→        category,
   434→        strategy: rule.strategy || this.defaultStrategy,
   435→        enabled: rule.enabled ?? true,
   436→        ...rule,
   437→      });
   438→    }
   439→    logger.info({ category, strategy: rule.strategy }, 'Redaction rule updated');
   440→  }
   441→
   442→  /**
   443→   * Get all active patterns.
   444→   */
   445→  getPatterns(): PIIPattern[] {
   446→    return [...this.patterns];
   447→  }
   448→
   449→  /**
   450→   * Get all redaction rules.
   451→   */
   452→  getRedactionRules(): RedactionRule[] {
   453→    return Array.from(this.redactionRules.values());
   454→  }
   455→
   456→  // --- Private Helpers ------------------------------------------------------
   457→
   458→  private applyRedaction(text: string, category: PIICategory, rule?: RedactionRule): string {
   459→    const strategy = rule?.enabled ? rule.strategy : this.defaultStrategy;
   460→
   461→    switch (strategy) {
   462→      case RedactionStrategy.MASK: {
   463→        const maskChar = rule?.maskChar || '*';
   464→        const keepPrefix = rule?.maskKeepPrefix ?? 0;
   465→        const keepSuffix = rule?.maskKeepSuffix ?? 0;
   466→        const maskableLength = Math.max(0, text.length - keepPrefix - keepSuffix);
   467→        const masked =
   468→          text.slice(0, keepPrefix) +
   469→          maskChar.repeat(maskableLength) +
   470→          text.slice(text.length - keepSuffix);
   471→        return masked;
   472→      }
   473→
   474→      case RedactionStrategy.HASH: {
   475→        return 'SHA256:' + crypto.createHash('sha256').update(text).digest('hex').slice(0, 16);
   476→      }
   477→
   478→      case RedactionStrategy.REMOVE: {
   479→        return '';
   480→      }
   481→
   482→      case RedactionStrategy.REDACTED: {
   483→        const label = rule?.replacementText || `[${category.toUpperCase()} REDACTED]`;
   484→        return label;
   485→      }
   486→
   487→      default:
   488→        return '[REDACTED]';
   489→    }
   490→  }
   491→
   492→  private deepRedact(obj: any, detections: PIIDetectionResult[]): any {
   493→    if (obj === null || obj === undefined) return obj;
   494→
   495→    if (typeof obj === 'string') {
   496→      const { redactedText, detections: dets } = this.redactText(obj);
   497→      detections.push(...dets);
   498→      return redactedText;
   499→    }
   500→
   501→    if (Array.isArray(obj)) {
   502→      return obj.map((item) => this.deepRedact(item, detections));
   503→    }
   504→
   505→    if (typeof obj === 'object') {
   506→      const result: Record<string, any> = {};
   507→      for (const [key, value] of Object.entries(obj)) {
   508→        result[key] = this.deepRedact(value, detections);
   509→      }
   510→      return result;
   511→    }
   512→
   513→    return obj;
   514→  }
   515→}

   548→  /**
   549→   * Process a data subject access request (GDPR Article 15).
   550→   * Returns all data associated with the given subject identifier.
   551→   */
   552→  async processDataAccessRequest(
   553→    subjectIdentifier: string,
   554→    requestorEmail: string,
   555→  ): Promise<DataSubjectRequest> {
   556→    const requestId = `dsr-${crypto.randomUUID()}`;
   557→    const deadline = new Date();
   558→    deadline.setDate(deadline.getDate() + 30); // GDPR: 30 days
   559→
   560→    const request: DataSubjectRequest = {
   561→      id: requestId,
   562→      type: DataSubjectRequestType.ACCESS,
   563→      subjectIdentifier,
   564→      requestorEmail,
   565→      requestDate: new Date(),
   566→      deadline,
   567→      status: 'in_progress',
   568→      notes: [`Access request received from ${requestorEmail}`],
   569→      dataCollected: [],
   570→    };
   571→
   572→    try {
   573→      // Search for all data records associated with this subject
   574→      const records = await this.findDataBySubject(subjectIdentifier);
   575→      request.dataCollected = records;
   576→
   577→      // Also check Redis cache for any cached PII
   578→      const cachedData = await this.findCachedDataBySubject(subjectIdentifier);
   579→      if (cachedData) {
   580→        request.notes.push(`Found cached data in ${cachedData} cache entries`);
   581→      }
   582→
   583→      request.status = 'completed';
   584→      request.completedDate = new Date();
   585→      request.notes.push(`Found ${records.length} data records`);
   586→
   587→      logger.info({
   588→        requestId,
   589→        subjectIdentifier,
   590→        recordsFound: records.length,
   591→      }, 'GDPR data access request completed');
   592→    } catch (error: any) {
   593→      request.status = 'pending';
   594→      request.notes.push(`Error processing request: ${error.message}`);
   595→      logger.error({ requestId, error: error.message }, 'Failed to process data access request');
   596→    }
   597→
   598→    this.dataSubjectRequests.set(requestId, request);
   599→
   600→    // Persist to database
   601→    await this.persistDataSubjectRequest(request);
   602→
   603→    return request;
   604→  }

   609→   */
   610→  async processErasureRequest(
   611→    subjectIdentifier: string,
   612→    requestorEmail: string,
   613→  ): Promise<DataSubjectRequest> {
   614→    const requestId = `dsr-${crypto.randomUUID()}`;
   615→    const deadline = new Date();
   616→    deadline.setDate(deadline.getDate() + 30);
   617→
   618→    const request: DataSubjectRequest = {
   619→      id: requestId,
   620→      type: DataSubjectRequestType.ERASURE,
   621→      subjectIdentifier,
   622→      requestorEmail,
   623→      requestDate: new Date(),
   624→      deadline,
   625→      status: 'in_progress',
   626→      notes: [`Erasure request received from ${requestorEmail}`],
   627→      dataCollected: [],
   628→    };
   629→
   630→    try {
   631→      const records = await this.findDataBySubject(subjectIdentifier);
   632→      request.dataCollected = records;
   633→
   634→      let deletedCount = 0;
   635→      let heldCount = 0;
   636→
   637→      for (const record of records) {
   638→        // Check for legal holds
   639→        if (record.legalHoldIds && record.legalHoldIds.length > 0) {
   640→          heldCount++;
   641→          request.notes.push(
   642→            `Record ${record.id} is under legal hold (${record.legalHoldIds.join(', ')}). Skipped deletion.`,
   643→          );
   644→          continue;
   645→        }
   646→
   647→        await this.deleteDataRecord(record);
   648→        deletedCount++;
   649→      }
   650→
   651→      // Clear Redis cache entries
   652→      await this.clearCachedDataBySubject(subjectIdentifier);
   653→
   654→      request.status = 'completed';
   655→      request.completedDate = new Date();
   656→      request.notes.push(`Deleted ${deletedCount} records, ${heldCount} under legal hold`);
   657→
   658→      logger.info({
   659→        requestId,
   660→        subjectIdentifier,
   661→        deletedCount,
   662→        heldCount,
   663→      }, 'GDPR erasure request completed');
   664→    } catch (error: any) {
   665→      request.status = 'pending';
   666→      request.notes.push(`Error processing request: ${error.message}`);
   667→      logger.error({ requestId, error: error.message }, 'Failed to process erasure request');
   668→    }
   669→
   670→    this.dataSubjectRequests.set(requestId, request);
   671→    await this.persistDataSubjectRequest(request);
   672→
   673→    return request;
   674→  }

   717→   */
   718→  detectConsentBanner(html: string, url: string): ConsentBannerResult {
   719→    const detected = this.consentBannerPatterns.some((pattern) => pattern.test(html));
   720→
   721→    if (!detected) {
   722→      return {
   723→        detected: false,
   724→        bannerType: null,
   725→        consentOptions: [],
   726→        recommendedAction: 'No consent banner detected; proceed with caution',
   727→        url,
   728→      };
   729→    }
   730→
   731→    // Determine banner type
   732→    let bannerType = 'unknown';
   733→    if (/gdpr/i.test(html)) bannerType = 'gdpr';
   734→    else if (/ccpa/i.test(html)) bannerType = 'ccpa';
   735→    else if (/cookie/i.test(html)) bannerType = 'cookie';
   736→
   737→    // Extract consent options
   738→    const consentOptions: string[] = [];
   739→    if (/accept\s+all/i.test(html)) consentOptions.push('accept_all');
   740→    if (/reject\s+all/i.test(html)) consentOptions.push('reject_all');
   741→    if (/manage\s+(?:preferences|consent)/i.test(html)) consentOptions.push('manage_preferences');
   742→    if (/necessary\s+only/i.test(html)) consentOptions.push('necessary_only');
   743→
   744→    return {
   745→      detected: true,
   746→      bannerType,
   747→      consentOptions,
   748→      recommendedAction: 'Respect consent banner; do not auto-accept without user direction',
   749→      url,
   750→    };

   919→  private startRetentionEnforcement(): void {

   921→      await this.enforceRetentionPolicies();
   922→    }, 3600_000); // Check every hour
   923→

   930→  private async enforceRetentionPolicies(): Promise<void> {
   931→    const now = new Date();
   932→    let expiredCount = 0;
   933→
   934→    try {
   935→      // Find records past retention period
   936→      for (const policy of this.retentionPolicies.values()) {
   937→        if (!policy.autoDelete) continue;
   938→
   939→        const cutoffDate = new Date(now.getTime() - policy.retentionDays * 86400_000);
   940→
   941→        for (const pattern of policy.appliesTo) {
   942→          try {
   943→            const expired = await db.scrapedData.findMany({
   944→              where: {
   945→                createdAt: { lt: cutoffDate },
   946→                deletedAt: null,
   947→                url: { contains: pattern },
   948→              },
   949→              take: 100,
   950→            });
   951→
   952→            for (const record of expired) {
   953→              // Check legal hold
   954→              if ((record as any).legalHoldIds?.length > 0) {
   955→                continue;
   956→              }
   957→
   958→              await db.scrapedData.update({
   959→                where: { id: record.id },
   960→                data: { deletedAt: now },
   961→              });
   962→
   963→              expiredCount++;
   964→            }
   965→          } catch (error: any) {
   966→            logger.warn({ pattern, error: error.message }, 'Failed to enforce retention for pattern');
   967→          }
   968→        }
   969→      }
   970→
   971→      // Also enforce default retention for records without a specific policy
   972→      const defaultCutoff = new Date(now.getTime() - this.defaultRetentionDays * 86400_000);
   973→      try {
   974→        const defaultExpired = await db.scrapedData.findMany({
   975→          where: {
   976→            createdAt: { lt: defaultCutoff },
   977→            deletedAt: null,
   978→          },
   979→          take: 100,
   980→        });
   981→
   982→        for (const record of defaultExpired) {
   983→          if ((record as any).legalHoldIds?.length > 0) continue;
   984→          await db.scrapedData.update({
   985→            where: { id: record.id },
   986→            data: { deletedAt: now },
   987→          });
   988→          expiredCount++;
   989→        }
   990→      } catch (error: any) {
   991→        logger.warn({ error: error.message }, 'Failed to enforce default retention');
   992→      }
   993→
   994→      if (expiredCount > 0) {
   995→        logger.info({ expiredCount }, 'Retention policies enforced');
   996→      }
   997→    } catch (error: any) {
   998→      logger.error({ error: error.message }, 'Error during retention enforcement');
   999→    }
  1000→  }
  1001→
  1002→  private urlMatchesPattern(url: string, pattern: string): boolean {
  1003→    if (pattern.startsWith('*.')) {
  1004→      // Domain wildcard
  1005→      const domain = pattern.slice(2);
  1006→      return url.includes(domain);
  1007→    }
  1008→    if (pattern.startsWith('/')) {
  1009→      // Path pattern
  1010→      try {
  1011→        const urlPath = new URL(url).pathname;
  1012→        return urlPath.startsWith(pattern);
  1013→      } catch {
  1014→        return false;
  1015→      }
  1016→    }
  1017→    return url.includes(pattern);
  1018→  }

  1020→  private recordsToCSV(records: ScrapeDataRecord[]): string {
  1021→    if (records.length === 0) return '';
  1022→
  1023→    const headers = ['id', 'url', 'domain', 'scrapedAt', 'scrapedBy', 'piiRedacted', 'doNotSell'];
  1024→    const rows = records.map((r) =>
  1025→      headers.map((h) => {
  1026→        const val = (r as any)[h];
  1027→        return typeof val === 'string' ? `"${val.replace(/"/g, '""')}"` : val;
  1028→      }).join(','),
  1029→    );
  1030→
  1031→    return [headers.join(','), ...rows].join('\n');
  1032→  }

  1286→// --- Audit Logger (Tamper-Proof with Hash Chaining) ---------------------------

  1288→export class AuditLogger {
  1289→  private lastHash: string = '0'.repeat(64); // Genesis hash
  1290→  private logBuffer: AuditLogEntry[] = [];
  1291→  private flushInterval: NodeJS.Timer | null = null;
  1292→  private readonly BUFFER_SIZE = 100;
  1293→  private readonly FLUSH_INTERVAL_MS = 10_000;
  1294→
  1295→  constructor() {
  1296→    this.loadLastHash();
  1297→    this.startFlushCycle();
  1298→  }
  1299→
  1300→  /**
  1301→   * Log an audit event with hash-chaining for tamper proofing.
  1302→   */
  1303→  async log(
  1304→    eventType: AuditEventType,
  1305→    userId: string,
  1306→    details: Record<string, any>,
  1307→    options?: {
  1308→      url?: string;
  1309→      domain?: string;
  1310→      piiFound?: boolean;
  1311→      piiRedacted?: boolean;
  1312→    },
  1313→  ): Promise<AuditLogEntry> {
  1314→    const entry: AuditLogEntry = {
  1315→      id: `audit-${crypto.randomUUID()}`,
  1316→      timestamp: new Date(),
  1317→      eventType,
  1318→      userId,
  1319→      url: options?.url,
  1320→      domain: options?.domain,
  1321→      details,
  1322→      piiFound: options?.piiFound ?? false,
  1323→      piiRedacted: options?.piiRedacted ?? false,
  1324→      previousHash: this.lastHash,
  1325→      currentHash: '', // Will be computed
  1326→    };
  1327→
  1328→    // Compute hash: SHA-256 of entry data + previous hash
  1329→    const hashPayload = JSON.stringify({
  1330→      id: entry.id,
  1331→      timestamp: entry.timestamp.toISOString(),
  1332→      eventType,
  1333→      userId,
  1334→      url: entry.url,
  1335→      domain: entry.domain,
  1336→      details,
  1337→      piiFound: entry.piiFound,
  1338→      piiRedacted: entry.piiRedacted,
  1339→      previousHash: entry.previousHash,
  1340→    });
  1341→
  1342→    entry.currentHash = crypto.createHash('sha256').update(hashPayload).digest('hex');
  1343→    this.lastHash = entry.currentHash;
  1344→
  1345→    // Buffer for batch persist
  1346→    this.logBuffer.push(entry);
  1347→
  1348→    // Flush if buffer is full
  1349→    if (this.logBuffer.length >= this.BUFFER_SIZE) {
  1350→      await this.flush();
  1351→    }
  1352→
  1353→    return entry;
  1354→  }
  1355→
  1356→  /**
  1357→   * Query audit logs with filters.
  1358→   */
  1359→  async query(filters: AuditLogQuery): Promise<AuditLogEntry[]> {
  1360→    const cacheKey = `compliance:audit:query:${crypto
  1361→      .createHash('md5')
  1362→      .update(JSON.stringify(filters))
  1363→      .digest('hex')}`;
  1364→
  1365→    // Try cache first
  1366→    const cached = await cacheGet<AuditLogEntry[]>(cacheKey);
  1367→    if (cached) return cached;
  1368→
  1369→    try {
  1370→      // Build where clause for database query
  1371→      const where: Record<string, any> = {};
  1372→
  1373→      if (filters.startDate || filters.endDate) {
  1374→        where.timestamp = {};
  1375→        if (filters.startDate) where.timestamp.gte = filters.startDate;
  1376→        if (filters.endDate) where.timestamp.lte = filters.endDate;
  1377→      }
  1378→      if (filters.userId) where.userId = filters.userId;
  1379→      if (filters.domain) where.domain = filters.domain;
  1380→      if (filters.eventType) where.eventType = filters.eventType;
  1381→      if (filters.piiFound !== undefined) where.piiFound = filters.piiFound;
  1382→
  1383→      const results = await db.auditLog.findMany({
  1384→        where,
  1385→        orderBy: { createdAt: 'desc' },
  1386→        take: filters.limit || 100,
  1387→        skip: filters.offset || 0,
  1388→      });
  1389→
  1390→      const entries = results.map((r: any) => ({
  1391→        id: r.id,
  1392→        timestamp: r.createdAt,
  1393→        eventType: r.eventType as AuditEventType,
  1394→        userId: r.userId,
  1395→        url: r.url,
  1396→        domain: r.domain,
  1397→        details: r.details as Record<string, any>,
  1398→        piiFound: r.piiFound ?? false,
  1399→        piiRedacted: r.piiRedacted ?? false,
  1400→        previousHash: r.previousHash || '',
  1401→        currentHash: r.currentHash || '',
  1402→      }));
  1403→
  1404→      // Cache results briefly
  1405→      await cacheSet(cacheKey, entries, 300);
  1406→
  1407→      return entries;
  1408→    } catch (error: any) {
  1409→      logger.warn({ error: error.message }, 'Failed to query audit logs from DB; searching buffer');
  1410→      return this.searchBuffer(filters);
  1411→    }
  1412→  }
  1413→
  1414→  /**
  1415→   * Verify integrity of the audit log chain.
  1416→   * Returns true if all hashes link correctly.
  1417→   */
  1418→  async verifyIntegrity(): Promise<{ valid: boolean; brokenAt: string | null; totalChecked: number }> {
  1419→    try {
  1420→      const entries = await db.auditLog.findMany({
  1421→        orderBy: { createdAt: 'asc' },
  1422→        take: 10000,
  1423→      });
  1424→
  1425→      let previousHash = '0'.repeat(64);
  1426→      let totalChecked = 0;
  1427→
  1428→      for (const entry of entries) {
  1429→        totalChecked++;
  1430→
  1431→        if (entry.previousHash !== previousHash) {
  1432→          logger.error({ entryId: entry.id }, 'Audit log chain integrity broken');
  1433→          return { valid: false, brokenAt: entry.id, totalChecked };
  1434→        }
  1435→
  1436→        // Recompute hash
  1437→        const hashPayload = JSON.stringify({
  1438→          id: entry.id,
  1439→          timestamp: entry.createdAt.toISOString(),
  1440→          eventType: entry.eventType,
  1441→          userId: entry.userId,
  1442→          url: entry.url,
  1443→          domain: entry.domain,
  1444→          details: entry.details,
  1445→          piiFound: entry.piiFound,
  1446→          piiRedacted: entry.piiRedacted,
  1447→          previousHash: entry.previousHash,
  1448→        });
  1449→
  1450→        const computedHash = crypto.createHash('sha256').update(hashPayload).digest('hex');
  1451→        if (computedHash !== entry.currentHash) {
  1452→          logger.error({ entryId: entry.id }, 'Audit log entry hash mismatch');
  1453→          return { valid: false, brokenAt: entry.id, totalChecked };
  1454→        }
  1455→
  1456→        previousHash = entry.currentHash;
  1457→      }
  1458→
  1459→      logger.info({ totalChecked }, 'Audit log integrity verified');
  1460→      return { valid: true, brokenAt: null, totalChecked };
  1461→    } catch (error: any) {
  1462→      logger.error({ error: error.message }, 'Failed to verify audit log integrity');
  1463→      return { valid: false, brokenAt: null, totalChecked: 0 };
  1464→    }
  1465→  }
  1466→
  1467→  /**
  1468→   * Flush buffered audit log entries to persistent storage.
  1469→   */
  1470→  async flush(): Promise<void> {
  1471→    if (this.logBuffer.length === 0) return;
  1472→
  1473→    const entries = [...this.logBuffer];
  1474→    this.logBuffer = [];
  1475→
  1476→    try {
  1477→      // Batch insert into database
  1478→      await db.auditLog.createMany({
  1479→        data: entries.map((e) => ({
  1480→          id: e.id,
  1481→          action: e.eventType,
  1482→          resource: 'compliance',
  1483→          category: 'compliance',
  1484→          eventType: e.eventType,
  1485→          url: e.url,
  1486→          domain: e.domain,
  1487→          details: e.details,
  1488→          piiFound: e.piiFound,
  1489→          piiRedacted: e.piiRedacted,
  1490→          previousHash: e.previousHash,
  1491→          currentHash: e.currentHash,
  1492→          createdAt: e.timestamp,
  1493→        })),
  1494→      });
  1495→
  1496→      // Update last hash in cache
  1497→      if (entries.length > 0) {
  1498→        await cacheSet('compliance:audit:lastHash', this.lastHash, 86400);
  1499→      }
  1500→
  1501→      logger.debug({ count: entries.length }, 'Audit log entries flushed');
  1502→    } catch (error: any) {
  1503→      // Re-add to buffer for retry
  1504→      this.logBuffer.unshift(...entries);
  1505→      logger.error({ error: error.message, count: entries.length }, 'Failed to flush audit log entries');
  1506→    }
  1507→  }
  1508→
  1509→  /**
  1510→   * Shutdown: flush remaining entries.
  1511→   */
  1512→  async shutdown(): Promise<void> {
  1513→    if (this.flushInterval) {
  1514→      clearInterval(this.flushInterval as any);
  1515→    }
  1516→    await this.flush();
  1517→    logger.info('Audit logger shut down');
  1518→  }
  1519→
  1520→  // --- Private Helpers ------------------------------------------------------
  1521→
  1522→  private async loadLastHash(): Promise<void> {
  1523→    try {
  1524→      const cached = await cacheGet<string>('compliance:audit:lastHash');
  1525→      if (cached) {
  1526→        this.lastHash = cached;
  1527→      }
  1528→    } catch (error: any) {
  1529→      logger.warn({ error: error.message }, 'Failed to load last audit hash; using genesis');
  1530→    }
  1531→  }
  1532→
  1533→  private startFlushCycle(): void {
  1534→    this.flushInterval = setInterval(async () => {
  1535→      await this.flush();
  1536→    }, this.FLUSH_INTERVAL_MS);
  1537→
  1538→    if ((this.flushInterval as any).unref) {
  1539→      (this.flushInterval as any).unref();
  1540→    }
  1541→  }
  1542→
  1543→  private searchBuffer(filters: AuditLogQuery): AuditLogEntry[] {
  1544→    let results = [...this.logBuffer];
  1545→
  1546→    if (filters.startDate) {
  1547→      results = results.filter((e) => e.timestamp >= filters.startDate!);
  1548→    }
  1549→    if (filters.endDate) {
  1550→      results = results.filter((e) => e.timestamp <= filters.endDate!);
  1551→    }
  1552→    if (filters.userId) {
  1553→      results = results.filter((e) => e.userId === filters.userId);
  1554→    }
  1555→    if (filters.domain) {
  1556→      results = results.filter((e) => e.domain === filters.domain);
  1557→    }
  1558→    if (filters.eventType) {
  1559→      results = results.filter((e) => e.eventType === filters.eventType);
  1560→    }
  1561→    if (filters.piiFound !== undefined) {
  1562→      results = results.filter((e) => e.piiFound === filters.piiFound);
  1563→    }
  1564→
  1565→    results.sort((a, b) => b.timestamp.getTime() - a.timestamp.getTime());
  1566→
  1567→    return results.slice(filters.offset || 0, (filters.offset || 0) + (filters.limit || 100));
  1568→  }
  1569→}

  1573→export class RobotsComplianceEnhancer {
  1574→  private crawlDelays: Map<string, { delayMs: number; lastAccess: number }> = new Map();
  1575→  private overrideLog: Map<string, { url: string; timestamp: Date; userId: string }[]> = new Map();
  1576→  private respectRobots: boolean;
  1577→  private logOverrides: boolean;
  1578→
  1579→  constructor(respectRobots: boolean = true, logOverrides: boolean = true) {
  1580→    this.respectRobots = respectRobots;
  1581→    this.logOverrides = logOverrides;
  1582→  }
  1583→
  1584→  /**
  1585→   * Check if a URL is compliant with robots.txt, including crawl-delay enforcement.
  1586→   */
  1587→  async checkCompliance(
  1588→    url: string,
  1589→    userAgent: string = 'ScrapeSuite',
  1590→    overrideRobots: boolean = false,
  1591→  ): Promise<RobotsComplianceResult> {
  1592→    const domain = new URL(url).hostname;
  1593→
  1594→    // Check robots.txt via the existing parser
  1595→    const result = await robotsParser.isAllowed(url, userAgent, true);
  1596→
  1597→    // Get sitemaps
  1598→    const sitemaps = await robotsParser.getSitemaps(domain);
  1599→
  1600→    // Get disallowed paths for this domain
  1601→    const disallowedPaths = await this.getDisallowedPaths(domain, userAgent);
  1602→
  1603→    // Check crawl-delay compliance
  1604→    if (result.crawlDelay) {
  1605→      const delayCompliant = this.checkCrawlDelay(domain, result.crawlDelay);
  1606→      if (!delayCompliant.compliant) {
  1607→        logger.warn({
  1608→          domain,
  1609→          requiredDelay: result.crawlDelay,
  1610→          remainingMs: delayCompliant.remainingMs,
  1611→        }, 'Crawl-delay not yet satisfied');
  1612→      }
  1613→    }
  1614→
  1615→    // Handle override
  1616→    let overrideLogged = false;
  1617→    if (!result.allowed && overrideRobots) {
  1618→      overrideLogged = true;
  1619→      this.logOverride(url, 'system'); // Will be updated with actual user ID
  1620→      logger.warn({ url, domain }, 'Robots.txt override applied');
  1621→    }
  1622→
  1623→    return {
  1624→      url,
  1625→      allowed: overrideRobots || result.allowed,
  1626→      crawlDelay: result.crawlDelay,
  1627→      sitemaps,
  1628→      overrideLogged,
  1629→      disallowedPaths,
  1630→    };
  1631→  }
  1632→
  1633→  /**
  1634→   * Check crawl-delay and return remaining wait time.
  1635→   */
  1636→  checkCrawlDelay(domain: string, requiredDelaySeconds: number): {
  1637→    compliant: boolean;
  1638→    remainingMs: number;
  1639→  } {
  1640→    const entry = this.crawlDelays.get(domain);
  1641→    if (!entry) {
  1642→      return { compliant: true, remainingMs: 0 };
  1643→    }
  1644→
  1645→    const elapsed = Date.now() - entry.lastAccess;
  1646→    const requiredMs = requiredDelaySeconds * 1000;
  1647→    const remainingMs = Math.max(0, requiredMs - elapsed);
  1648→
  1649→    return {
  1650→      compliant: elapsed >= requiredMs,
  1651→      remainingMs,
  1652→    };
  1653→  }
  1654→
  1655→  /**
  1656→   * Record a crawl access for crawl-delay tracking.
  1657→   */
  1658→  recordAccess(domain: string, crawlDelaySeconds: number | null): void {
  1659→    this.crawlDelays.set(domain, {
  1660→      delayMs: (crawlDelaySeconds || 0) * 1000,
  1661→      lastAccess: Date.now(),
  1662→    });
  1663→  }
  1664→
  1665→  /**
  1666→   * Wait for crawl-delay if needed, then record the access.
  1667→   */
  1668→  async waitForCrawlDelay(domain: string, crawlDelaySeconds: number | null): Promise<void> {
  1669→    if (!crawlDelaySeconds || crawlDelaySeconds <= 0) return;
  1670→
  1671→    const entry = this.crawlDelays.get(domain);
  1672→    if (!entry) {
  1673→      this.recordAccess(domain, crawlDelaySeconds);
  1674→      return;
  1675→    }
  1676→
  1677→    const elapsed = Date.now() - entry.lastAccess;
  1678→    const requiredMs = crawlDelaySeconds * 1000;
  1679→    const remainingMs = requiredMs - elapsed;
  1680→
  1681→    if (remainingMs > 0) {
  1682→      logger.debug({ domain, waitMs: remainingMs }, 'Waiting for crawl-delay');
  1683→      await new Promise((resolve) => setTimeout(resolve, remainingMs));
  1684→    }
  1685→
  1686→    this.recordAccess(domain, crawlDelaySeconds);
  1687→  }
  1688→
  1689→  /**
  1690→   * Get disallowed paths for a domain.
  1691→   */
  1692→  async getDisallowedPaths(domain: string, userAgent: string = '*'): Promise<string[]> {
  1693→    try {
  1694→      const robotsUrl = `https://${domain}/robots.txt`;
  1695→      const parsed = await robotsParser.fetchAndParse(robotsUrl);
  1696→
  1697→      if (!parsed) return [];
  1698→
  1699→      return parsed.rules
  1700→        .filter((r) => !r.allow)
  1701→        .map((r) => r.path);
  1702→    } catch (error: any) {
  1703→      logger.warn({ domain, error: error.message }, 'Failed to get disallowed paths');
  1704→      return [];
  1705→    }
  1706→  }
  1707→
  1708→  /**
  1709→   * Get override history for a domain.
  1710→   */
  1711→  getOverrideHistory(domain: string): { url: string; timestamp: Date; userId: string }[] {
  1712→    return this.overrideLog.get(domain) || [];
  1713→  }
  1714→
  1715→  /**
  1716→   * Check if robots.txt compliance is enabled.
  1717→   */
  1718→  isRespectRobots(): boolean {
  1719→    return this.respectRobots;
  1720→  }
  1721→
  1722→  /**
  1723→   * Update robots.txt compliance setting.
  1724→   */
  1725→  setRespectRobots(respect: boolean): void {

  1727→    logger.info({ respectRobots: respect }, 'Robots.txt respect setting updated');
  1728→  }
  1729→
  1730→  // --- Private Helpers ------------------------------------------------------
  1731→
  1732→  private logOverride(url: string, userId: string): void {
  1733→    if (!this.logOverrides) return;
  1734→
  1735→    const domain = new URL(url).hostname;
  1736→    const entry = { url, timestamp: new Date(), userId };
  1737→
  1738→    const existing = this.overrideLog.get(domain) || [];
  1739→    existing.push(entry);
  1740→    this.overrideLog.set(domain, existing);
  1741→
  1742→    // Also persist to cache for durability
  1743→    cacheSet(`compliance:robots:override:${domain}`, existing, 86400 * 30).catch(() => {});
  1744→  }
  1745→}
