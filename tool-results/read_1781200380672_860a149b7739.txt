     1→/**
     2→ * X-Bogus CDN Extractor -- ScrapeSuite Engine
     3→ *
     4→ * Extracts the real X-Bogus algorithm from TikTok's CDN JavaScript and executes
     5→ * it in a controlled browser context to generate authentic signatures.
     6→ *
     7→ * Architecture:
     8→ *  1. Fetch TikTok's main page HTML
     9→ *  2. Extract script URLs containing the X-Bogus algorithm
    10→ *  3. Download and cache the algorithm JS
    11→ *  4. Execute the algorithm in a sandboxed Playwright context
    12→ *  5. Return the generated X-Bogus signature
    13→ *
    14→ * This replaces the custom murmur-hash approach with the REAL TikTok algorithm,
    15→ * ensuring signatures pass server-side validation.
    16→ */
    17→
    18→import { chromium, type Browser, type BrowserContext, type Page } from 'playwright';
    19→import { createChildLogger } from '../../utils/logger';
    20→import { cacheGet, cacheSet } from '../../utils/redis';
    21→import { db } from '../../utils/db';
    22→import { xbogusSigner } from './xbogus-signer';
    23→import type { XBogusParams, XBogusResult, TikTokDeviceType } from './types';
    24→
    25→const logger = createChildLogger('xbogus-cdn-extractor');
    26→
    27→// ===============================================================================
    28→// CONSTANTS
    29→// ===============================================================================
    30→
    31→/** Patterns that identify X-Bogus algorithm code inside a JS bundle */
    32→const XBOGUS_SIGNATURE_PATTERNS: RegExp[] = [
    33→  /X-Bogus/i,
    34→  /Dkdpgh4ZKsQB80\/Mfvw36XI1R25-WtUVEGayPuHJOjcLNqiz9mCeTFS/,
    35→  /DFSzswVO/,
    36→  /function\s*\w*\s*\([^)]*\)\s*\{[^}]*xor/i,
    37→  /\b0x5BD1E995\b/,
    38→];
    39→
    40→/** Known TikTok CDN hostname prefixes for algorithm bundles */
    41→const TIKTOK_CDN_HOSTS = [
    42→  'sf16-ies-middleware.bytetos.com',
    43→  'sf16-shortev.bytetos.com',
    44→  'sf16-muse-va.bytetos.com',
    45→  'p16-sign-sg.tiktokcdn.com',
    46→  'sf19-ies-middleware.bytetos.com',
    47→];
    48→
    49→/** Version prefix patterns used to detect the algorithm version */
    50→const VERSION_PATTERNS: RegExp[] = [
    51→  /DFSzswVO/,
    52→  /DFSzswV[0-9]/,
    53→  /"[A-Za-z0-9]{8}"/,
    54→];
    55→
    56→/** Default User-Agent for extraction browser context */
    57→const DEFAULT_UA =
    58→  'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36';
    59→
    60→/** Signature validity window (5 minutes, matching TikTok server-side check) */
    61→const SIGNATURE_VALIDITY_MS = 5 * 60 * 1000;
    62→
    63→// ===============================================================================
    64→// INTERFACE: Cached Algorithm Entry
    65→// ===============================================================================
    66→
    67→interface CachedAlgorithm {
    68→  /** The raw JavaScript containing the X-Bogus algorithm */
    69→  js: string;
    70→  /** Timestamp when this was extracted from CDN */
    71→  extractedAt: number;
    72→  /** Detected algorithm version string (e.g. "DFSzswVO") */
    73→  version: string;
    74→}

    76→// ===============================================================================
    77→// X-BOGUS CDN EXTRACTOR
    78→// ===============================================================================
    79→
    80→export class XBogusCDNExtractor {
    81→  // --- In-memory algorithm cache keyed by version ----------------------------
    82→  private algorithmCache = new Map<string, CachedAlgorithm>();
    83→
    84→  // --- Lifecycle state -------------------------------------------------------
    85→  private initialized = false;
    86→  private browserPage: Page | null = null;
    87→  private browser: Browser | null = null;
    88→
    89→  // --- Cache durations -------------------------------------------------------
    90→  /** Algorithm cache duration: 2 hours (TikTok updates frequently) */
    91→  private static ALGORITHM_CACHE_MS = 2 * 60 * 60 * 1000;
    92→  /** Redis cache key for persisted algorithm JS */
    93→  private static ALGORITHM_REDIS_KEY = 'tiktok:xbogus:algorithm';
    94→
    95→  // --- Operational stats -----------------------------------------------------
    96→  private stats = {
    97→    extractionAttempts: 0,
    98→    extractionSuccesses: 0,
    99→    extractionFailures: 0,
   100→    cacheHits: 0,
   101→    cacheMisses: 0,
   102→    fallbackUsed: 0,
   103→    algorithmVersion: 'unknown',
   104→    lastExtractionAt: 0 as number,
   105→    lastExtractionDurationMs: 0 as number,
   106→    sandboxExecutions: 0,
   107→    sandboxFailures: 0,
   108→    totalSignaturesGenerated: 0,
   109→  };
   110→
   111→  // ===========================================================================
   112→  // LIFECYCLE
   113→  // ===========================================================================
   114→
   115→  /**
   116→   * Initialize the extractor -- launch a headless Chromium instance and prepare
   117→   * the sandbox page used for algorithm execution.
   118→   */
   119→  async initialize(): Promise<void> {
   120→    if (this.initialized) {
   121→      logger.debug('XBogusCDNExtractor already initialized -- skipping');
   122→      return;
   123→    }
   124→
   125→    try {
   126→      logger.info('Initializing XBogusCDNExtractor -- launching headless browser');
   127→
   128→      this.browser = await chromium.launch({
   129→        headless: true,
   130→        args: [
   131→          '--no-sandbox',
   132→          '--disable-setuid-sandbox',
   133→          '--disable-dev-shm-usage',
   134→          '--disable-blink-features=AutomationControlled',
   135→          '--disable-features=IsolateOrigins,site-per-process',
   136→          '--no-first-run',
   137→          '--no-default-browser-check',
   138→          '--disable-gpu',
   139→        ],
   140→      });
   141→
   142→      this.initialized = true;
   143→      logger.info('XBogusCDNExtractor initialized successfully');
   144→
   145→      // Attempt an initial algorithm extraction in the background
   146→      this.refreshAlgorithm().catch((err) => {
   147→        logger.warn({ error: (err as Error).message }, 'Initial algorithm extraction failed -- will retry on demand');
   148→      });
   149→    } catch (err: unknown) {
   150→      const message = err instanceof Error ? err.message : String(err);
   151→      logger.error({ error: message }, 'Failed to initialize XBogusCDNExtractor browser');
   152→      throw err;
   153→    }
   154→  }
   155→
   156→  // ===========================================================================
   157→  // PUBLIC API
   158→  // ===========================================================================
   159→
   160→  /**
   161→   * Extract the real X-Bogus algorithm from TikTok's CDN (if not cached) and
   162→   * execute it in a sandboxed Playwright context to sign the given parameters.
   163→   *
   164→   * Falls back to the custom murmur-hash {@link XBogusSignerEngine} when CDN
   165→   * extraction is unavailable or fails.
   166→   */
   167→  async extractAndExecute(params: XBogusParams): Promise<XBogusResult> {
   168→    const startTime = performance.now();
   169→    this.stats.totalSignaturesGenerated++;
   170→
   171→    try {
   172→      // Ensure browser is initialized
   173→      if (!this.initialized || !this.browser) {
   174→        await this.initialize();
   175→      }
   176→
   177→      // Step 1: Get the algorithm JS (from cache or CDN)
   178→      const algorithm = await this.getAlgorithm();
   179→      if (!algorithm) {
   180→        logger.warn('No algorithm available from CDN -- falling back to murmur signer');
   181→        this.stats.fallbackUsed++;
   182→        return xbogusSigner.generate(params);
   183→      }
   184→
   185→      // Step 2: Execute in sandbox
   186→      const xBogus = await this.executeInSandbox(algorithm.js, params);
   187→
   188→      const generationTimeMs = performance.now() - startTime;
   189→
   190→      return {
   191→        xBogus,
   192→        version: algorithm.version,
   193→        generationTimeMs,
   194→        isValid: true,
   195→        expiresAt: Date.now() + SIGNATURE_VALIDITY_MS,
   196→      };
   197→    } catch (err: unknown) {
   198→      const message = err instanceof Error ? err.message : String(err);
   199→      logger.error({ error: message }, 'CDN extraction + execution failed -- falling back to murmur signer');
   200→      this.stats.fallbackUsed++;
   201→      return xbogusSigner.generate(params);
   202→    }
   203→  }
   204→
   205→  /**
   206→   * Force a fresh extraction of the X-Bogus algorithm from TikTok's CDN.
   207→   * Invalidates all caches (in-memory and Redis).
   208→   */
   209→  async refreshAlgorithm(): Promise<void> {
   210→    this.stats.extractionAttempts++;
   211→    const extractStart = performance.now();
   212→
   213→    try {
   214→      logger.info('Refreshing X-Bogus algorithm from TikTok CDN');
   215→
   216→      // Step 1: Discover script URLs from TikTok's main page
   217→      const scriptUrls = await this.fetchAlgorithmScripts();
   218→      if (scriptUrls.length === 0) {
   219→        throw new Error('No algorithm script URLs discovered from TikTok page');
   220→      }
   221→
   222→      logger.info({ scriptCount: scriptUrls.length }, 'Discovered candidate script URLs');
   223→
   224→      // Step 2: Download and identify the algorithm JS
   225→      const algorithmJS = await this.downloadAlgorithmJS(scriptUrls);
   226→      if (!algorithmJS) {
   227→        throw new Error('Downloaded scripts did not contain X-Bogus algorithm');
   228→      }
   229→
   230→      // Step 3: Detect version
   231→      const version = await this.detectAlgorithmVersion(algorithmJS);
   232→
   233→      // Step 4: Cache in-memory
   234→      const cached: CachedAlgorithm = {
   235→        js: algorithmJS,
   236→        extractedAt: Date.now(),
   237→        version,
   238→      };
   239→      this.algorithmCache.set(version, cached);
   240→      this.stats.algorithmVersion = version;
   241→      this.stats.lastExtractionAt = Date.now();
   242→      this.stats.lastExtractionDurationMs = performance.now() - extractStart;
   243→      this.stats.extractionSuccesses++;
   244→
   245→      // Step 5: Persist to Redis with 2-hour TTL
   246→      try {
   247→        await cacheSet(
   248→          XBogusCDNExtractor.ALGORITHM_REDIS_KEY,
   249→          { js: algorithmJS, version, extractedAt: cached.extractedAt },
   250→          Math.floor(XBogusCDNExtractor.ALGORITHM_CACHE_MS / 1000),
   251→        );
   252→      } catch (redisErr: unknown) {
   253→        logger.debug({ error: (redisErr as Error).message }, 'Redis cache write failed -- continuing with in-memory cache only');
   254→      }
   255→
   256→      logger.info(
   257→        { version, jsSize: algorithmJS.length, durationMs: this.stats.lastExtractionDurationMs.toFixed(0) },
   258→        'X-Bogus algorithm extracted and cached',
   259→      );
   260→    } catch (err: unknown) {
   261→      const message = err instanceof Error ? err.message : String(err);
   262→      this.stats.extractionFailures++;
   263→      logger.error({ error: message }, 'Failed to refresh X-Bogus algorithm from CDN');
   264→    }
   265→  }
   266→
   267→  /**
   268→   * Return operational statistics for monitoring / diagnostics.
   269→   */
   270→  getStats(): Record<string, unknown> {
   271→    return {
   272→      ...this.stats,
   273→      initialized: this.initialized,
   274→      algorithmCacheSize: this.algorithmCache.size,
   275→      algorithmCacheEntries: Array.from(this.algorithmCache.entries()).map(([version, entry]) => ({
   276→        version,
   277→        jsSize: entry.js.length,
   278→        ageMs: Date.now() - entry.extractedAt,
   279→      })),
   280→    };
   281→  }
   282→
   283→  // ===========================================================================
   284→  // ALGORITHM DISCOVERY
   285→  // ===========================================================================
   286→
   287→  /**
   288→   * Navigate to TikTok's main page in a Playwright context, intercept loaded
   289→   * script resources, and return URLs for scripts that may contain the X-Bogus
   290→   * signing algorithm.
   291→   *
   292→   * Strategy:
   293→   *  - Create an incognito browser context with realistic headers
   294→   *  - Intercept network responses for JS content-type
   295→   *  - Filter URLs against known CDN hostname patterns
   296→   *  - Also parse the initial HTML for `<script src="...">` tags
   297→   */
   298→  private async fetchAlgorithmScripts(): Promise<string[]> {
   299→    if (!this.browser) {
   300→      throw new Error('Browser not initialized');
   301→    }
   302→
   303→    const collectedUrls = new Set<string>();
   304→    let context: BrowserContext | null = null;
   305→
   306→    try {
   307→      context = await this.browser.newContext({
   308→        userAgent: DEFAULT_UA,
   309→        viewport: { width: 1920, height: 1080 },
   310→        locale: 'en-US',
   311→        timezoneId: 'America/New_York',
   312→        extraHTTPHeaders: {
   313→          'Accept-Language': 'en-US,en;q=0.9',
   314→          'Sec-CH-UA': '"Not_A Brand";v="8", "Chromium";v="120", "Google Chrome";v="120"',
   315→          'Sec-CH-UA-Mobile': '?0',
   316→          'Sec-CH-UA-Platform': '"Windows"',
   317→        },
   318→      });
   319→
   320→      // Mask Playwright automation signals
   321→      await context.addInitScript(() => {
   322→        Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
   323→        const w = globalThis as Record<string, unknown>;
   324→        delete w.__playwright;
   325→        delete w.__pw_manual;
   326→        delete w.__PW_inspect;
   327→      });
   328→
   329→      const page = await context.newPage();
   330→
   331→      // Intercept JS responses and collect their URLs
   332→      page.on('response', (response) => {
   333→        const url = response.url();
   334→        const contentType = response.headers()['content-type'] || '';
   335→        if (
   336→          contentType.includes('javascript') ||
   337→          url.endsWith('.js') ||
   338→          url.includes('.js?') ||
   339→          TIKTOK_CDN_HOSTS.some((host) => url.includes(host))
   340→        ) {
   341→          collectedUrls.add(url);
   342→        }
   343→      });
   344→
   345→      // Navigate to TikTok main page (tolerate non-200; we still collect scripts)
   346→      try {
   347→        await page.goto('https://www.tiktok.com', {
   348→          waitUntil: 'domcontentloaded',
   349→          timeout: 30_000,
   350→        });
   351→      } catch (navErr: unknown) {
   352→        logger.debug({ error: (navErr as Error).message }, 'TikTok page navigation issue -- collected scripts anyway');
   353→      }
   354→
   355→      // Wait a moment for lazy-loaded scripts
   356→      await page.waitForTimeout(3000);
   357→
   358→      // Also parse the initial HTML for <script src> tags as a fallback
   359→      const htmlScriptUrls = await page.evaluate(() => {
   360→        const scripts = Array.from(document.querySelectorAll('script[src]'));
   361→        return scripts.map((s) => s.getAttribute('src') || '');
   362→      });
   363→
   364→      for (const url of htmlScriptUrls) {
   365→        if (url) collectedUrls.add(url);
   366→      }
   367→
   368→      await page.close();
   369→    } catch (err: unknown) {
   370→      const message = err instanceof Error ? err.message : String(err);
   371→      logger.error({ error: message }, 'Error while fetching algorithm script URLs from TikTok');
   372→    } finally {
   373→      if (context) {
   374→        try { await context.close(); } catch {}
   375→      }
   376→    }
   377→
   378→    return Array.from(collectedUrls);
   379→  }
   380→
   381→  // ===========================================================================
   382→  // ALGORITHM DOWNLOAD
   383→  // ===========================================================================
   384→
   385→  /**
   386→   * Download each candidate script, inspect its content for X-Bogus algorithm
   387→   * signatures, and return the concatenated matching JS.
   388→   *
   389→   * Checks the Redis cache first before downloading.
   390→   */
   391→  private async downloadAlgorithmJS(scriptUrls: string[]): Promise<string | null> {
   392→    // Check Redis cache first
   393→    try {
   394→      const cached = await cacheGet<{ js: string; version: string; extractedAt: number }>(
   395→        XBogusCDNExtractor.ALGORITHM_REDIS_KEY,
   396→      );
   397→      if (cached && cached.js && Date.now() - cached.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   398→        this.stats.cacheHits++;
   399→        logger.info({ version: cached.version }, 'Algorithm cache hit (Redis)');
   400→        const version = await this.detectAlgorithmVersion(cached.js);
   401→        this.algorithmCache.set(version, {
   402→          js: cached.js,
   403→          extractedAt: cached.extractedAt,
   404→          version,
   405→        });
   406→        return cached.js;
   407→      }
   408→    } catch (err: unknown) {
   409→      logger.debug({ error: (err as Error).message }, 'Redis cache read failed -- proceeding to download');
   410→    }
   411→
   412→    // Check in-memory cache
   413→    for (const [version, entry] of this.algorithmCache.entries()) {
   414→      if (Date.now() - entry.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   415→        this.stats.cacheHits++;
   416→        logger.info({ version }, 'Algorithm cache hit (memory)');
   417→        return entry.js;
   418→      }
   419→    }
   420→
   421→    this.stats.cacheMisses++;
   422→
   423→    // Download and scan each candidate script
   424→    const algorithmChunks: string[] = [];
   425→
   426→    for (const url of scriptUrls) {
   427→      try {
   428→        const response = await fetch(url, {
   429→          headers: {
   430→            'User-Agent': DEFAULT_UA,
   431→            'Accept': '*/*',
   432→            'Accept-Language': 'en-US,en;q=0.9',
   433→            Referer: 'https://www.tiktok.com/',
   434→          },
   435→          signal: AbortSignal.timeout(15_000),
   436→        });
   437→
   438→        if (!response.ok) continue;
   439→
   440→        const js = await response.text();
   441→
   442→        // Test if this script contains X-Bogus algorithm indicators
   443→        if (this.containsAlgorithm(js)) {
   444→          algorithmChunks.push(js);
   445→          logger.debug({ url: url.substring(0, 120), size: js.length }, 'Identified X-Bogus algorithm script');
   446→        }
   447→      } catch (err: unknown) {
   448→        logger.debug({ url: url.substring(0, 120), error: (err as Error).message }, 'Failed to download candidate script');
   449→      }
   450→    }
   451→
   452→    if (algorithmChunks.length === 0) {
   453→      logger.warn({ urlCount: scriptUrls.length }, 'No algorithm scripts found among candidates');
   454→      return null;
   455→    }
   456→
   457→    return algorithmChunks.join('\n;//---SEPARATOR---;\n');
   458→  }
   459→
   460→  /**
   461→   * Check whether a JS string contains patterns consistent with the X-Bogus
   462→   * signing algorithm.
   463→   */
   464→  private containsAlgorithm(js: string): boolean {
   465→    // Must match at least two distinct patterns to reduce false positives
   466→    let matchCount = 0;
   467→    for (const pattern of XBOGUS_SIGNATURE_PATTERNS) {
   468→      if (pattern.test(js)) {
   469→        matchCount++;
   470→        if (matchCount >= 2) return true;
   471→      }
   472→    }
   473→    return false;
   474→  }
   475→
   476→  // ===========================================================================
   477→  // SANDBOX EXECUTION
   478→  // ===========================================================================
   479→
   480→  /**
   481→   * Execute the extracted X-Bogus algorithm inside a fresh Playwright
   482→   * incognito context and return the generated signature.
   483→   *
   484→   * Security model:
   485→   *  - A brand-new incognito context is created per call
   486→   *  - Realistic TikTok headers and cookies (ttwid, msToken) are set
   487→   *  - The algorithm JS is injected via page.addScriptTag
   488→   *  - The signing function is invoked via page.evaluate()
   489→   *  - The context is destroyed after extracting the result
   490→   */
   491→  private async executeInSandbox(algorithmJS: string, params: XBogusParams): Promise<string> {
   492→    if (!this.browser) {
   493→      throw new Error('Browser not initialized -- cannot execute in sandbox');
   494→    }
   495→
   496→    this.stats.sandboxExecutions++;
   497→    let context: BrowserContext | null = null;
   498→
   499→    try {
   500→      // Step 1: Create incognito context with realistic TikTok headers
   501→      context = await this.browser.newContext({
   502→        userAgent: params.userAgent || DEFAULT_UA,
   503→        viewport: { width: 1920, height: 1080 },
   504→        locale: 'en-US',
   505→        timezoneId: 'America/New_York',
   506→        extraHTTPHeaders: {
   507→          'Accept-Language': 'en-US,en;q=0.9',
   508→          'Sec-CH-UA': '"Not_A Brand";v="8", "Chromium";v="120", "Google Chrome";v="120"',
   509→          'Sec-CH-UA-Mobile': '?0',
   510→          'Sec-CH-UA-Platform': '"Windows"',
   511→        },
   512→      });
   513→
   514→      // Set realistic TikTok cookies
   515→      await context.addCookies([
   516→        {
   517→          name: 'ttwid',
   518→          value: this.generateTTWid(),
   519→          domain: '.tiktok.com',
   520→          path: '/',
   521→        },
   522→        {
   523→          name: 'msToken',
   524→          value: this.generateMsToken(),
   525→          domain: '.tiktok.com',
   526→          path: '/',
   527→        },
   528→        {
   529→          name: 'odin_tt',
   530→          value: this.generateOdinTT(),
   531→          domain: '.tiktok.com',
   532→          path: '/',
   533→        },
   534→      ]);
   535→
   536→      // Mask automation indicators
   537→      await context.addInitScript(() => {
   538→        Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
   539→        const w = globalThis as Record<string, unknown>;
   540→        delete w.__playwright;
   541→        delete w.__pw_manual;
   542→        delete w.__PW_inspect;
   543→      });
   544→
   545→      const page = await context.newPage();
   546→
   547→      // Step 2: Navigate to a blank TikTok-origin page so the algorithm JS
   548→      // runs in the correct origin context
   549→      await page.goto('about:blank');
   550→
   551→      // Step 3: Inject the algorithm JavaScript
   552→      await page.addScriptTag({ content: algorithmJS });
   553→
   554→      // Step 4: Attempt to call the X-Bogus signing function
   555→      // TikTok's algorithm is typically exposed via a global function or
   556→      // accessible through their module system. We try multiple known entry points.
   557→      const xBogus = await page.evaluate(
   558→        ({ url, queryString, userAgent, timestamp, body }) => {
   559→          // Attempt 1: Direct global function (byted_acrawler.sign)
   560→          try {
   561→            const byted = (window as Record<string, any>).byted_acrawler;
   562→            if (byted && typeof byted.sign === 'function') {
   563→              const result = byted.sign({
   564→                url: url + queryString,
   565→                userAgent,
   566→                timestamp,
   567→                body: body || '',
   568→              });
   569→              if (typeof result === 'string' && result.length > 10) return result;
   570→              if (result && typeof result.x_bogus === 'string') return result.x_bogus;
   571→              if (result && typeof result['X-Bogus'] === 'string') return result['X-Bogus'];
   572→            }
   573→          } catch {}
   574→
   575→          // Attempt 2: window._bytedParam via module system
   576→          try {
   577→            const param = (window as Record<string, any>)._bytedParam;
   578→            if (param && typeof param.sign === 'function') {
   579→              return param.sign(url + queryString, userAgent, timestamp, body || '');
   580→            }
   581→          } catch {}
   582→
   583→          // Attempt 3: _webmssdk_env or similar entry
   584→          try {
   585→            const env = (window as Record<string, any>)._webmssdk_env;
   586→            if (env && typeof env.sign === 'function') {
   587→              return env.sign(url + queryString, body || '', userAgent);
   588→            }
   589→          } catch {}
   590→
   591→          // Attempt 4: Search for the signing function in global scope
   592→          const globalKeys = Object.keys(window);
   593→          for (const key of globalKeys) {
   594→            try {
   595→              const val = (window as Record<string, any>)[key];
   596→              if (val && typeof val === 'object' && typeof val.sign === 'function') {
   597→                const result = val.sign({ url: url + queryString, userAgent, timestamp, body: body || '' });
   598→                if (typeof result === 'string' && result.length > 10) return result;
   599→                if (result && typeof result.x_bogus === 'string') return result.x_bogus;
   600→                if (result && typeof result['X-Bogus'] === 'string') return result['X-Bogus'];
   601→              }
   602→            } catch {}
   603→          }
   604→
   605→          return null;
   606→        },
   607→        {
   608→          url: params.url,
   609→          queryString: params.queryString,
   610→          userAgent: params.userAgent,
   611→          timestamp: params.timestamp,
   612→          body: params.body || '',
   613→        },
   614→      );
   615→
   616→      await page.close();
   617→
   618→      if (typeof xBogus === 'string' && xBogus.length > 10) {
   619→        return xBogus;
   620→      }
   621→
   622→      throw new Error('Sandbox execution did not produce a valid X-Bogus signature');
   623→    } catch (err: unknown) {
   624→      const message = err instanceof Error ? err.message : String(err);
   625→      this.stats.sandboxFailures++;
   626→      logger.error({ error: message }, 'Sandbox execution failed');
   627→      throw err;
   628→    } finally {
   629→      if (context) {
   630→        try { await context.close(); } catch {}
   631→      }
   632→    }
   633→  }

   636→  // ALGORITHM VERSION DETECTION
   637→  // ===========================================================================
   638→
   639→  /**
   640→   * Detect the X-Bogus algorithm version from the extracted JavaScript.
   641→   * Looks for version prefixes like "DFSzswVO" and other identifiers.
   642→   */
   643→  private async detectAlgorithmVersion(js: string): Promise<string> {
   644→    for (const pattern of VERSION_PATTERNS) {
   645→      const match = js.match(pattern);
   646→      if (match) {
   647→        return match[0].replace(/"/g, '');
   648→      }
   649→    }
   650→
   651→    // Fallback: hash the first 512 bytes to create a stable version id
   652→    const sample = js.substring(0, 512);
   653→    let hash = 0;
   654→    for (let i = 0; i < sample.length; i++) {
   655→      const chr = sample.charCodeAt(i);
   656→      hash = ((hash << 5) - hash + chr) | 0;
   657→    }
   658→    return `custom-${Math.abs(hash).toString(16).slice(0, 8)}`;
   659→  }

   665→  /**
   666→   * Retrieve the algorithm JS from in-memory cache, Redis, or by performing
   667→   * a fresh CDN extraction.
   668→   */
   669→  private async getAlgorithm(): Promise<CachedAlgorithm | null> {
   670→    // Check in-memory cache (any version, as long as it's not expired)
   671→    for (const [version, entry] of this.algorithmCache.entries()) {
   672→      if (Date.now() - entry.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   673→        this.stats.cacheHits++;
   674→        return entry;
   675→      }
   676→    }
   677→
   678→    // Check Redis cache
   679→    try {
   680→      const cached = await cacheGet<{ js: string; version: string; extractedAt: number }>(
   681→        XBogusCDNExtractor.ALGORITHM_REDIS_KEY,
   682→      );
   683→      if (cached && cached.js && Date.now() - cached.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   684→        this.stats.cacheHits++;
   685→        this.algorithmCache.set(cached.version, {
   686→          js: cached.js,
   687→          extractedAt: cached.extractedAt,
   688→          version: cached.version,
   689→        });
   690→        return this.algorithmCache.get(cached.version) ?? null;
   691→      }
   692→    } catch (err: unknown) {
   693→      logger.debug({ error: (err as Error).message }, 'Redis cache read failed');
   694→    }
   695→
   696→    // Fresh extraction required
   697→    this.stats.cacheMisses++;
   698→    await this.refreshAlgorithm();
   699→
   700→    // Return from in-memory cache after extraction
   701→    for (const [, entry] of this.algorithmCache.entries()) {
   702→      if (Date.now() - entry.extractedAt < XBogusCDNExtractor.ALGORITHM_CACHE_MS) {
   703→        return entry;
   704→      }
   705→    }
   706→
   707→    return null;
   708→  }

   711→   * Generate a plausible ttwid cookie value.
   712→   */
   713→  private generateTTWid(): string {
   714→    const ts = Date.now();
   715→    const rand = Math.random().toString(36).substring(2, 15);
   716→    return `${ts}_${rand}`;
   717→  }

   719→  /**
   720→   * Generate a plausible msToken cookie value (base64-like).
   721→   */
   722→  private generateMsToken(): string {
   723→    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=';
   724→    const length = 107 + Math.floor(Math.random() * 21);
   725→    let token = '';
   726→    for (let i = 0; i < length; i++) {
   727→      token += chars.charAt(Math.floor(Math.random() * chars.length));
   728→    }
   729→    return token;
   730→  }

   735→  private generateOdinTT(): string {
