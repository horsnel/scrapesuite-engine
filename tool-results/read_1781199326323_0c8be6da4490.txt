    37→    37→interface PendingRequest {
    38→    38→  promise: Promise<ScrapeJobResult>;
    39→    39→  resolve: (result: ScrapeJobResult) => void;

    44→    44→
    45→    45→function getDedupeKey(data: ScrapeJobData): string {
    46→    46→  // Dedupe by URL + strategy + extraction instruction + session
    47→    47→  return `${data.url}:${data.strategy || 'auto'}:${data.extract || ''}:${data.proxyTier || ''}:${data.proxyCountry || ''}:${data.proxyCity || ''}:${data.sessionId || ''}`;

    51→    51→
    52→    52→export class Orchestrator {
    53→    53→  /**
    54→    54→   * Process a scrape job through the full pipeline:
    55→    55→   *   dedupe → robots → cache → session → intelligence → adaptive rate limit →
    56→    56→   *   proxy → stealth → fetch → smart retry → record → cache →
    57→    57→   *   template extraction → structured → NL extract → captcha →
    58→    58→   *   output pipeline → bill → persist
    59→    59→   */
    60→    60→  async processJob(data: ScrapeJobData): Promise<ScrapeJobResult> {
    61→    61→    // -- Request Deduplication --------------------------------------------
    62→    62→    const dedupeKey = getDedupeKey(data);
    63→    63→    const existing = inflightRequests.get(dedupeKey);
    64→    64→    if (existing) {
    65→    65→      logger.info({ jobId: data.jobId, url: data.url }, 'Deduped -- waiting for identical in-flight request');
    66→    66→      return existing.promise;
    67→    67→    }
    68→    68→
    69→    69→    const pending: Partial<PendingRequest> = {};
    70→    70→    pending.promise = new Promise<ScrapeJobResult>((resolve, reject) => {
    71→    71→      pending.resolve = resolve;
    72→    72→      pending.reject = reject;
    73→    73→    });
    74→    74→    inflightRequests.set(dedupeKey, pending as PendingRequest);
    75→    75→
    76→    76→    try {
    77→    77→      const result = await this._processJob(data);
    78→    78→      pending.resolve!(result);
    79→    79→      return result;
    80→    80→    } catch (err: any) {
    81→    81→      const result: ScrapeJobResult = {
    82→    82→        jobId: data.jobId,
    83→    83→        status: 'failed',
    84→    84→        strategy: 'http',
    85→    85→        creditsUsed: 0,
    86→    86→        creditsCharged: 0,
    87→    87→        statusCode: 502,
    88→    88→        responseMs: 0,
    89→    89→        error: err.message,
    90→    90→        cached: false,
    91→    91→      };
    92→    92→      pending.resolve!(result);
    93→    93→      return result;
    94→    94→    } finally {
    95→    95→      inflightRequests.delete(dedupeKey);
    96→    96→    }
    97→    97→  }

    99→    99→  private async _processJob(data: ScrapeJobData): Promise<ScrapeJobResult> {
   100→   100→    const startTime = Date.now();
   101→   101→    const domain = data.domain;
   102→   102→    let strategy: JobStrategy = 'http';
   103→   103→    let creditsUsed = 0;
   104→   104→    let creditsCharged = 0;
   105→   105→    let cached = false;
   106→   106→    let html = '';
   107→   107→    let statusCode = 200;
   108→   108→    let extracted: Record<string, any> | undefined;
   109→   109→    let structuredData: Record<string, any> | undefined;
   110→   110→    let proxyId: string | undefined;
   111→   111→    let proxyCountry: string | undefined;
   112→   112→    let proxyUrl: string | undefined;
   113→   113→    let captchaSolved = false;
   114→   114→    let error: string | undefined;
   115→   115→    let retryCount = 0;
   116→   116→    let bandwidthBytes = 0;
   117→   117→    let responseHeaders: Record<string, string> = {};
   118→   118→    let finalUrl = data.url;
   119→   119→    let outputCompressionRatio: number | undefined;
   120→   120→    let appliedOutputFormat: OutputFormat | undefined;
   121→   121→
   122→   122→    // Session state -- populated when sessionId is provided
   123→   123→    let sessionState: import('../session').SessionState | null = null;
   124→   124→
   125→   125→    try {
   126→   126→      // -- Step 1: Check robots.txt --------------------------------------
   127→   127→      if (data.respectRobotsTxt !== false) {
   128→   128→        const robotsCheck = await robotsParser.isAllowed(data.url, 'ScrapeSuite');
   129→   129→        if (!robotsCheck.allowed) {
   130→   130→          logger.info({ jobId: data.jobId, url: data.url }, 'Blocked by robots.txt');
   131→   131→          return {
   132→   132→            jobId: data.jobId,
   133→   133→            status: 'failed',
   134→   134→            strategy: 'cache',
   135→   135→            creditsUsed: 0,
   136→   136→            creditsCharged: 0,
   137→   137→            statusCode: 403,
   138→   138→            responseMs: Date.now() - startTime,
   139→   139→            error: 'URL disallowed by robots.txt',
   140→   140→            cached: false,
   141→   141→          };
   142→   142→        }
   143→   143→      }
   144→   144→
   145→   145→      // -- Step 2: Check Redis cache -------------------------------------
   146→   146→      const cachedResult = await this.checkCache(data.url);
   147→   147→      if (cachedResult) {
   148→   148→        cached = true;
   149→   149→        creditsUsed = 0;
   150→   150→        creditsCharged = 0;
   151→   151→        html = cachedResult.html;
   152→   152→        statusCode = cachedResult.statusCode;
   153→   153→        strategy = 'cache';
   154→   154→        logger.info({ jobId: data.jobId, url: data.url, strategy: 'cache' }, 'Cache hit');
   155→   155→      }
   156→   156→
   157→   157→      if (!cached) {
   158→   158→        // -- Step 3: Session Manager Integration -------------------------
   159→   159→        // If a sessionId is provided, retrieve or create the session state
   160→   160→        // to reuse the same proxy and fingerprint for geo-consistency.
   161→   161→        if (data.sessionId) {
   162→   162→          try {
   163→   163→            sessionState = await sessionManager.getSession(data.sessionId);
   164→   164→            if (sessionState) {
   165→   165→              // Refresh the session TTL on use (sliding window)
   166→   166→              sessionState = await sessionManager.refreshSession(data.sessionId);
   167→   167→              if (sessionState) {
   168→   168→                logger.info(
   169→   169→                  { jobId: data.jobId, sessionId: data.sessionId, proxyId: sessionState.proxyId, proxyCountry: sessionState.proxyCountry },
   170→   170→                  'Session found -- using session proxy and fingerprint',
   171→   171→                );
   172→   172→                // Use the session's proxy and fingerprint
   173→   173→                proxyUrl = sessionState.proxyUrl;
   174→   174→                proxyId = sessionState.proxyId;
   175→   175→                proxyCountry = sessionState.proxyCountry;
   176→   176→              }
   177→   177→            } else {
   178→   178→              // Session doesn't exist or expired -- create a new one
   179→   179→              logger.info(
   180→   180→                { jobId: data.jobId, sessionId: data.sessionId },
   181→   181→                'Session not found or expired -- creating new session',
   182→   182→              );
   183→   183→              try {
   184→   184→                sessionState = await sessionManager.createSession({
   185→   185→                  userId: data.userId,
   186→   186→                  proxyTier: (data.proxyTier as any) || 'residential',
   187→   187→                  proxyCountry: data.proxyCountry,
   188→   188→                  proxyCity: data.proxyCity,
   189→   189→                  proxyAsn: data.proxyAsn,
   190→   190→                  domain,
   191→   191→                });
   192→   192→                proxyUrl = sessionState.proxyUrl;
   193→   193→                proxyId = sessionState.proxyId;
   194→   194→                proxyCountry = sessionState.proxyCountry;
   195→   195→                logger.info(
   196→   196→                  { jobId: data.jobId, sessionId: sessionState.sessionId, proxyId, proxyCountry },
   197→   197→                  'New session created with proxy',
   198→   198→                );
   199→   199→              } catch (sessionErr: any) {
   200→   200→                logger.warn(
   201→   201→                  { jobId: data.jobId, sessionId: data.sessionId, error: sessionErr.message },
   202→   202→                  'Failed to create session -- proceeding without session',
   203→   203→                );
   204→   204→              }
   205→   205→            }
   206→   206→          } catch (sessionErr: any) {
   207→   207→            logger.warn(
   208→   208→              { jobId: data.jobId, sessionId: data.sessionId, error: sessionErr.message },
   209→   209→              'Session manager error -- proceeding without session',
   210→   210→            );
   211→   211→          }
   212→   212→        }
   213→   213→
   214→   214→        // -- Step 4: Consult Domain Intelligence -------------------------
   215→   215→        const recommendation = await domainIntelligence.recommendStrategy(domain);
   216→   216→
   217→   217→        // -- Step 5: Acquire adaptive rate limit token -------------------
   218→   218→        // Uses the new AdaptiveRateLimiter instead of the simple Redis counter.
   219→   219→        try {
   220→   220→          const rateResult = await adaptiveRateLimiter.acquireToken(domain);
   221→   221→          if (!rateResult.allowed) {
   222→   222→            logger.info(
   223→   223→              { jobId: data.jobId, domain, waitMs: rateResult.waitMs, currentRps: rateResult.currentRps },
   224→   224→              'Rate limited -- waiting before proceeding',
   225→   225→            );
   226→   226→            await this.sleep(rateResult.waitMs);
   227→   227→            // Try one more time after waiting
   228→   228→            const retryResult = await adaptiveRateLimiter.acquireToken(domain);
   229→   229→            if (!retryResult.allowed) {
   230→   230→              logger.warn(
   231→   231→                { jobId: data.jobId, domain, waitMs: retryResult.waitMs },
   232→   232→                'Rate limit token still not available after wait -- proceeding anyway',
   233→   233→              );
   234→   234→            }
   235→   235→          }
   236→   236→        } catch (rateErr: any) {
   237→   237→          logger.warn(
   238→   238→            { jobId: data.jobId, domain, error: rateErr.message },
   239→   239→            'Adaptive rate limiter error -- falling back to simple rate limit',
   240→   240→          );
   241→   241→          // Fallback to the old simple rate limiter
   242→   242→          await this.acquireSimpleRateLimitToken(domain, recommendation.safeRps);
   243→   243→        }
   244→   244→
   245→   245→        // -- Step 6: Get proxy from Smart IP Pool (v3.1+) -----------------
   246→   246→        // Uses the unified mega-pool with reputation tracking, IP cooling,
   247→   247→        // auto-scaling, and multi-provider aggregation. Falls back to
   248→   248→        // legacy getProxy() if the smart pool fails.
   249→   249→        // If session already provided a proxy, skip this step.
   250→   250→        if (!proxyUrl) {
   251→   251→          const proxySelection = await proxyManager.getSmartProxy(
   252→   252→            domain,
   253→   253→            (data.proxyTier as any) || recommendation.proxyTier,
   254→   254→            data.proxyCountry,
   255→   255→            {
   256→   256→              city: data.proxyCity,
   257→   257→              asn: data.proxyAsn,
   258→   258→              sessionId: data.sessionId,
   259→   259→              // For domains known to have anti-bot, require IPs with good reputation
   260→   260→              requireReputation: recommendation.strategy === 'stealth-browser',
   261→   261→            },
   262→   262→          );
   263→   263→
   264→   264→          proxyUrl = proxySelection?.proxyUrl;
   265→   265→          proxyId = proxySelection?.proxyId;
   266→   266→          proxyCountry = proxySelection?.country;
   267→   267→        }
   268→   268→
   269→   269→        // -- Step 7: Execute fetch (with smart retry logic) --------------
   270→   270→        const effectiveStrategy: JobStrategy =
   271→   271→          data.strategy === 'auto' || !data.strategy
   272→   272→            ? recommendation.strategy
   273→   273→            : (data.strategy as JobStrategy);
   274→   274→
   275→   275→        // If renderJs is requested, force browser strategy
   276→   276→        const forceBrowser = data.renderJs === true;
   277→   277→        let actualStrategy: JobStrategy = forceBrowser && effectiveStrategy === 'http'
   278→   278→          ? 'browser'
   279→   279→          : effectiveStrategy;
   280→   280→
   281→   281→        // If session has a fingerprint profile, use it for stealth browser requests
   282→   282→        // (handled inside fetchViaStealthBrowser / fetchViaHttp via stealthEngine.getRandomProfile())
   283→   283→
   284→   284→        let fetchResult = await this.executeFetch(
   285→   285→          actualStrategy, data.url, domain, proxyUrl,
   286→   286→          data.waitForSelector, data.timeout, data.headers,
   287→   287→        );
   288→   288→
   289→   289→        html = fetchResult.html;
   290→   290→        statusCode = fetchResult.statusCode;
   291→   291→        strategy = fetchResult.strategy;
   292→   292→        bandwidthBytes = Buffer.byteLength(html, 'utf8');
   293→   293→        responseHeaders = fetchResult.headers || {};
   294→   294→        finalUrl = fetchResult.finalUrl || data.url;
   295→   295→
   296→   296→        // Record the initial response to the adaptive rate limiter
   297→   297→        const isSuccess = statusCode >= 200 && statusCode < 400;
   298→   298→        try {
   299→   299→          await adaptiveRateLimiter.recordResponse(domain, {
   300→   300→            success: isSuccess,
   301→   301→            statusCode,
   302→   302→            responseMs: Date.now() - startTime,
   303→   303→          });
   304→   304→        } catch (rateErr: any) {
   305→   305→          logger.debug({ domain, error: rateErr.message }, 'Failed to record response to adaptive rate limiter');
   306→   306→        }
   307→   307→
   308→   308→        // -- Step 8: Smart Retry -- Empty HTML → browser -----------------
   309→   309→        if (strategy === 'http' && this.isEmptyHtml(html) && retryCount < MAX_RETRIES) {
   310→   310→          retryCount++;
   311→   311→          logger.info({ jobId: data.jobId, url: data.url, retry: retryCount }, 'Empty HTML -- escalating to browser');
   312→   312→
   313→   313→          // Rotate fingerprint before retry
   314→   314→          const rotatedProfile = stealthEngine.getRandomProfile();
   315→   315→          logger.debug(
   316→   316→            { jobId: data.jobId, userAgent: rotatedProfile.userAgent },
   317→   317→            'Rotated fingerprint profile for retry',
   318→   318→          );
   319→   319→
   320→   320→          fetchResult = await this.executeFetch(
   321→   321→            'browser', data.url, domain, proxyUrl,
   322→   322→            data.waitForSelector, data.timeout, data.headers,
   323→   323→          );
   324→   324→          html = fetchResult.html;
   325→   325→          statusCode = fetchResult.statusCode;
   326→   326→          strategy = fetchResult.strategy;
   327→   327→          bandwidthBytes = Buffer.byteLength(html, 'utf8');
   328→   328→          responseHeaders = fetchResult.headers || {};
   329→   329→
   330→   330→          // Record retry outcome
   331→   331→          try {
   332→   332→            await adaptiveRateLimiter.recordResponse(domain, {
   333→   333→              success: statusCode >= 200 && statusCode < 400,
   334→   334→              statusCode,
   335→   335→              responseMs: Date.now() - startTime,
   336→   336→            });
   337→   337→          } catch {}
   338→   338→        }
   339→   339→
   340→   340→        // -- Step 9: Smart Retry -- 429 Rate Limited ---------------------
   341→   341→        if (statusCode === 429 && retryCount < MAX_RETRIES) {
   342→   342→          retryCount++;
   343→   343→          logger.info(
   344→   344→            { jobId: data.jobId, url: data.url, retry: retryCount, currentRps: (await adaptiveRateLimiter.getDomainStatus(domain)).currentRps },
   345→   345→            '429 rate-limited -- waiting cooldown then retrying with different proxy',
   346→   346→          );
   347→   347→
   348→   348→          // Wait based on adaptive rate limiter's cooldown
   349→   349→          const domainStatus = await adaptiveRateLimiter.getDomainStatus(domain).catch(() => null);
   350→   350→          const cooldownMs = domainStatus?.cooldownRemainingMs || 5_000;
   351→   351→          logger.info({ jobId: data.jobId, domain, cooldownMs }, 'Waiting for rate limit cooldown');
   352→   352→          await this.sleep(cooldownMs + 500); // Add small buffer
   353→   353→
   354→   354→          // Get a DIFFERENT proxy for the retry
   355→   355→          const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   356→   356→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   357→   357→
   358→   358→          // Rotate fingerprint
   359→   359→          stealthEngine.getRandomProfile();
   360→   360→
   361→   361→          // Re-acquire rate limit token after cooldown
   362→   362→          try {
   363→   363→            const retryToken = await adaptiveRateLimiter.acquireToken(domain);
   364→   364→            if (!retryToken.allowed) {
   365→   365→              await this.sleep(retryToken.waitMs);
   366→   366→            }
   367→   367→          } catch {}
   368→   368→
   369→   369→          // Retry with stealth browser (more resilient to rate limits)
   370→   370→          fetchResult = await this.executeFetch(
   371→   371→            'stealth-browser', data.url, domain, retryProxyUrl,
   372→   372→            data.waitForSelector, data.timeout, data.headers,
   373→   373→          );
   374→   374→
   375→   375→          if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   376→   376→            html = fetchResult.html;
   377→   377→            statusCode = fetchResult.statusCode;
   378→   378→            strategy = fetchResult.strategy;
   379→   379→            bandwidthBytes = Buffer.byteLength(html, 'utf8');
   380→   380→            responseHeaders = fetchResult.headers || {};
   381→   381→            if (retryProxy) {
   382→   382→              proxyId = retryProxy.proxyId;
   383→   383→              proxyCountry = retryProxy.country;
   384→   384→              proxyUrl = retryProxyUrl;
   385→   385→            }
   386→   386→          }
   387→   387→
   388→   388→          // Record retry outcome
   389→   389→          try {
   390→   390→            await adaptiveRateLimiter.recordResponse(domain, {
   391→   391→              success: fetchResult.statusCode >= 200 && fetchResult.statusCode < 400,
   392→   392→              statusCode: fetchResult.statusCode,
   393→   393→              responseMs: Date.now() - startTime,
   394→   394→            });
   395→   395→          } catch {}
   396→   396→        }
   397→   397→
   398→   398→        // -- Step 10: Smart Retry -- 403 with Anti-bot -------------------
   399→   399→        const antiBot = this.detectAntiBot(html);
   400→   400→        if (
   401→   401→          (statusCode === 403 && (antiBot.confidenceScore > 0.3 || antiBot.cloudflare || antiBot.datadome || antiBot.akamai)) &&
   402→   402→          retryCount < MAX_RETRIES
   403→   403→        ) {
   404→   404→          retryCount++;
   405→   405→          logger.info(
   406→   406→            { jobId: data.jobId, url: data.url, retry: retryCount, antiBotScore: antiBot.confidenceScore },
   407→   407→            '403 with anti-bot detected -- rotating proxy AND fingerprint, retrying with stealth',
   408→   408→          );
   409→   409→
   410→   410→          // Rotate BOTH proxy AND fingerprint
   411→   411→          const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   412→   412→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   413→   413→          const rotatedProfile = stealthEngine.getRandomProfile();
   414→   414→          logger.debug(
   415→   415→            { jobId: data.jobId, rotatedUserAgent: rotatedProfile.userAgent, rotatedPlatform: rotatedProfile.platform },
   416→   416→            'Rotated fingerprint profile for anti-bot retry',
   417→   417→          );
   418→   418→
   419→   419→          fetchResult = await this.executeFetch(
   420→   420→            'stealth-browser', data.url, domain, retryProxyUrl,
   421→   421→            data.waitForSelector, data.timeout, data.headers,
   422→   422→          );
   423→   423→
   424→   424→          if (fetchResult.statusCode >= 200 && fetchResult.statusCode < 400) {
   425→   425→            html = fetchResult.html;
   426→   426→            statusCode = fetchResult.statusCode;
   427→   427→            strategy = fetchResult.strategy;
   428→   428→            bandwidthBytes = Buffer.byteLength(html, 'utf8');
   429→   429→            responseHeaders = fetchResult.headers || {};
   430→   430→            if (retryProxy) {
   431→   431→              proxyId = retryProxy.proxyId;
   432→   432→              proxyCountry = retryProxy.country;
   433→   433→              proxyUrl = retryProxyUrl;
   434→   434→            }
   435→   435→          }
   436→   436→
   437→   437→          // Record retry outcome
   438→   438→          try {
   439→   439→            await adaptiveRateLimiter.recordResponse(domain, {
   440→   440→              success: fetchResult.statusCode >= 200 && fetchResult.statusCode < 400,
   441→   441→              statusCode: fetchResult.statusCode,
   442→   442→              responseMs: Date.now() - startTime,
   443→   443→              hadAntiBot: true,
   444→   444→            });
   445→   445→          } catch {}
   446→   446→        }
   447→   447→
   448→   448→        // -- Step 11: Smart Retry -- 503 Service Unavailable --------------
   449→   449→        if (statusCode === 503 && retryCount < MAX_RETRIES) {
   450→   450→          retryCount++;
   451→   451→          logger.info(
   452→   452→            { jobId: data.jobId, url: data.url, retry: retryCount },
   453→   453→            '503 service unavailable -- retrying with different proxy after delay',
   454→   454→          );
   455→   455→
   456→   456→          // Short delay before retry
   457→   457→          await this.sleep(PROXY_RETRY_DELAY_MS);
   458→   458→
   459→   459→          // Get a different proxy
   460→   460→          const retryProxy = await proxyManager.getProxy(domain, 'residential', data.proxyCountry, 'random');
   461→   461→          const retryProxyUrl = retryProxy?.proxyUrl || proxyUrl;
   462→   462→
   463→   463→          // Rotate fingerprint
   464→   464→          stealthEngine.getRandomProfile();
