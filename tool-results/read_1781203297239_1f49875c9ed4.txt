   526→  /**
   527→   * Get the best proxy from any available source.
   528→   * Enhanced: 10-dimension unified scoring, intelligence-guided selection.
   529→   */
   530→  async getProxy(request: MegaPoolRequest): Promise<MegaPoolResult | null> {
   531→    const {
   532→      domain,
   533→      tier = 'residential',
   534→      country,
   535→      city,
   536→      asn,
   537→      sessionId,
   538→      maxCostPerGb,
   539→      requireReputation = false,
   540→      preferSource,
   541→      avoidSources = [],
   542→    } = request;
   543→
   544→    this.totalAcquisitions++;
   545→    const startTime = Date.now();
   546→
   547→    // Check pool intelligence for source hints
   548→    const intelligenceHint = this.getIntelligenceHint(domain, tier, country);
   549→
   550→    const sourceOrder = this.buildSourceOrder(domain, tier, country || 'any');
   551→
   552→    let filteredSources = sourceOrder.filter(s => !avoidSources.includes(s));
   553→
   554→    // Apply intelligence hint
   555→    if (intelligenceHint && !avoidSources.includes(intelligenceHint)) {
   556→      const hintIdx = filteredSources.indexOf(intelligenceHint);
   557→      if (hintIdx > 0) {
   558→        filteredSources.splice(hintIdx, 1);
   559→        filteredSources.unshift(intelligenceHint);
   560→      }
   561→    }
   562→
   563→    if (preferSource && !avoidSources.includes(preferSource)) {
   564→      const preferredIdx = filteredSources.indexOf(preferSource);
   565→      if (preferredIdx > 0) {
   566→        filteredSources.splice(preferredIdx, 1);
   567→        filteredSources.unshift(preferSource);
   568→      } else if (preferredIdx === -1) {
   569→        filteredSources.unshift(preferSource);
   570→      }
   571→    }
   572→
   573→    // Step 1: Check sticky session
   574→    if (sessionId && !avoidSources.includes('database')) {
   575→      const stickyResult = await this.tryStickySession(sessionId, domain);
   576→      if (stickyResult) {
   577→        this.recordAcquisitionSuccess(stickyResult.source, startTime);
   578→        return stickyResult;
   579→      }
   580→    }
   581→
   582→    // Step 2: Check pre-warmed pool
   583→    if (!avoidSources.includes('database')) {
   584→      const prewarmedResult = await this.tryPrewarmedPool(domain, tier, country);
   585→      if (prewarmedResult) {
   586→        this.recordAcquisitionSuccess(prewarmedResult.source, startTime);
   587→        return prewarmedResult;
   588→      }
   589→    }
   590→
   591→    // Steps 3+: Try sources in priority order
   592→    for (const source of filteredSources) {
   593→      if (source === 'database') continue;
   594→
   595→      const sourceCost = SOURCE_COST[source];
   596→      if (maxCostPerGb && sourceCost > maxCostPerGb) continue;
   597→
   598→      const health = this.sourceHealthMap.get(source);
   599→      if (health && !health.healthy) continue;
   600→
   601→      let result: MegaPoolResult | null = null;
   602→
   603→      try {
   604→        switch (source) {
   605→          case 'rotating':
   606→            result = await this.tryRotatingSession(domain, tier, country, city, asn);
   607→            break;
   608→          case 'subnet':
   609→            result = await this.trySubnetExpander(domain, tier, country);
   610→            break;
   611→          case 'bulk':
   612→            result = await this.tryBulkSession(domain, tier, country);
   613→            break;
   614→          case 'brightdata':
   615→          case 'oxylabs':
   616→          case 'smartproxy':
   617→          case 'iproyal':
   618→          case 'webshare':
   619→            result = await this.tryResidentialProvider(source, domain, tier, country, city, asn, maxCostPerGb);
   620→            break;
   621→          case 'free':
   622→            result = await this.tryFreeProxy(domain, tier, country);
   623→            break;
   624→          case 'tor':
   625→            result = await this.tryTorPool(domain, country);
   626→            break;
   627→          case 'fusion':
   628→            result = await this.tryFusionSource(domain, tier, country);
   629→            break;
   630→          case 'chain':
   631→          case 'breeder':
   632→          case 'quantum':
   633→          case 'plasma':
   634→            // Advanced sources -- not yet implemented
   635→            break;
   636→          default:
   637→            break;
   638→        }
   639→      } catch (err: any) {
   640→        logger.warn({ source, domain, error: err.message }, 'Source acquisition failed');
   641→        this.recordSourceFailure(source);
   642→      }
   643→
   644→      if (result) {
   645→        // Compute unified score (10 dimensions)
   646→        result.unifiedScore = this.computeUnifiedScore(result, request);
   647→        result.scoreBreakdown = this.computeScoreBreakdown(result, request);
   648→
   649→        if (sessionId) {
   650→          await this.setStickySession(sessionId, result);
   651→        }
   652→
   653→        this.recordAcquisitionSuccess(result.source, startTime);
   654→        this.recordIntelligence(domain, result.source, true, result.reputationScore);
   655→        return result;
   656→      }
   657→    }
   658→
   659→    // Last resort: Try database proxies
   660→    if (!avoidSources.includes('database')) {
   661→      try {
   662→        const poolResult = await smartIPPool.getProxy({
   663→          domain, tier, country, city, asn, sessionId, requireReputation, maxCostPerGb,
   664→        });
   665→
   666→        if (poolResult) {
   667→          const result = this.convertIPPoolResult(poolResult, 'database');
   668→          result.unifiedScore = this.computeUnifiedScore(result, request);
   669→          result.scoreBreakdown = this.computeScoreBreakdown(result, request);
   670→          if (sessionId) await this.setStickySession(sessionId, result);
   671→          this.recordAcquisitionSuccess('database', startTime);
   672→          this.recordIntelligence(domain, 'database', true, result.reputationScore);
   673→          return result;
   674→        }
   675→      } catch (err: any) {
   676→        logger.warn({ domain, error: err.message }, 'Database proxy selection failed');
   677→      }
   678→    }
   679→
   680→    this.recordDemandSignal(domain, tier, country);
   681→    this.totalFailedAcquisitions++;
   682→    this.recordIntelligence(domain, 'none' as any, false, 0);
   683→
   684→    logger.warn({ domain, tier, country, sourcesTried: filteredSources.length, acquisitionTimeMs: Date.now() - startTime }, 'No proxy available');
   685→
   686→    return null;
   687→  }

   691→  /**
   692→   * Compute a unified score (0-100) for a proxy across 10 dimensions.
   693→   */
   694→  computeUnifiedScore(result: MegaPoolResult, request: MegaPoolRequest): number {
   695→    const breakdown = this.computeScoreBreakdown(result, request);
   696→    return breakdown.totalScore;
   697→  }

   699→  /**
   700→   * Compute score breakdown across 10 dimensions.
   701→   */
   702→  computeScoreBreakdown(result: MegaPoolResult, request: MegaPoolRequest): ScoreBreakdown {
   703→    // 1. Success Rate Score (0-100)
   704→    const successRateScore = Math.round(result.reputationScore * 100);
   705→
   706→    // 2. Latency Score (0-100) -- lower is better
   707→    const latencyScore = Math.max(0, Math.round(100 - (result.costPerGb * 5)));
   708→
   709→    // 3. Anonymity Score (0-100) -- based on tier/source
   710→    const anonymityScore = Math.round(SOURCE_QUALITY[result.source] * 100);
   711→
   712→    // 4. Geo Match Score (0-100)
   713→    const geoMatchScore = (request.country && result.country === request.country.toUpperCase()) ? 100 : 30;
   714→
   715→    // 5. Cost Efficiency Score (0-100) -- lower cost is better
   716→    const costEfficiencyScore = Math.max(0, Math.round(100 - SOURCE_COST[result.source] * 5));
   717→
   718→    // 6. Availability Score (0-100) -- based on source health
   719→    const health = this.sourceHealthMap.get(result.source);
   720→    const availabilityScore = health ? Math.round(health.successRate * 100) : 50;
   721→
   722→    // 7. Reliability Score (0-100) -- based on source outcomes
   723→    const outcomes = this.sourceOutcomes.get(result.source);
   724→    const reliabilityScore = outcomes
   725→      ? Math.round((outcomes.successes / Math.max(1, outcomes.successes + outcomes.failures)) * 100)
   726→      : 50;
   727→
   728→    // 8. Freshness Score (0-100) -- recently acquired is better
   729→    const ageMs = Date.now() - result.acquiredAt;
   730→    const freshnessScore = Math.max(0, Math.round(100 - (ageMs / 60000) * 10));
   731→
   732→    // 9. Tier Match Score (0-100)
   733→    const tierMatchScore = (request.tier && result.tier === request.tier) ? 100 : 50;
   734→
   735→    // 10. Historical Performance Score (0-100) -- from intelligence
   736→    const intelKey = `${request.domain}:${result.source}`;
   737→    const intel = this.poolIntelligence.get(intelKey);
   738→    const historicalPerformanceScore = intel
   739→      ? Math.round((intel.successCount / Math.max(1, intel.successCount + intel.failureCount)) * 100)
   740→      : 50;
   741→
   742→    const totalScore = Math.round(
   743→      successRateScore * SCORING_WEIGHTS.successRate +
   744→      latencyScore * SCORING_WEIGHTS.latency +
   745→      anonymityScore * SCORING_WEIGHTS.anonymity +
   746→      geoMatchScore * SCORING_WEIGHTS.geoMatch +
   747→      costEfficiencyScore * SCORING_WEIGHTS.costEfficiency +
   748→      availabilityScore * SCORING_WEIGHTS.availability +
   749→      reliabilityScore * SCORING_WEIGHTS.reliability +
   750→      freshnessScore * SCORING_WEIGHTS.freshness +
   751→      tierMatchScore * SCORING_WEIGHTS.tierMatch +
   752→      historicalPerformanceScore * SCORING_WEIGHTS.historicalPerformance
   753→    );
   754→
   755→    return {
   756→      successRateScore,
   757→      latencyScore,
   758→      anonymityScore,
   759→      geoMatchScore,
   760→      costEfficiencyScore,
   761→      availabilityScore,
   762→      reliabilityScore,
   763→      freshnessScore,
   764→      tierMatchScore,
   765→      historicalPerformanceScore,
   766→      totalScore: Math.max(0, Math.min(100, totalScore)),
   767→    };
   768→  }

   797→  /**
   798→   * Get a source hint from pool intelligence.
   799→   */
   800→  private getIntelligenceHint(domain: string, tier: string, country?: string): MegaPoolSource | null {
   801→    let bestSource: MegaPoolSource | null = null;
   802→    let bestScore = 0;
   803→
   804→    for (const [key, entry] of this.poolIntelligence) {
   805→      if (!key.startsWith(`${domain}:`)) continue;
   806→
   807→      const totalAttempts = entry.successCount + entry.failureCount;
   808→      if (totalAttempts < 3) continue;
   809→
   810→      const successRate = entry.successCount / totalAttempts;
   811→      if (successRate > bestScore) {
   812→        bestScore = successRate;
   813→        bestSource = entry.source;
   814→      }
   815→    }
   816→
   817→    return bestScore >= 0.6 ? bestSource : null;
   818→  }

   866→  /**
   867→   * Run pool compression -- merge duplicate proxy entries.
   868→   * Identifies proxies that are the same underlying IP but stored
   869→   * under different URLs (e.g., different credentials or ports).
   870→   */
   871→  async runPoolCompression(): Promise<{
   872→    duplicatesFound: number;
   873→    entriesMerged: number;
   874→    spaceSavedPercent: number;
   875→  }> {
   876→    let duplicatesFound = 0;
   877→    let entriesMerged = 0;
   878→
   879→    try {
   880→      const proxies = await db.proxy.findMany({
   881→        where: { retired: false },
   882→        select: { id: true, url: true, country: true, successRate: true },
   883→        take: 50_000,
   884→      });
   885→
   886→      // Group by hostname to find duplicates
   887→      const hostnameMap = new Map<string, Array<{ id: string; url: string; country: string; successRate: number }>>();
   888→
   889→      for (const proxy of proxies) {
   890→        try {
   891→          const parsed = new URL(proxy.url);
   892→          const hostname = parsed.hostname;
   893→
   894→          if (!hostnameMap.has(hostname)) {
   895→            hostnameMap.set(hostname, []);
   896→          }
   897→          hostnameMap.get(hostname)!.push(proxy);
   898→        } catch {
   899→          // Invalid URL -- skip
   900→        }
   901→      }
   902→
   903→      // Find hostnames with multiple entries
   904→      const mergePromises: Promise<number>[] = [];
   905→
   906→      for (const [hostname, entries] of hostnameMap) {
   907→        if (entries.length <= 1) continue;
   908→
   909→        duplicatesFound += entries.length - 1;
   910→
   911→        // Keep the one with highest success rate, retire the rest
   912→        entries.sort((a, b) => b.successRate - a.successRate);
   913→        const keeper = entries[0];
   914→
   915→        for (let i = 1; i < entries.length; i++) {
   916→          const duplicate = entries[i];
   917→          mergePromises.push((async () => {
   918→            try {
   919→              await db.proxy.update({
   920→                where: { id: duplicate.id },
   921→                data: { retired: true, lastChecked: new Date() },
   922→              });
   923→              return 1;
   924→            } catch {
   925→              return 0;
   926→            }
   927→          })());
   928→        }
   929→      }
   930→
   931→      // Process merges in parallel batches
   932→      for (let i = 0; i < mergePromises.length; i += 100) {
   933→        const batch = mergePromises.slice(i, i + 100);
   934→        const results = await Promise.allSettled(batch);
   935→        for (const result of results) {
   936→          if (result.status === 'fulfilled') entriesMerged += result.value;
   937→        }
   938→      }
   939→
   940→      this.compressionState.entriesCompressed += entriesMerged;
   941→      this.compressionState.spaceSavedPercent = proxies.length > 0
   942→        ? Math.round((entriesMerged / proxies.length) * 100 * 100) / 100
   943→        : 0;
   944→      this.compressionState.lastRunAt = Date.now();
   945→
   946→      if (entriesMerged > 0) {
   947→        logger.info(
   948→          { duplicatesFound, entriesMerged, spaceSavedPercent: this.compressionState.spaceSavedPercent },
   949→          'Pool compression: duplicate entries merged',
   950→        );
   951→      }
   952→    } catch (err: any) {
   953→      logger.debug({ error: err.message }, 'Pool compression failed');
   954→    }
   955→
   956→    return {
   957→      duplicatesFound,
   958→      entriesMerged,
   959→      spaceSavedPercent: this.compressionState.spaceSavedPercent,
   960→    };
   961→  }

   969→  async runPoolFusion(): Promise<{
   970→    proxiesFused: number;
   971→    peersChecked: number;
   972→  }> {

   979→      if (peerData && Array.isArray(peerData)) {
   980→        for (const peer of peerData) {
   981→          if (Date.now() - peer.timestamp > 5 * 60 * 1000) continue; // Skip stale peers
   982→          this.fusionState.peersConnected++;
   983→          peersChecked++;
   984→        }
   985→      }
   986→    } catch {}
   987→
   988→    // Try to fuse proxies from shared cache

   990→      const sharedProxies = await cacheGet<Array<{
   991→        url: string;
   992→        provider: string;
   993→        country: string;
   994→        tier: string;
   995→        successRate: number;
   996→      }>>('mega_pool:shared_proxies');
   997→
   998→      if (sharedProxies && Array.isArray(sharedProxies)) {
   999→        const newProxies = sharedProxies.filter(sp => sp.successRate >= 0.5);
  1000→
  1001→        for (const proxy of newProxies.slice(0, 50)) {
  1002→          try {
  1003→            const existing = await db.proxy.findFirst({
  1004→              where: { url: proxy.url, retired: false },
  1005→            });
  1006→
  1007→            if (!existing) {
  1008→              await db.proxy.create({
  1009→                data: {
  1010→                  id: `fusion-${Buffer.from(proxy.url).toString('base64url').slice(0, 24)}`,
  1011→                  url: proxy.url,
  1012→                  provider: proxy.provider || 'fusion',
  1013→                  country: (proxy.country || 'unknown').toUpperCase(),
  1014→                  tier: (proxy.tier || 'residential') as any,
  1015→                  successRate: proxy.successRate,
  1016→                  failures: 0,
  1017→                  consecutiveFailures: 0,
  1018→                  retired: false,
  1019→                  sticky: false,
  1020→                  addedAt: new Date(),
  1021→                },
  1022→              });
  1023→              proxiesFused++;
  1024→            }
  1025→          } catch {}
  1026→        }
  1027→      }
  1028→    } catch (err: any) {
  1029→      logger.debug({ error: err.message }, 'Pool fusion failed');
  1030→    }
  1031→
  1032→    // Share our best proxies
  1033→    try {
  1034→      const ourBestProxies = await db.proxy.findMany({
  1035→        where: { retired: false, successRate: { gte: 0.7 } },
  1036→        take: 100,
  1037→        orderBy: { successRate: 'desc' },

  1041→        url: p.url,
  1042→        provider: p.provider,
  1043→        country: p.country,
  1044→        tier: p.tier,
  1045→        successRate: p.successRate,
  1046→      })), 300).catch(() => {});
  1047→    } catch {}
  1048→
  1049→    this.fusionState.proxiesFused += proxiesFused;
  1050→    this.fusionState.lastRunAt = Date.now();
  1051→
  1052→    if (proxiesFused > 0) {
  1053→      logger.info({ proxiesFused, peersChecked }, 'Pool fusion: proxies fused from peers');
  1054→    }
  1055→
  1056→    return { proxiesFused, peersChecked };
  1057→  }

  1065→  async runTierRebalance(): Promise<{
  1066→    promotions: number;
  1067→    demotions: number;
  1068→  }> {
  1069→    let promotions = 0;
  1070→    let demotions = 0;
  1071→

  1081→
  1082→        let tierPromotions = 0;
  1083→        let tierDemotions = 0;
  1084→
  1085→        for (const proxy of proxies) {
  1086→          // Promote high-performing datacenter/mobile to residential
  1087→          if (tier === 'datacenter' && proxy.successRate >= 0.8) {

  1089→              await db.proxy.update({
  1090→                where: { id: proxy.id },

  1092→              });
  1093→              tierPromotions++;
  1094→            } catch {}
  1095→          }
  1096→
  1097→          // Demote low-performing residential to datacenter
  1098→          if (tier === 'residential' && proxy.successRate < 0.3) {
  1099→            try {
  1100→              await db.proxy.update({
  1101→                where: { id: proxy.id },
  1102→                data: { tier: 'datacenter' as any },
  1103→              });
  1104→              tierDemotions++;
  1105→            } catch {}
  1106→          }
  1107→
  1108→          // Promote high-performing ISP to residential
  1109→          if (tier === 'isp' && proxy.successRate >= 0.7) {
  1110→            try {
  1111→              await db.proxy.update({
  1112→                where: { id: proxy.id },
  1113→                data: { tier: 'residential' as any },
  1114→              });
  1115→              tierPromotions++;
  1116→            } catch {}
  1117→          }
  1118→        }
  1119→
  1120→        // Update tier stats
  1121→        const stats = this.tierStats.get(tier) || {
  1122→          tier,
  1123→          count: proxies.length,
  1124→          avgScore: proxies.length > 0 ? proxies.reduce((s, p) => s + p.successRate, 0) / proxies.length : 0,
  1125→          promotions: 0,
  1126→          demotions: 0,
  1127→          lastRebalancedAt: Date.now(),
  1128→        };
  1129→
  1130→        stats.count = proxies.length;
  1131→        stats.promotions += tierPromotions;
  1132→        stats.demotions += tierDemotions;
  1133→        stats.lastRebalancedAt = Date.now();
  1134→        this.tierStats.set(tier, stats);
  1135→
  1136→        promotions += tierPromotions;
  1137→        demotions += tierDemotions;
  1138→      }
  1139→    } catch (err: any) {
  1140→      logger.debug({ error: err.message }, 'Tier rebalance failed');
  1141→    }
  1142→
  1143→    if (promotions > 0 || demotions > 0) {
  1144→      logger.info({ promotions, demotions }, 'Tier rebalance completed');
  1145→    }
  1146→
  1147→    return { promotions, demotions };
  1148→  }

  1201→    }
  1202→
  1203→    if (source) {
  1204→      this.recordSourceOutcome(source, success);
  1205→
  1206→      try {
  1207→        switch (source) {
  1208→          case 'brightdata': case 'oxylabs': case 'smartproxy': case 'iproyal': case 'webshare':
  1209→            await residentialProxyManager.releaseProxy(source as ProxyProvider, success);
  1210→            break;
  1211→          case 'rotating':
  1212→            if (this.rotatingSessionsModule?.releaseSession) await this.rotatingSessionsModule.releaseSession(proxyId, success);
  1213→            break;
  1214→          case 'subnet':
  1215→            if (this.subnetExpanderModule?.releaseVirtualIP) await this.subnetExpanderModule.releaseVirtualIP(proxyId, success);
  1216→            break;
  1217→          case 'bulk':
  1218→            if (this.bulkSessions?.releaseSession) await this.bulkSessions.releaseSession(proxyId, success);
  1219→            break;
  1220→          case 'database':
  1221→            await smartIPPool.releaseProxy(proxyId, domain, success, statusCode);
  1222→            break;
  1223→        }
  1224→      } catch (err: any) {
  1225→        logger.warn({ source, proxyId, error: err.message }, 'Failed to release proxy back to source');
  1226→      }
  1227→    }
  1228→
  1229→    if (!success && (statusCode === 403 || statusCode === 429)) {
  1230→      const cooldownMs = statusCode === 429 ? 5 * 60 * 1000 : 10 * 60 * 1000;
  1231→      try {
  1232→        await ipReputationTracker.blacklistProxy(proxyId, domain, `http_${statusCode}`, cooldownMs);
  1233→      } catch {}
  1234→    }
  1235→
  1236→    this.fulfillDemand(domain);
  1237→  }

  1241→  buildSourceOrder(domain: string, tier: string, country: string): MegaPoolSource[] {
  1242→    const defaultOrder: MegaPoolSource[] = [
  1243→      'rotating', 'subnet', 'bulk', 'brightdata', 'oxylabs', 'smartproxy',
  1244→      'iproyal', 'webshare', 'database', 'fusion', 'free', 'tor',
  1245→    ];
  1246→
  1247→    const scoredSources = defaultOrder.map(source => {
  1248→      let score = 0;
  1249→
  1250→      score += SOURCE_QUALITY[source] * 30;
  1251→
  1252→      const health = this.sourceHealthMap.get(source);
  1253→      if (health) {
  1254→        if (health.healthy) {
  1255→          score += 20 + health.successRate * 15;
  1256→        } else {
  1257→          score -= 30;
  1258→        }
  1259→      }
  1260→
  1261→      const cost = SOURCE_COST[source];
  1262→      score += cost === 0 ? 10 : -cost * 0.5;
  1263→
  1264→      if (tier === 'residential' && ['brightdata', 'oxylabs', 'smartproxy', 'rotating', 'subnet'].includes(source)) score += 10;
  1265→      else if (tier === 'datacenter' && ['database', 'webshare', 'free'].includes(source)) score += 10;
  1266→      else if (tier === 'mobile' && ['brightdata', 'oxylabs', 'rotating'].includes(source)) score += 10;
  1267→
  1268→      if (country && country !== 'any') {
  1269→        const providerCountries = this.getProviderCountryAvailability(source);
  1270→        if (providerCountries.includes(country.toUpperCase())) score += 15;
  1271→        else if (providerCountries.length > 0) score -= 5;
  1272→      }
  1273→
  1274→      // Intelligence bonus
  1275→      const intelKey = `${domain}:${source}`;
  1276→      const intel = this.poolIntelligence.get(intelKey);
  1277→      if (intel) {
  1278→        const intelScore = intel.successCount / Math.max(1, intel.successCount + intel.failureCount);
  1279→        score += intelScore * 10;
  1280→      }
  1281→
  1282→      return { source, score };
  1283→    });
  1284→
  1285→    scoredSources.sort((a, b) => b.score - a.score);
  1286→    return scoredSources.map(s => s.source);
  1287→  }

  1861→  private recordSourceOutcome(source: MegaPoolSource, success: boolean): void {
  1862→    const outcomes = this.sourceOutcomes.get(source) || { successes: 0, failures: 0 };
  1863→    if (success) outcomes.successes++;
  1864→    else outcomes.failures++;
  1865→    this.sourceOutcomes.set(source, outcomes);
  1866→
  1867→    // Update source health
  1868→    const health = this.sourceHealthMap.get(source);
  1869→    if (health) {
  1870→      const total = outcomes.successes + outcomes.failures;
  1871→      health.successRate = total > 0 ? outcomes.successes / total : SOURCE_QUALITY[source];
  1872→      health.healthy = health.successRate >= MIN_SOURCE_HEALTH_RATE;
  1873→    }
  1874→  }
