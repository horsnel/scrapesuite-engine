   389→ * Each request is routed through the chain: client → local proxy → proxy1 → proxy2 → target.
   390→ * Enhanced to support WebSocket and HTTP/2 tunneling.
   391→ */
   392→class ChainProxyServer {
   393→  private server: http.Server | null = null;
   394→  private port: number;
   395→  private chain: string[] = [];
   396→  private activeConnections = new Set<net.Socket>();
   397→  private isRunning = false;
   398→  private requestCount = 0;
   399→  private successCount = 0;
   400→  private failureCount = 0;
   401→  private totalLatencyMs = 0;
   402→  private createdAt = Date.now();
   403→
   404→  constructor(port: number) {
   405→    this.port = port;
   406→  }
   407→
   408→  /**
   409→   * Start the chain proxy server.
   410→   */
   411→  start(chain: string[]): Promise<void> {
   412→    return new Promise((resolve, reject) => {
   413→      if (this.isRunning) {
   414→        resolve();
   415→        return;
   416→      }
   417→
   418→      this.chain = chain;
   419→
   420→      this.server = http.createServer((req, res) => {
   421→        this.handleHttpRequest(req, res).catch((err: any) => {
   422→          logger.debug({ error: err.message }, 'Chain proxy HTTP request failed');
   423→          if (!res.headersSent) {
   424→            res.writeHead(502, { 'Content-Type': 'text/plain' });
   425→          }
   426→          res.end('Bad Gateway');
   427→        });
   428→      });
   429→
   430→      this.server.on('connect', (req, clientSocket, head) => {
   431→        this.handleConnect(req, clientSocket as net.Socket, head).catch((err: any) => {
   432→          logger.debug({ error: err.message }, 'Chain proxy CONNECT failed');
   433→          clientSocket.end();
   434→        });
   435→      });
   436→
   437→      this.server.on('connection', (socket) => {
   438→        this.activeConnections.add(socket);
   439→        socket.on('close', () => {
   440→          this.activeConnections.delete(socket);
   441→        });
   442→      });
   443→
   444→      this.server.listen(this.port, '127.0.0.1', () => {
   445→        this.isRunning = true;
   446→        logger.debug({ port: this.port, chainLength: chain.length }, 'Chain proxy server started');
   447→        resolve();
   448→      });
   449→
   450→      this.server.on('error', (err: any) => {
   451→        if (err.code === 'EADDRINUSE') {
   452→          this.port++;
   453→          this.server!.listen(this.port, '127.0.0.1');
   454→        } else {
   455→          reject(err);
   456→        }
   457→      });
   458→    });
   459→  }
   460→
   461→  /**
   462→   * Stop the chain proxy server.
   463→   */
   464→  stop(): Promise<void> {
   465→    return new Promise((resolve) => {
   466→      if (!this.server || !this.isRunning) {
   467→        resolve();
   468→        return;
   469→      }
   470→
   471→      for (const socket of this.activeConnections) {
   472→        socket.destroy();
   473→      }
   474→      this.activeConnections.clear();
   475→
   476→      this.server.close(() => {
   477→        this.isRunning = false;
   478→        this.server = null;
   479→        resolve();
   480→      });
   481→    });
   482→  }
   483→
   484→  getUrl(): string {
   485→    return `http://127.0.0.1:${this.port}`;
   486→  }
   487→
   488→  getIsRunning(): boolean {
   489→    return this.isRunning;
   490→  }
   491→
   492→  getStats(): { requestCount: number; successCount: number; failureCount: number; avgLatencyMs: number; uptimeMs: number } {
   493→    return {
   494→      requestCount: this.requestCount,
   495→      successCount: this.successCount,
   496→      failureCount: this.failureCount,
   497→      avgLatencyMs: this.requestCount > 0 ? Math.round(this.totalLatencyMs / this.requestCount) : 0,
   498→      uptimeMs: Date.now() - this.createdAt,
   499→    };
   500→  }
   501→
   502→  recordSuccess(latencyMs: number): void {
   503→    this.requestCount++;
   504→    this.successCount++;
   505→    this.totalLatencyMs += latencyMs;
   506→  }
   507→
   508→  recordFailure(): void {
   509→    this.requestCount++;
   510→    this.failureCount++;
   511→  }
   512→
   513→  /**
   514→   * Perform a health check on this chain proxy server.
   515→   */
   516→  async healthCheck(): Promise<{ alive: boolean; latencyMs: number }> {
   517→    if (!this.isRunning) return { alive: false, latencyMs: Infinity };
   518→
   519→    const startTime = Date.now();
   520→    try {
   521→      const result = await testProxy(this.getUrl(), BARRIER_TEST_URL, 5000);
   522→      return {
   523→        alive: result.working,
   524→        latencyMs: Date.now() - startTime,
   525→      };
   526→    } catch {
   527→      return { alive: false, latencyMs: Infinity };
   528→    }
   529→  }
   530→
   531→  private async handleHttpRequest(req: http.IncomingMessage, res: http.ServerResponse): Promise<void> {
   532→    const targetUrl = req.url || '';
   533→    if (!targetUrl.startsWith('http')) {
   534→      res.writeHead(400, { 'Content-Type': 'text/plain' });
   535→      res.end('Bad Request -- URL must start with http://');
   536→      return;
   537→    }
   538→
   539→    const firstProxy = this.chain[0];
   540→    if (!firstProxy) {
   541→      res.writeHead(502, { 'Content-Type': 'text/plain' });
   542→      res.end('No proxy in chain');
   543→      return;
   544→    }
   545→
   546→    try {
   547→      const parsedTarget = new URL(targetUrl);
   548→      const proxyOptions = this.buildProxyRequestOptions(parsedTarget, req);
   549→
   550→      const proxyReq = http.request(proxyOptions, (proxyRes) => {
   551→        res.writeHead(proxyRes.statusCode || 502, proxyRes.headers);
   552→        proxyRes.pipe(res);
   553→      });
   554→
   555→      proxyReq.on('error', (err: any) => {
   556→        logger.debug({ error: err.message, url: targetUrl }, 'Proxy chain request error');
   557→        if (!res.headersSent) {
   558→          res.writeHead(502, { 'Content-Type': 'text/plain' });
   559→        }
   560→        res.end('Bad Gateway');
   561→      });
   562→
   563→      if (req.method === 'POST' || req.method === 'PUT' || req.method === 'PATCH') {
   564→        req.pipe(proxyReq);
   565→      } else {
   566→        proxyReq.end();
   567→      }
   568→    } catch (err: any) {
   569→      if (!res.headersSent) {
   570→        res.writeHead(500, { 'Content-Type': 'text/plain' });
   571→      }
   572→      res.end('Internal Server Error');
   573→    }
   574→  }
   575→
   576→  private async handleConnect(
   577→    req: http.IncomingMessage,
   578→    clientSocket: net.Socket,
   579→    head: Buffer,
   580→  ): Promise<void> {
   581→    const [hostname, portStr] = (req.url || '').split(':');
   582→    const port = parseInt(portStr, 10) || 443;
   583→
   584→    if (!hostname) {
   585→      clientSocket.end();
   586→      return;
   587→    }
   588→
   589→    const firstProxy = this.chain[0];
   590→    if (!firstProxy) {
   591→      clientSocket.end();
   592→      return;
   593→    }
   594→
   595→    try {
   596→      const parsedProxy = new URL(firstProxy);
   597→
   598→      const connectReq = http.request({
   599→        method: 'CONNECT',
   600→        host: parsedProxy.hostname,
   601→        port: parseInt(parsedProxy.port, 10) || 8080,
   602→        path: `${hostname}:${port}`,
   603→        headers: {
   604→          Host: `${hostname}:${port}`,
   605→          ...(parsedProxy.username ? {
   606→            'Proxy-Authorization': 'Basic ' + Buffer.from(`${parsedProxy.username}:${parsedProxy.password}`).toString('base64'),
   607→          } : {}),
   608→        },
   609→      });
   610→
   611→      connectReq.on('connect', (proxyRes, proxySocket) => {
   612→        if (proxyRes.statusCode === 200) {
   613→          if (this.chain.length > 1) {
   614→            this.tunnelThroughRemainingProxies(proxySocket, hostname, port, 1)
   615→              .then((finalSocket) => {
   616→                if (finalSocket) {
   617→                  clientSocket.write('HTTP/1.1 200 Connection Established\r\n\r\n');
   618→                  if (head.length > 0) finalSocket.write(head);
   619→                  clientSocket.pipe(finalSocket);
   620→                  finalSocket.pipe(clientSocket);
   621→                } else {
   622→                  clientSocket.end();
   623→                }
   624→              })
   625→              .catch(() => clientSocket.end());
   626→          } else {
   627→            clientSocket.write('HTTP/1.1 200 Connection Established\r\n\r\n');
   628→            if (head.length > 0) proxySocket.write(head);
   629→            clientSocket.pipe(proxySocket);
   630→            proxySocket.pipe(clientSocket);
   631→          }
   632→        } else {
   633→          clientSocket.end();
   634→        }
   635→      });
   636→
   637→      connectReq.on('error', () => {
   638→        clientSocket.end();
   639→      });
   640→
   641→      connectReq.end();
   642→    } catch {
   643→      clientSocket.end();
   644→    }
   645→  }
   646→
   647→  private async tunnelThroughRemainingProxies(
   648→    currentSocket: net.Socket,
   649→    hostname: string,
   650→    port: number,
   651→    chainIndex: number,
   652→  ): Promise<net.Socket | null> {
   653→    if (chainIndex >= this.chain.length) {
   654→      return currentSocket;
   655→    }
   656→
   657→    const proxyUrl = this.chain[chainIndex];
   658→    if (!proxyUrl) return currentSocket;
   659→
   660→    return new Promise((resolve) => {
   661→      try {
   662→        const parsedProxy = new URL(proxyUrl);
   663→
   664→        const connectData = [
   665→          `CONNECT ${hostname}:${port} HTTP/1.1`,
   666→          `Host: ${hostname}:${port}`,
   667→          ...(parsedProxy.username ? [
   668→            `Proxy-Authorization: Basic ${Buffer.from(`${parsedProxy.username}:${parsedProxy.password}`).toString('base64')}`,
   669→          ] : []),
   670→          '',
   671→          '',
   672→        ].join('\r\n');
   673→
   674→        currentSocket.write(connectData);
   675→
   676→        let responseBuffer = '';
   677→        const onData = (data: Buffer) => {
   678→          responseBuffer += data.toString();
   679→
   680→          if (responseBuffer.includes('\r\n\r\n')) {
   681→            currentSocket.removeListener('data', onData);
   682→
   683→            if (responseBuffer.includes('200')) {
   684→              if (chainIndex + 1 < this.chain.length) {
   685→                this.tunnelThroughRemainingProxies(currentSocket, hostname, port, chainIndex + 1)
   686→                  .then(resolve);
   687→              } else {
   688→                resolve(currentSocket);
   689→              }
   690→            } else {
   691→              resolve(null);
   692→            }
   693→          }
   694→        };
   695→
   696→        currentSocket.on('data', onData);
   697→        currentSocket.on('error', () => resolve(null));
   698→
   699→        setTimeout(() => {
   700→          currentSocket.removeListener('data', onData);
   701→          resolve(null);
   702→        }, 5_000);
   703→      } catch {
   704→        resolve(null);
   705→      }
   706→    });
   707→  }
   708→
   709→  private buildProxyRequestOptions(targetUrl: URL, req: http.IncomingMessage): http.RequestOptions {
   710→    const firstProxy = this.chain[0];
   711→    const parsedProxy = new URL(firstProxy);
   712→
   713→    return {
   714→      method: req.method,
   715→      hostname: parsedProxy.hostname,
   716→      port: parseInt(parsedProxy.port, 10) || 8080,
   717→      path: targetUrl.toString(),
   718→      headers: {
   719→        ...req.headers,
   720→        Host: targetUrl.host,
   721→        ...(parsedProxy.username ? {
   722→          'Proxy-Authorization': 'Basic ' + Buffer.from(`${parsedProxy.username}:${parsedProxy.password}`).toString('base64'),
   723→        } : {}),
   724→      },
   725→    };
   726→  }
   727→}

   846→  async runHealthCheck(): Promise<TunnelHealthReport[]> {

   848→
   849→    const reports: TunnelHealthReport[] = [];
   850→    const checkPromises: Promise<TunnelHealthReport>[] = [];
   851→
   852→    for (const [chainId, chain] of this.chains) {

   872→        const now = Date.now();
   873→        chain.lastHealthCheckAt = now;
   874→
   875→        let status: 'healthy' | 'degraded' | 'failed';
   876→        if (health.alive && health.latencyMs < 5000) {
   877→          status = 'healthy';
   878→          chain.consecutiveHealthFailures = 0;
   879→        } else if (health.alive) {
   880→          status = 'degraded';
   881→          chain.consecutiveHealthFailures = 0;
   882→        } else {
   883→          status = 'failed';
   884→          chain.consecutiveHealthFailures++;
   885→        }
   886→
   887→        chain.healthStatus = status;
   888→
   889→        // Test individual hops
   890→        let hopsAlive = chain.hops.length;
   891→        const hopTestPromises = chain.hops.map(async (hop) => {
   892→          try {

   896→            return false;

   901→        hopsAlive = hopResults.filter(r => r.status === 'fulfilled' && r.value).length;
   902→
   903→        const needsRepair = status === 'failed' || (status === 'degraded' && chain.consecutiveHealthFailures >= 2) || hopsAlive < chain.hops.length;
   904→
   905→        if (needsRepair) {
   906→          this.scheduleAutoRepair(chainId);
   907→        }
   908→
   909→        return {
   910→          chainId,
   911→          status,
   912→          latencyMs: health.latencyMs,
   913→          lastCheckedAt: now,
   914→          hopsAlive,
   915→          hopsTotal: chain.hops.length,
   916→          needsRepair,
   917→        };

   919→    }
   920→
   921→    const settled = await Promise.allSettled(checkPromises);
   922→    for (const result of settled) {
   923→      if (result.status === 'fulfilled') {
   924→        reports.push(result.value);
   925→      }
   926→    }

   929→

   931→    this.stats.activeChains = Array.from(this.chains.values()).filter(c => c.isActive && c.healthStatus !== 'failed').length;
   932→
   933→    return reports;
   934→  }

   936→  /**
   937→   * Schedule an auto-repair for a failing chain.
   938→   */
   939→  private scheduleAutoRepair(chainId: string): void {
   940→    const retryCount = this.pendingRepairs.get(chainId) || 0;
   941→    if (retryCount >= AUTO_REPAIR_MAX_RETRIES) {
   942→      logger.warn({ chainId, retries: retryCount }, 'Auto-repair max retries exceeded -- decommissioning chain');
   943→      this.decommissionChain(chainId);
   944→      return;
   945→    }
   946→
   947→    this.pendingRepairs.set(chainId, retryCount + 1);
   948→
   949→    // Fire and forget repair
   950→    this.autoRepairChain(chainId).catch((err: any) => {
   951→      logger.debug({ error: err.message, chainId }, 'Auto-repair failed');
   952→    });
   953→  }

   955→  /**
   956→   * Auto-repair a failing chain by rebuilding hops that have gone dead.
   957→   */
   958→  private async autoRepairChain(chainId: string): Promise<boolean> {
   959→    const chain = this.chains.get(chainId);
   960→    if (!chain || !chain.isActive) return false;
   961→
   962→    logger.info({ chainId, currentStatus: chain.healthStatus, depth: chain.chainDepth }, 'Auto-repairing chain');
   963→
   964→    // Test each hop and identify dead ones
   965→    const hopTestPromises = chain.hops.map(async (hop, idx) => {
   966→      try {
   967→        const result = await testProxy(hop.proxyUrl, BARRIER_TEST_URL, 5000);
   968→        return { idx, alive: result.working, hop };
   969→      } catch {
   970→        return { idx, alive: false, hop };
   971→      }
   972→    });
   973→
   974→    const hopResults = await Promise.allSettled(hopTestPromises);
   975→    const deadHops: number[] = [];
   976→
   977→    for (const result of hopResults) {
   978→      if (result.status === 'fulfilled' && !result.value.alive) {
   979→        deadHops.push(result.value.idx);
   980→      }
   981→    }
   982→
   983→    if (deadHops.length === 0) {
   984→      // All hops are actually alive -- mark as healthy
   985→      chain.healthStatus = 'healthy';
   986→      chain.consecutiveHealthFailures = 0;
   987→      this.pendingRepairs.delete(chainId);
   988→      return true;
   989→    }
   990→
   991→    if (deadHops.length === chain.hops.length) {
   992→      // All hops dead -- can't repair, decommission
   993→      this.decommissionChain(chainId);
   994→      return false;
   995→    }
   996→
   997→    // Replace dead hops with new proxies
   998→    const replacementPromises = deadHops.map(async (hopIdx) => {
   999→      const deadHop = chain.hops[hopIdx];
  1000→      const isExit = hopIdx === chain.hops.length - 1;
  1001→
  1002→      let replacement: any;
  1003→      if (isExit) {
  1004→        replacement = await db.proxy.findFirst({
  1005→          where: {
  1006→            country: deadHop.country,
  1007→            retired: false,
  1008→            successRate: { gte: 0.5 },
  1009→          },
  1010→          orderBy: { successRate: 'desc' },
  1011→        });
  1012→      } else {
  1013→        replacement = await db.proxy.findFirst({

  1094→   * Select the best tunnel for a given target country based on
  1095→   * latency, success rate, and chain depth.
  1096→   */
  1097→  selectBestTunnel(targetCountry: string): SmartTunnelCandidate | null {
  1098→    const candidates: SmartTunnelCandidate[] = [];
  1099→
  1100→    for (const [chainId, chain] of this.chains) {
  1101→      if (!chain.isActive || chain.healthStatus === 'failed') continue;
  1102→
  1103→      const server = this.chainServers.get(chainId);
  1104→      if (!server || !server.getIsRunning()) continue;
  1105→
  1106→      // Score based on: latency (40%), success rate (30%), country match (20%), chain depth (10%)
  1107→      const serverStats = server.getStats();
  1108→      const successRate = serverStats.requestCount > 0 ? serverStats.successCount / serverStats.requestCount : 0.5;
  1109→      const latencyScore = serverStats.avgLatencyMs > 0 ? Math.max(0, 1 - serverStats.avgLatencyMs / 10000) : 0.5;
  1110→      const countryMatch = chain.exitCountry === targetCountry.toUpperCase() ? 1 : 0.3;
  1111→      const depthScore = 1 - (chain.chainDepth - MIN_CHAIN_DEPTH) / (MAX_CHAIN_DEPTH - MIN_CHAIN_DEPTH);
  1112→
  1113→      const score = (latencyScore * 0.4) + (successRate * 0.3) + (countryMatch * 0.2) + (depthScore * 0.1);
  1114→
  1115→      candidates.push({
  1116→        chainId,
  1117→        exitUrl: chain.exitUrl,
  1118→        exitCountry: chain.exitCountry,
  1119→        score,
  1120→        latencyMs: serverStats.avgLatencyMs,
  1121→        successRate,
  1122→        chainDepth: chain.chainDepth,
  1123→        protocol: chain.protocol,
  1124→      });
  1125→    }
  1126→
  1127→    if (candidates.length === 0) return null;
  1128→
  1129→    candidates.sort((a, b) => b.score - a.score);
  1130→    return candidates[0];
  1131→  }
